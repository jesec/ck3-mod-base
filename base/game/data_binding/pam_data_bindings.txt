	description = "Can this dead Faith be researched by the character?"
	definition = "KnowsDeadFaith(ScopeCharacter,Faith)"
	replace_with = "GetScriptedGui( 'pam_knows_dead_faith' ).IsValid( GuiScope.SetRoot( ScopeCharacter.MakeScope ).AddScope( 'faith', Faith.MakeScope ).End )"
}
