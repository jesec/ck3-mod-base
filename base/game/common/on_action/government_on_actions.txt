

# root - Character who changed government
# scope:old_government - Character's old government type
on_government_change = {
	effect = {
		# GDP Invalidate adventurer and merchant contracts on government change
		if = {
			limit = {
				scope:old_government = government_type:merchant_government
			}
			merchant_inspector_clear_specialized_market_effect = yes
			gdp_invalidate_merchant_task_contracts_on_government_change_effect = yes
			gdp_update_usury_reputation_effect = yes
		}
		if = {
			limit = {
				government_has_flag = government_is_merchant
				NOT = { scope:old_government = government_type:merchant_government }
			}
			set_variable = {
				name = gdp_recently_became_merchant_cooldown
				years = 5
			}
		}
		if = {
			limit = {
				scope:old_government = government_type:landless_adventurer_government
			}
			gdp_invalidate_landless_adventurer_task_contracts_on_government_change_effect = yes
		}
		# Otherwise new adventurers sit in Undecided until the situation's next periodic group refresh
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
				exists = situation:guelph_ghibelline
			}
			recalculate_participant_group = situation:guelph_ghibelline
		}

		# No longer Wanua
		if = {
			limit = {
					scope:old_government = government_type:wanua_government
					NOT = { government_has_flag = government_is_wanua }
			}
			every_held_title = {
				   limit = {
							tier = tier_barony
							exists = title_province
				   }
				   title_province = {
							# Remove Wanua-only resource buildings so they do not become worthless if you stop being a Wanua.
							remove_wanua_resource_barter_buildings_effect = yes
				   }
			}
		}
		christian_clergy_learn_liturgical_language_effect = yes

		# MPO remove nomad title
		if = {
			limit = {
				scope:old_government = government_type:nomad_government
			}
			every_held_title = {
				limit = { is_nomad_title = yes }
				root = {
					destroy_title = prev
				}
			}
		}
		# MPO confederation
		if = {
			limit = {
				is_confederation_member = yes
				# A house-based confederation is left by the house, not by individual members
				confederation = { is_house_based = no }
				NOR = {
					government_has_flag = government_is_nomadic
					government_has_flag = government_is_tribal
					confederation = {
						OR = {
							is_title_based = yes
							has_variable = confederation_ignore_government
							confederation_type = confederation_type:lombard_league
						}
					}
				}
			}
			confederation = {
				remove_confederation_member = root
			}
		}
		# ACH coronation
		if = {
			limit = {
				coronation_trigger = yes
				NOT = { has_game_rule = coronation_laws_off }
			}
			if = {
				limit = {
					has_variable = crowned_emperor_var
					NOT = { has_realm_law = crowned_emperor }
				}
				add_realm_law = crowned_emperor
			}
			else_if = {
				limit = {
					has_variable = crowned_king_var
					NOT = { has_realm_law = crowned_king }
				}
				add_realm_law = crowned_king
			}
		}
		# TGP Mandala clean modifier
		if = {
			limit = { scope:old_government = government_type:mandala_government }
			#Remove Aspect modifiers/variables
			if = {
				limit = { has_character_modifier = mandala_increments_of_peace_modifier }
				remove_character_modifier = mandala_increments_of_peace_modifier
			}
			if = {
				limit = { has_character_modifier = mandala_creator_modifier }
				remove_character_modifier = mandala_creator_modifier
			}
			remove_variable = mandala_house_power_accumulated_increments_of_peace
			remove_variable = mandala_house_power_accumulated_creator
			#Remove max piety cap modifiers/variables
			if = {
				limit = { has_character_modifier = budding_divinity_modifier }
				remove_character_modifier = budding_divinity_modifier
			}
			if = {
				limit = { has_character_modifier = budding_deity_modifier }
				remove_character_modifier = budding_deity_modifier
			}
			if = {
				limit = { has_character_modifier = budding_godhood_modifier }
				remove_character_modifier = budding_godhood_modifier
			}
			remove_variable = not_subject_to_succession_trials
			remove_variable = tribute_mission_grace
		}
		if = {
			limit = {
				government_has_flag = government_is_mandala
				is_house_head = yes
			}
			#Set the house aspect to default, please
			house = {
				set_house_aspiration = { type = no_aspect }
			}
		}
		# TGP clean ministry
		if = {
			limit = {
				has_title = title:h_china
				scope:old_government = { government_type_has_flag = government_is_celestial }
				tgp_has_access_to_ministry_trigger = no
			}
			every_vassal = {
				limit = { tgp_has_minister_title = yes }
				every_held_title = {
					limit = { tgp_is_minister_title = yes }
					prev = { destroy_title = prev }
				}
			}
		}
		# PAM align theocracy government - the change re-enters this on_action once, then stops
		pam_align_theocracy_with_faith_government_effect = yes
		pam_strip_illegal_cardinalate_effect = yes

		# Merchant/republic swap
		if = {
			limit = {
				government_has_flag = government_is_merchant
				scope:old_government = government_type:republic_government
				any_held_title = {
					is_noble_family_title = yes
					save_temporary_scope_as = title_temp
				}
			}
			save_scope_as = new_merchant
			scope:title_temp ?= { # silently fails if already present, so we can try every time
				add_to_global_variable_list = {
					name = merchants_tally
					target = this
				}
				set_title_name_dynamic = merchant_name_001
			}
		}
		else_if = {
			limit = {
				government_has_flag = government_is_republic
				scope:old_government = government_type:merchant_government
				any_held_title = {
					is_noble_family_title = yes
					save_temporary_scope_as = title_temp
				}
			}
			save_scope_as = new_patrician
			scope:title_temp ?= { # silently fails, so we can try every time
				remove_list_global_variable = {
					name = merchants_tally
					target = this
				}
				set_title_name_dynamic = patrician_family_name_001
			}
		}

		# GDP achievement: Wearer of Many Hats
		ep5_09_wearer_of_many_hats_achievement_record_effect = yes
	}
}

# quarterly_playable_pulse runs this because on_government_change misses silent government swaps
ep5_09_wearer_of_many_hats_achievement_pulse = {
	trigger = {
		is_ai = no
	}
	effect = {
		ep5_09_wearer_of_many_hats_achievement_record_effect = yes
	}
}
