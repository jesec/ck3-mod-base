# Republic Popular Opinion: Annual City Climate Check
# Runs from yearly_playable_pulse
# Each tier on_action can be called directly from the console to test it

gdp_republic_popular_opinion_pulse = {
	trigger = {
		landed_republican_trigger = yes
	}

	effect = {
		# Negative branch
		if = {
			limit = {
				capital_county = { popular_opinion < 0 }
			}
			random = {
				chance = {
					value = 20
					if = {
						limit = { capital_county = { county_control < county_control_baseline } }
						add = 10
					}
					if = {
						limit = { capital_county = { county_control < county_control_relative_20 } }
						add = 10
					}
				}

				if = {
					limit = { capital_county = { popular_opinion < -50 } }
					trigger_event = { on_action = gdp_republic_popular_opinion_severe_neg }
				}
				else = {
					trigger_event = { on_action = gdp_republic_popular_opinion_mild_neg }
				}
			}
		}

		# Positive branch
		if = {
			limit = {
				capital_county = { popular_opinion >= 25 }
			}
			random = {
				chance = {
					value = 15
					if = {
						limit = { capital_county = { county_control >= county_control_relative_60 } }
						add = 10
					}
					if = {
						limit = { capital_county = { county_control >= county_control_relative_75 } }
						add = 10
					}
				}

				if = {
					limit = { capital_county = { popular_opinion >= 50 } }
					trigger_event = { on_action = gdp_republic_popular_opinion_strong_pos }
				}
				else = {
					trigger_event = { on_action = gdp_republic_popular_opinion_mild_pos }
				}
			}
		}
	}
}
