### GDP Trade Opportunities
# Challenge Competitor
gdp_challenge_competitor = {
	trigger = {
		can_create_task_contract = { type_name = challenge_competitor }
	}
	
	effect = {
		# Find destination
		random_trade_route = {
			limit = {
				has_trader = yes
				current_capacity < max_capacity
				any_trade_market_on_route = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					gdp_market_competitiveness_contested_trigger = { ROUTE_OWNER = root }
					NOR = {
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_failure_modifier
						}
					}
				}
			}
			save_scope_as = route
			trader ?= { save_scope_as = associate }
			random_trade_market_on_route = {
				limit = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					# prefer having contracts outside capital market
					NOT = { this = root.domicile.domicile_location.barony.trade_market }
					gdp_market_competitiveness_contested_trigger = { ROUTE_OWNER = root }
					NOR = {
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_failure_modifier
						}
					}
				}
				alternative_limit = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					gdp_market_competitiveness_contested_trigger = { ROUTE_OWNER = root }
					NOR = {
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_success_modifier
						}
						has_competitiveness_modifier_for_character = {
							character = root
							modifier = gdp_challenge_competitor_trade_opportunity_failure_modifier
						}
					}
				}
				save_scope_as = market
				trade_market_owner.title_province = { save_scope_as = destination }
				random_character_in_market = {
					limit = {
						gdp_profit_potential_rival_trigger = {
							ROUTE_OWNER = root
							MARKET = scope:market
						}
					}
					save_scope_as = challenge_competitor_target
				}
			}
		}

		# Save useful scopes
		trade_opportunity_task_contract_scopes_effect = yes

		# Create
		create_task_contract = {
			task_contract_type = challenge_competitor
			task_contract_tier = scope:contract_tier
			target = scope:challenge_competitor_target
			location = scope:destination
			task_contract_employer = scope:associate # for UI
			save_scope_as = trade_opportunity
		}
		
		# General post-creation set up
		manage_new_trade_opportunity_contract_effect = yes
	}
}

# Promote Goods
gdp_promote_goods = {
	trigger = {
		can_create_task_contract = { type_name = promote_goods }
	}
	
	effect = {
		# Find destination
		random_trade_route = {
			limit = {
				has_trader = yes
				NOT = { has_variable = promote_goods_trade_opportunity_active }
				save_temporary_scope_as = promote_goods_route_check
				any_trade_market_on_route = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					save_temporary_scope_as = promote_goods_market_check
					scope:promote_goods_route_check = {
						trade_route_selling_any_good_in_market_trigger = { MARKET = scope:promote_goods_market_check }
					}
				}
			}
			save_scope_as = route
			trader ?= { save_scope_as = associate }
			random_trade_market_on_route = {
				limit = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					this = root.domicile.domicile_location.barony.trade_market
					save_temporary_scope_as = promote_goods_market_check
					scope:route = {
						trade_route_selling_any_good_in_market_trigger = { MARKET = scope:promote_goods_market_check }
					}
				}
				alternative_limit = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					save_temporary_scope_as = promote_goods_market_check
					scope:route = {
						trade_route_selling_any_good_in_market_trigger = { MARKET = scope:promote_goods_market_check }
					}
				}
				save_scope_as = market
				trade_market_owner.title_province = { save_scope_as = destination }
			}
		}

		# Save useful scopes
		trade_opportunity_task_contract_scopes_effect = yes

		# Create the contract
		create_task_contract = {
			task_contract_type = promote_goods
			task_contract_tier = scope:contract_tier
			location = scope:destination
			task_contract_employer = scope:associate # for UI
			save_scope_as = trade_opportunity
		}
		
		# Set a cooldown
		add_character_flag = {
			flag = trade_opportunity_cooldown_promote_goods
			years = 10
		}
		
		# General post-creation set up
		manage_new_trade_opportunity_contract_effect = yes
		promote_goods_task_contract_pick_trade_good_effect = yes
	}
}

# Exploit/Alleviate Upheaval
gdp_exploit_upheaval = {
	trigger = {
		can_create_task_contract = { type_name = exploit_upheaval }
	}
	
	effect = {
		# Find destination
		random_trade_route = {
			limit = {
				has_trader = yes
				any_trade_market_on_route = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					exploit_upheaval_market_modifier_trigger = yes
				}
			}
			save_scope_as = route
			trader ?= { save_scope_as = associate }
			random_trade_market_on_route = {
				limit = {
					NOT = { this = root.domicile.domicile_location.barony.trade_market }
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					exploit_upheaval_market_modifier_trigger = yes
				}
				alternative_limit = {
					gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
					exploit_upheaval_market_modifier_trigger = yes
				}
				weight = {
					base = 50
					modifier = { # prefer having contracts outside capital market
						this = root.domicile.domicile_location.barony.trade_market
						add = -100
					}
				}
				save_scope_as = market
				trade_market_owner.title_province = { save_scope_as = destination }
			}
		}
		
		# Save useful scopes
		trade_opportunity_task_contract_scopes_effect = yes
		
		# Create the contract
		create_task_contract = {
			task_contract_type = exploit_upheaval
			task_contract_tier = scope:contract_tier
			location = scope:destination
			task_contract_employer = scope:associate # for UI
			save_scope_as = trade_opportunity
		}
		# Create the other contract
		create_task_contract = {
			task_contract_type = alleviate_upheaval
			task_contract_tier = scope:contract_tier
			location = scope:destination
			task_contract_employer = scope:associate # for UI
			save_scope_as = other_trade_opportunity
		}
		scope:trade_opportunity ?= {
			set_variable = {
				name = paired_contract
				value = scope:other_trade_opportunity
			}
		}
		scope:other_trade_opportunity ?= {
			set_variable = {
				name = paired_contract
				value = scope:trade_opportunity
			}
		}

		# General post-creation set up
		manage_new_trade_opportunity_contract_effect = yes
	}
}

# Build Category Connections
gdp_build_category_connections = {
	trigger = {
		can_create_task_contract = { type_name = build_category_connections }
	}

	effect = {
		save_scope_as = merchant
		# Find destination
		random_trade_route = {
			limit = {
				has_trader = yes
				any_traded_good = {
					any_trade_category = {
						has_trade_good_category_xp_track = yes
						save_temporary_scope_as = category_temp
						"scope:merchant.primary_title.trade_good_category_level(scope:category_temp)" < 55
					}
				}
			}
			weight = {
				base = 1
				weight_up_trade_route_by_traded_good_with_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 5 WEIGHT = 150 }
				weight_up_trade_route_by_traded_good_with_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 25 WEIGHT = 50 }
				weight_up_trade_route_by_traded_good_with_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 50 WEIGHT = 10 }
			}
			save_scope_as = route 
			trader ?= { save_scope_as = associate }
			random_traded_good = {
				limit = {
					any_trade_category = {
						has_trade_good_category_xp_track = yes
						save_temporary_scope_as = category_temp
						"scope:merchant.primary_title.trade_good_category_level(scope:category_temp)" < 55
					}
				}
				weight = {
					base = 1
					weight_up_trade_good_by_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 5 WEIGHT = 150 }
					weight_up_trade_good_by_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 25 WEIGHT = 50 }
					weight_up_trade_good_by_category_xp_below_modifier = { MERCHANT = scope:merchant BELOW = 50 WEIGHT = 10 }
				}
				save_scope_as = good
			}
			scope:good = {
				ordered_trade_category = {
					limit = {
						has_trade_good_category_xp_track = yes
						save_temporary_scope_as = category_temp
						"scope:merchant.primary_title.trade_good_category_level(scope:category_temp)" < 55
					}
					order_by = {
						value = 0
						if = {
							limit = {
								save_temporary_scope_as = category_order_temp
								always = yes
							}
							add = {
								value = "scope:merchant.primary_title.trade_good_category_level(scope:category_order_temp)"
								multiply = -1
							}
						}
					}
					save_scope_as = category
				}
			}
			random_trade_market_on_route = {
				limit = {
					trade_market_production_of_trade_good_need_category = { target = scope:category value > 0 }
				}
				alternative_limit = { always = yes }
				save_scope_as = market
				trade_market_owner.title_province = { save_scope_as = destination }
			}
		}

		# Save useful scopes
		trade_opportunity_task_contract_scopes_effect = yes

		# Create
		create_task_contract = {
			task_contract_type = build_category_connections
			task_contract_tier = scope:contract_tier
			location = scope:destination
			task_contract_employer = scope:associate # for UI
			save_scope_as = trade_opportunity
		}

		# General post-creation set up
		manage_new_trade_opportunity_contract_effect = yes

		scope:trade_opportunity ?= {
			set_variable = {
				name = task_contract_trade_good
				value = scope:good
			}
			set_variable = {
				name = task_contract_trade_category
				value = scope:category
			}
		}
	}
}

# Disrupt Charter
gdp_disrupt_charter = {
	trigger = {
		can_create_task_contract = { type_name = disrupt_charter }
	}

	effect = {
		save_scope_as = merchant
		random_in_global_list = {
			variable = merchants_tally
			limit = {
				holder = {
					disrupt_charter_trade_opportunity_target_trigger = { SCHEME_OWNER = scope:merchant }
				}
			}
			weight = {
				base = 1
				modifier = {
					add = 10
					holder = {
						has_any_bad_relationship_with_character_trigger = { CHARACTER = root }
					}
				}
				modifier = {
					add = 10
					holder.suzerain = scope:merchant.suzerain
				}
				modifier = {
					add = 5
					holder.suzerain.top_liege = scope:merchant.suzerain.top_liege

		random_trade_route = {
			limit = {
				has_trader = yes

				}

# Entrepot Buy/Sell
gdp_entrepot_buy = {
	trigger = {
		OR = {
			can_create_task_contract = { type_name = entrepot_buy }
			can_create_task_contract = { type_name = entrepot_sell }
		}
	}

	effect = {
		save_scope_as = merchant
		# Check both up front, as creating either contract invalidates valid_to_create for the other
		if = {
			limit = { can_create_task_contract = { type_name = entrepot_buy } }
			save_scope_value_as = {
				name = entrepot_buy_allowed
				value = yes
			}
		}
		if = {
			limit = { can_create_task_contract = { type_name = entrepot_sell } }
			save_scope_value_as = {
				name = entrepot_sell_allowed
				value = yes
			}
		}
		# Find a trade route passing through an entrepot, where it is buying or selling goods
		random_trade_route = {
			limit = {
				has_trader = yes
				NOR = {
					has_trade_route_modifier = gdp_entrepot_buy_trade_opportunity_success_modifier
					has_trade_route_modifier = gdp_entrepot_buy_trade_opportunity_success_critical_modifier
					has_trade_route_modifier = gdp_entrepot_buy_trade_opportunity_failure_modifier
					has_trade_route_modifier = gdp_entrepot_sell_trade_opportunity_success_modifier
					has_trade_route_modifier = gdp_entrepot_sell_trade_opportunity_success_critical_modifier
					has_trade_route_modifier = gdp_entrepot_sell_trade_opportunity_failure_modifier
				}
				save_temporary_scope_as = entrepot_route_check
				any_trade_market_on_route = {
					gdp_entrepot_trade_opportunity_market_available_trigger = { MERCHANT = root }
					save_temporary_scope_as = entrepot_market_check
					scope:entrepot_route_check = {
						OR = {
							AND = {
								exists = scope:entrepot_buy_allowed
								trade_route_buying_any_good_in_market_trigger = { MARKET = scope:entrepot_market_check }
							}
							AND = {
								exists = scope:entrepot_sell_allowed
								trade_route_selling_any_good_in_market_trigger = { MARKET = scope:entrepot_market_check }
							}
						}
					}
				}
			}
			save_scope_as = route 
			trader ?= { save_scope_as = associate }
			# Save the market
			random_trade_market_on_route = {
				limit = {
					gdp_entrepot_trade_opportunity_market_available_trigger = { MERCHANT = root }
					save_temporary_scope_as = entrepot_market_check
					scope:route = {
						OR = {
							AND = {
								exists = scope:entrepot_buy_allowed
								trade_route_buying_any_good_in_market_trigger = { MARKET = scope:entrepot_market_check }
							}
							AND = {
								exists = scope:entrepot_sell_allowed
								trade_route_selling_any_good_in_market_trigger = { MARKET = scope:entrepot_market_check }
							}
						}
					}
				}
				save_scope_as = market
				# Save the entrepot as the location
				trade_market_owner.duchy = {
					random_de_jure_county = {
						limit = {
							any_county_province = {
								gdp_entrepot_trade_opportunity_location_available_trigger = { MERCHANT = root }
							}
						}
						random_county_province = {
							limit = {
								gdp_entrepot_trade_opportunity_location_available_trigger = { MERCHANT = root }
							}
							save_scope_as = destination
						}
					}
				}
				# Find a good the route buys here for buy price, preferring one with a surplus
				if = {
					limit = { exists = scope:entrepot_buy_allowed }
					scope:route = {
						random_traded_good = {
							limit = {
								trade_good_bought_by_route_in_market_trigger = {
									ROUTE = scope:route
									MARKET = scope:market
								}
								save_temporary_scope_as = good_temp
								scope:market = {
									any_trade_good_with_surplus = { this = scope:good_temp }
								}
							}
							alternative_limit = {
								trade_good_bought_by_route_in_market_trigger = {
									ROUTE = scope:route
									MARKET = scope:market
								}
							}
							save_scope_as = buy_good
						}
					}
					scope:buy_good ?= {
						random_trade_category = {
							limit = { has_trade_good_category_xp_track = yes }
							alternative_limit = { always = yes }
							save_scope_as = buy_category
						}
					}
				}
				# Find a good the route sells here for sell price, preferring one with a need
				if = {
					limit = { exists = scope:entrepot_sell_allowed }
					scope:route = {
						random_traded_good = {
							limit = {
								trade_good_sold_by_route_in_market_trigger = {
									ROUTE = scope:route
									MARKET = scope:market
								}
								save_temporary_scope_as = good_temp
								scope:market = {
									any_trade_good_with_need = { this = scope:good_temp }
								}
							}
							alternative_limit = {
								trade_good_sold_by_route_in_market_trigger = {
									ROUTE = scope:route
									MARKET = scope:market
								}
							}
							save_scope_as = sell_good
						}
					}
					scope:sell_good ?= {
						random_trade_category = {
							limit = { has_trade_good_category_xp_track = yes }
							alternative_limit = { always = yes }
							save_scope_as = sell_category
						}
					}
				}
			}
		}

		# Save useful scopes
		trade_opportunity_task_contract_scopes_effect = yes

		# Create
		if = {
			limit = { exists = scope:buy_good }
			create_task_contract = {
				task_contract_type = entrepot_buy
				task_contract_tier = scope:contract_tier
				location = scope:destination
				task_contract_employer = scope:associate # for UI
				save_scope_as = entrepot_buy_opportunity
			}
			scope:entrepot_buy_opportunity = {
				save_scope_as = trade_opportunity
				# Special tracking
				set_variable = {
					name = task_contract_trade_good
					value = scope:buy_good
				}
				if = {
					limit = { exists = scope:buy_category }
					set_variable = {
						name = task_contract_trade_category
						value = scope:buy_category
					}
				}
			}
		}
		if = {
			limit = { exists = scope:sell_good }
			create_task_contract = {
				task_contract_type = entrepot_sell
				task_contract_tier = scope:contract_tier
				location = scope:destination
				task_contract_employer = scope:associate # for UI
				save_scope_as = entrepot_sell_opportunity
			}
			scope:entrepot_sell_opportunity = {
				if = {
					limit = { exists = scope:trade_opportunity }
					save_scope_as = other_trade_opportunity
				}
				else = {
					save_scope_as = trade_opportunity
				}
				# Special tracking
				set_variable = {
					name = task_contract_trade_good
					value = scope:sell_good
				}
				if = {
					limit = { exists = scope:sell_category }
					set_variable = {
						name = task_contract_trade_category
						value = scope:sell_category
					}
				}
			}
		}
		# Pair, if both were created
		if = {
			limit = {
				exists = scope:entrepot_buy_opportunity
				exists = scope:entrepot_sell_opportunity
			}
			scope:entrepot_buy_opportunity = {
				set_variable = {
					name = paired_contract
					value = scope:entrepot_sell_opportunity
				}
			}
			scope:entrepot_sell_opportunity = {
				set_variable = {
					name = paired_contract
					value = scope:entrepot_buy_opportunity
				}
			}
		}

		# General post-creation set up
		if = {
			limit = { exists = scope:trade_opportunity }
			manage_new_trade_opportunity_contract_effect = yes
		}
	}
}
