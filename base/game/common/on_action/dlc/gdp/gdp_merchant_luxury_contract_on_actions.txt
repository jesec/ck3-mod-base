#On actions relating to Merchant Luxury Contracts

#############################################
# ON ACTIONS FOR CREATING LUXURY CONTRACTS
#############################################

# Merchant government
spawn_luxury_contract_on_action = {
	trigger = {
		has_realm_law = company_purpose_caravaneer
	}
	# Select a random eligible contract to create
	random_on_action = {
		100 = gdp_menagerie_acquisition
		100 = gdp_find_curio
	}
}

### GDP Luxury Contracts
# Menagerie Acquisition
gdp_menagerie_acquisition = {
	trigger = {
		can_create_task_contract = { type_name = menagerie_acquisition }
	}

	weight_multiplier = {
		base = 151 # half curio, as it covers half as many artifact slots
		modifier = {
			add = {
				value = 0
				every_equipped_character_artifact = {
					limit = { artifact_type = creature }
					add = -50
				}
				min = -150
			}
		}
		modifier = {
			any_character_task_contract = { task_contract_type = menagerie_acquisition }
			factor = 0.5
		}
	}
	
	effect = {
		# Save where we're headed
		luxury_contract_save_destination_effect = { ORIGIN = domicile.domicile_location }
		if = {
			limit = { exists = scope:destination }
			# Save useful scopes
			luxury_contract_task_contract_scopes_effect = yes

			# Create
			create_task_contract = {
				task_contract_type = menagerie_acquisition
				task_contract_tier = scope:contract_tier
				location = scope:destination
				save_scope_as = luxury_contract
			}
			if = {
				limit = { exists = var:target_animal_type }
				scope:luxury_contract ?= {
					set_variable = { name = target_animal_type value = root.var:target_animal_type }
				}
			}
			
			# General post-creation set up
			manage_new_luxury_contract_effect = yes
		}
	}
}

# Find Curio
gdp_find_curio = {
	trigger = {
		can_create_task_contract = { type_name = find_curio }
	}

	weight_multiplier = {
		base = 301 # twice menagerie, as it covers twice as many artifact slots
		modifier = {
			add = {
				value = 0
				every_equipped_character_artifact = {
					limit = {
						OR = {
							artifact_type = curio
							artifact_type = library_item
						}
					}
					add = -50
				}
				min = -300
			}
		}
		modifier = {
			any_character_task_contract = { task_contract_type = find_curio }
			factor = 0.5
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { exists = scope:find_curio }
			}
			save_temporary_scope_as = find_curio
		}
		# Save where we're headed
		luxury_contract_save_destination_effect = { ORIGIN = domicile.domicile_location }
		if = {
			limit = { exists = scope:destination }
			# Save useful scopes
			luxury_contract_task_contract_scopes_effect = yes

			# Create
			create_task_contract = {
				task_contract_type = find_curio
				task_contract_tier = scope:contract_tier
				location = scope:destination
				save_scope_as = luxury_contract
			}
			
			# General post-creation set up
			manage_new_luxury_contract_effect = yes
		}
	}
}
