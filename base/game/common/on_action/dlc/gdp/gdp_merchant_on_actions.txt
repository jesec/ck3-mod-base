# AI merchants short on prestige for their next main building, or for the last tier's cost once maxed, convert part of their gold into prestige
gdp_merchant_ai_prestige_purchase_pulse = {
	trigger = {
		is_ai = yes
		government_has_flag = government_is_merchant
		NOT = { has_character_flag = gdp_merchant_ai_prestige_purchase_cooldown }
		gdp_merchant_ai_prestige_purchase_target_value > 0
		prestige < gdp_merchant_ai_prestige_purchase_target_value
		gold >= gdp_merchant_ai_prestige_purchase_min_gold_value
	}
	effect = {
		save_scope_value_as = {
			name = prestige_purchase_gold
			value = gdp_merchant_ai_prestige_purchase_gold_value
		}
		remove_short_term_gold = scope:prestige_purchase_gold
		add_prestige = scope:prestige_purchase_gold
		add_character_flag = {
			flag = gdp_merchant_ai_prestige_purchase_cooldown
			years = 5
		}
	}
}

gdp_merchant_company_debt_pulse = {
	trigger = {
		government_has_flag = government_is_merchant
		has_gdp_dlc_trigger = yes
		gdp_merchant_company_in_debt_trigger = yes
		NOR = {
			has_character_flag = merchant_company_in_debt
			has_character_flag = merchant_company_debt_grace
		}
	}
	effect = {
		# Acumen waits a year, so a short debt or an immediate re-entry does not apply it.
		add_character_flag = {
			flag = merchant_company_debt_grace
			months = company_debt_grace_flag_months_value
		}
		trigger_event = {
			on_action = gdp_merchant_company_debt_grace_pulse
			months = company_debt_grace_months_value
		}
	}
}

gdp_merchant_company_debt_grace_pulse = {
	effect = {
		remove_character_flag = merchant_company_debt_grace
		if = {
			limit = {
				government_has_flag = government_is_merchant
				has_gdp_dlc_trigger = yes
				gdp_merchant_company_in_debt_trigger = yes
				NOT = { has_character_flag = merchant_company_in_debt }
			}
			add_character_flag = merchant_company_in_debt
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = company_in_debt_title
					left_icon = root
					domicile ?= { add_acumen = company_debt_acumen_loss_value }
				}
			}
			else = {
				domicile ?= { add_acumen = company_debt_acumen_loss_value }
			}
			trigger_event = {
				on_action = gdp_merchant_company_debt_direction_pulse
				months = company_debt_direction_interval_months_value
			}
		}
	}
}

gdp_merchant_company_debt_direction_pulse = {
	trigger = {
		has_character_flag = merchant_company_in_debt
	}
	effect = {
		if = {
			limit = {
				government_has_flag = government_is_merchant
				gdp_merchant_company_in_debt_trigger = yes
			}
			if = {
				limit = { is_ai = no }
				send_interface_message = {
					type = event_generic_bad
					title = company_in_debt_direction_title
					left_icon = root
					house ?= {
						add_unity_value = {
							value = company_debt_direction_loss_value
							character = root
							desc = merchant_unity_company_in_debt_desc
						}
					}
				}
			}
			else = {
				house ?= {
					add_unity_value = {
						value = company_debt_direction_loss_value
						character = root
						desc = merchant_unity_company_in_debt_desc
					}
				}
			}
			trigger_event = {
				on_action = gdp_merchant_company_debt_direction_pulse
				months = company_debt_direction_interval_months_value
			}
		}
		else = {
			remove_character_flag = merchant_company_in_debt
		}
	}
}
