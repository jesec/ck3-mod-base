realm_maintenance_quarterly_pulse = {
	on_actions = {
		high_control_pulse
		county_corruption_escalation_pulse
		low_control_pulse
		in_debt_county_corruption_pulse
	}
}

#Existing corruption may worsen if not addressed
county_corruption_escalation_pulse = {
	trigger = {
		any_held_county = {
			county_control < low_county_control
			has_escalatable_county_corruption_trigger = yes
		}
	}
	effect = {
		every_held_county = {
			limit = {
				county_control < low_county_control
				has_escalatable_county_corruption_trigger = yes
			}
			random = {
				chance = 10
				save_scope_as = county
				holder = {
					save_scope_as = corruption_holder
					trigger_event = {
						id = realm_maintenance.0005
						days = { 10 40 }
					}
				}
			}
		}
	}
}

#Being in debt may add corruption
in_debt_county_corruption_pulse = {
	trigger = {
		is_in_debt = yes
		any_held_county = {
			is_susceptible_to_county_corruption_trigger = yes
		}
	}
	effect = {
		every_held_county = {
			limit = {
				is_susceptible_to_county_corruption_trigger = yes
			}
			save_temporary_scope_as = county
			random = {
				chance = in_debt_county_corruption_gain_chance
				save_scope_as = county
				holder = {
					save_scope_as = corruption_holder
					trigger_event = {
						id = realm_maintenance.0002
						days = { 10 40 }
					}
				}
			}
		}
	}
}

#Low County Control may add corruption
low_control_pulse = {
	effect = {
		every_held_county = {
			limit = {
				county_control < low_county_control
				is_susceptible_to_county_corruption_trigger = yes
			}
			random = {
				chance = county_control_corruption_gain_chance
				save_scope_as = county
				holder = {
					save_scope_as = corruption_holder
					trigger_event = {
						id = realm_maintenance.0003
						days = { 10 40 }
					}
				}
			}
		}
	}
}

#High County Control may remove corruption
high_control_pulse = {
	trigger = {
		gold >= 0
		any_held_county = {
			is_landless_type_title = no
			county_control > high_county_control_limit
			has_county_corruption_trigger = yes
		}
	}
	effect = {
		every_held_county = {
			limit = {
				is_landless_type_title = no
				county_control > high_county_control_limit
				has_county_corruption_trigger = yes
			}
			random = {
				chance = high_control_county_corruption_removal_chance
				save_scope_as = county
				holder = {
					save_scope_as = corruption_holder
					trigger_event = {
						id = realm_maintenance.0004
						days = { 10 40 }
					}
				}
			}
		}
	}
}
