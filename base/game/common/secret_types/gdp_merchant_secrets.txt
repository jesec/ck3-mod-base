secret_disguised_usury = {
	category = civil

	is_valid = {
		secret_disguised_usury_is_valid_trigger = {
			OWNER = scope:secret_owner
			TARGET = scope:secret_target
		}
	}

	is_shunned = {
		secret_disguised_usury_is_shunned_trigger = {
			OWNER = scope:secret_owner
			TARGET = scope:secret_target
		}
	}

	is_criminal = {
		secret_disguised_usury_is_criminal_trigger = {
			OWNER = scope:secret_owner
			TARGET = scope:secret_target
		}
	}

	on_discover = {
		save_scope_as = secret
		if = {
			limit = {
				exists = scope:discoverer
				exists = scope:secret_owner
				exists = scope:secret_target
				scope:discoverer != scope:secret_owner
				scope:secret_target = { is_alive = yes }
			}
			gdp_usury_penance_try_fire_effect = yes
		}
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			send_interface_toast = {
				type = event_secret_bad
				title = gdp_usury_secret_exposed_toast
				left_icon = scope:secret_target
				gdp_usury_lend_at_interest_effect = { BORROWER = scope:secret_target }
			}
			hidden_effect = {
				add_character_flag = gdp_usury_exposed
				gdp_update_usury_reputation_effect = yes
			}
		}
		if = {
			limit = {
				scope:secret_target = { is_alive = yes }
			}
			scope:secret_target = {
				add_opinion = {
					target = scope:secret_owner
					modifier = gdp_usurious_loan_opinion
				}
				every_owned_story = {
					limit = {
						story_type = story_cycle_loan
						var:lender = scope:secret_owner
						var:instrument = flag:disguised
					}
					set_variable = gdp_usury_known
				}
			}
		}
		secret_exposed_notification_effect = yes
	}
}
