	priority = 350
	combine_into_one = yes
	check_create_action = {

			limit = {
				government_has_flag = government_is_merchant
				save_temporary_scope_as = actor
				suzerain = { save_temporary_scope_as = recipient }
				is_character_interaction_valid = {
					interaction = patron_request_interaction
					recipient = suzerain

					AND = { # Competetiveness
						prestige >= merchant_competetiveness_prestige_value

					}
					AND = { # Wedding
						gold >= merchant_marriage_gold_value
						patron_request_marriage_available_trigger = yes
					}
					AND = { # Hunt
						gold >= medium_gold_value
						patron_request_hunt_available_trigger = yes

				important_action_type = action_can_make_patron_request_interaction

	}

		}

action_move_company_nearby_market = {
	priority = 50
	combine_into_one = yes
	check_create_action = {
		if = {
			limit = {
				government_has_flag = government_is_merchant
				NOT = { has_character_flag = merchant_recently_moved_company }
				save_temporary_scope_as = merchant_temp
				scope:merchant_temp.domicile.domicile_location = {
					save_temporary_scope_as = province_temp
					barony = {
						county = { save_temporary_scope_as = county_temp }
						trade_market = {
							save_temporary_scope_as = market_temp
							trade_market_owner.duchy = { save_temporary_scope_as = duchy_temp }
						}
					}
				}
			}
			scope:duchy_temp = {
				# Nearby Market has more prosperous county
				every_title_to_title_neighboring_and_across_water_duchy = {
					limit = { gdp_average_duchy_prosperity_value > scope:duchy_temp.gdp_average_duchy_prosperity_value }
					ordered_de_jure_county = {
						limit = {
							NOT = { title_province.barony.trade_market ?= scope:market_temp }
							OR = {
								prosperity_level > scope:province_temp.county.prosperity_level
								any_county_province = { province_size_level > scope:province_temp.province_size_level }
							}
						}
						order_by = {
							value = 1
							add = prosperity_level
							ordered_county_province = {
								order_by = province_size_level
								add = province_size_level
							}
						}
						title_province.barony.trade_market.trade_market_owner.title_province ?= {
							try_create_important_action = {
								important_action_type = action_move_company_nearby_market
								actor = root
								recipient = this.county.holder
								county = this.county
								province = this
								market = this.barony.trade_market
							}
						}
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:province }
			scope:province = { open_view_data = move_domicile_planner }
		}
		else = {
			open_view_data = move_domicile_planner
		}
	}
}

action_can_merchant_ransom = {
	combine_into_one = yes
	is_dangerous = no
	check_create_action = {
		save_scope_as = merchant
		if = {
			limit = { is_landless_merchant = yes }
			suzerain.top_liege = {
				if = {
					limit = { gdp_can_be_ransomed_by_merchant = yes }
					add_to_list = prisoners
				}
				every_prisoner = {
					limit = { gdp_can_be_ransomed_by_merchant = yes } 
					add_to_list = prisoners
				}
				every_vassal_or_below = {
					if = {
						limit = { gdp_can_be_ransomed_by_merchant = yes }
						add_to_list = prisoners
					}
					every_prisoner = {
						limit = { gdp_can_be_ransomed_by_merchant = yes }
						add_to_list = prisoners
					}
				}
				every_neighboring_and_across_water_realm_owner = {
					if = {
						limit = { gdp_can_be_ransomed_by_merchant = yes }
						add_to_list = prisoners
					}
					every_prisoner = {
						limit = { gdp_can_be_ransomed_by_merchant = yes }
						add_to_list = prisoners
					}
					every_vassal_or_below = {
						if = {
							limit = { gdp_can_be_ransomed_by_merchant = yes }
							add_to_list = prisoners
						}
						every_prisoner = {
							limit = { gdp_can_be_ransomed_by_merchant = yes }
							add_to_list = prisoners
						}
					}
				}
			}
			domicile.domicile_location.duchy = {
				every_de_jure_county = {
					limit = { holder.top_liege = root.suzerain.top_liege }
					holder.top_liege = {
						if = {
							limit = { gdp_can_be_ransomed_by_merchant = yes }
							add_to_list = prisoners
						}
						every_prisoner = {
							limit = { gdp_can_be_ransomed_by_merchant = yes } 
							add_to_list = prisoners
						}
						every_vassal_or_below = {
							if = {
								limit = { gdp_can_be_ransomed_by_merchant = yes }
								add_to_list = prisoners
							}
							every_prisoner = {
								limit = { gdp_can_be_ransomed_by_merchant = yes }
								add_to_list = prisoners
							}
						}
					}
				}
			}
		}
		every_in_list = {
			list = prisoners
			limit = { # We skip definite refusals from ai_accept, since is_character_interaction_potentially_accepted cannot handle multiple options
				character_is_prisoner_of_war = no
				NOR = {
					has_relation_nemesis = imprisoner
					is_at_war_with = imprisoner
					imprisoner = { is_at_war_with = root }
				}
			}
			try_create_important_action = {
				important_action_type = action_can_merchant_ransom
				actor = scope:merchant
				recipient = this.imprisoner
				secondary_recipient = this
			}
		}
	}
	unimportant = {
		OR = {
			gold < scope:secondary_recipient.ransom_cost_value
			scope:secondary_recipient = { has_relation_rival = scope:recipient }
		}
	}
	effect = {
		open_interaction_window = {
			interaction = merchant_ransom_interaction
			actor = scope:actor
			recipient = scope:recipient
			secondary_recipient = scope:secondary_recipient
		}
	}
}

action_commission_merchant = {
	priority = 300
	combine_into_one = yes
	icon = commission_trade_route_interaction
	check_create_action = {
		if = {
			limit = {
				is_landed = yes
				highest_held_title_tier >= tier_duchy
				any_tributary = {
					government_has_flag = government_is_merchant
				}
				NOT = {
					any_tributary = { has_commission_route_with = root }
				}
			}
			every_tributary = {
				limit = {
					government_has_flag = government_is_merchant
					NOR = {
						has_commission_route_with = root
						is_at_war_with = root
					}
					has_fewer_commission_routes_than = max_commission_routes
					save_temporary_scope_as = merchant_temp
					# Don't suggest unless there is a theoretical route that could be created
					root = {
						any_held_trade_market = {
							save_temporary_scope_as = market_temp
							OR = {
								any_trade_good_with_unfulfilled_need = {
									commission_trade_route_good_trigger = {
										COMMISSIONER = root
										MARKET = scope:market_temp
										GOOD = this
										MERCHANT = scope:merchant_temp
									}
								}
								any_trade_category_with_unfulfilled_need = {
									any_trade_good_in_category = {
										commission_trade_route_good_trigger = {
											COMMISSIONER = root
											MARKET = scope:market_temp
											GOOD = this
											MERCHANT = scope:merchant_temp
										}
									}
								}
							}
						}
					}
					root = {
						is_character_interaction_valid = {
							interaction = commission_trade_route_interaction
							recipient = prev
						}
					}
				}
				try_create_important_action = {
					important_action_type = action_commission_merchant
					actor = root
					recipient = this
				}
			}
		}
	}
	unimportant = {
		prestige < gdp_commission_route_cost_value
	}
	effect = {
		open_interaction_window = {
			interaction = commission_trade_route_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
