gdp_league_accession_cohesion_value = {
	value = 1

	if = {
		limit = { scope:title.tier >= tier_kingdom }
		add = 2
	}
	else_if = {
		limit = { scope:title.tier = tier_duchy }
		add = 1
	}

	if = {
		limit = {
			exists = scope:title.holder
			exists = scope:title.holder.capital_province
		}
		add = {
			value = scope:title.holder.capital_province.province_size_level
			multiply = 0.25
		}
	}
}

		add = 2
	}

	if = {
		limit = {

		}

	}

	multiply = -1
}
