displace_gg_ai_score_value = {
	value = 1
	scope:attacker = {
		if = {
			limit = {
				OR = {
					AND = {
						is_guelph_trigger = yes
						situation_guelph_score_percentage > 0.2
					}
					AND = {
						is_ghibelline_trigger = yes
						situation_ghibelline_score_percentage > 0.2
					}
				}
			}
			value = 10
			add = ai_boldness
		}
		if = {
			limit = {
				OR = {
					is_at_war = yes
					scope:defender.confederation ?= { combined_military_strength >= scope:attacker.current_military_strength }
					scope:defender.current_military_strength >= scope:attacker.current_military_strength
				}
			}
			multiply = 0
		}
	}
}

			add = {
				value = situation_guelph_rest_score
				desc = situation_gg_guelph_rest_ruler_score
			}
			#Has pope
			if = {
				limit = {
					any_situation_participant = {
						has_variable = guelph_leader
						AND = {
							any_held_title = { this = title:k_papal_state }
							primary_title = { is_head_of_faith = yes }
						}
					}
				}
				add = {
					value = 250
					desc = "situation_gg_guelph_leader_is_pope_score"
				}
			}
			#Imprisoned Guelph Leader?
			if = {
				limit = {
					any_situation_participant = {
						has_variable = guelph_leader
						is_imprisoned = yes
					}
				}
				subtract = {
					value = 250
					desc = "situation_gg_guelph_leader_imprisoned_score"
				}
			}
		}
	}
}
