merchant_purchase_land_cost_scale_by_era = {
	value = 0
	if = {
		limit = {
			exists = culture
		}
		culture = {
			if = {
				limit = {
					has_cultural_era_or_later = culture_era_late_medieval
				}
				add = 2
			}
			else_if = {
				limit = {
					has_cultural_era_or_later = culture_era_high_medieval
				}
				add = 1.5
			}
			else_if = {
				limit = {
					has_cultural_era_or_later = culture_era_early_medieval
				}
				add = 1
			}
			else = {
				add = 0.5
			}
		}
	}
}

			every_de_jure_county = {
				limit = { ep3_purchase_land_valid_county_trigger = yes }
				add = merchant_purchase_land_county_cost_value
			}
		}

		limit = {
			scope:actor = { has_variable = ep3_grace_to_the_family_var }
		}
		multiply = 0.5
	}
	if = {
		limit = { scope:forgive_loan = yes }
		scope:recipient = {
			ordered_owned_story = {
				limit = {
					story_type = story_cycle_loan
					var:lender = scope:actor
				}
				order_by = var:years_since
				add = {
					value = var:amount_owed
					multiply = -1
				}
			}
		}
	}
	if = {
		limit = { scope:offer_gold = yes }
		multiply = 2
	}
	min = 0
}

				limit = { is_defender = scope:patron_temp }

					value = 0

						limit = { defender_war_score > attacker_war_score }

					}
					else = {
						add = attacker_war_score
						multiply = 0.5
					}
					max = 50
					min = -50
				}
			}
			else = {
				add = {
					value = 0
					if = {
						limit = { attacker_war_score > defender_war_score }
						add = attacker_war_score
						divide = -2
					}
					else = {
						add = defender_war_score
						multiply = 0.5
					}
					max = 50
					min = -50
				}
			}
		}
	}
	min = 50
}

entice_merchants_order_value = {
	value = 0
	add = "opinion(scope:new_patron)"
	if = {
		limit = {
			exists = scope:old_patron
			scope:new_patron != scope:old_patron
		}
		add = {
			value = "opinion(scope:old_patron)"
			multiply = -1
		}
	}
	add = {
		value = "opinion(suzerain)"
		multiply = -1
	}
	# CULTURAL ACCEPTANCE
	if = {
		limit = { culture != scope:new_patron.culture }
		culture = {
			if = {
				limit = {
					cultural_acceptance = { target = scope:new_patron.culture value < 25 }
				}
				add = -30
			}
			else_if = {
				limit = {
					cultural_acceptance = { target = scope:new_patron.culture value < 50 }
					cultural_acceptance = { target = scope:new_patron.culture value >= 25 }
				}
				add = -15
			}
			else_if = {
				limit = {
					cultural_acceptance = { target = scope:new_patron.culture value < 75 }
					cultural_acceptance = { target = scope:new_patron.culture value >= 50 }
				}
				add = 15
			}
			else = { add = 30 }
		}
	}
	# FAITH HOSTILITY
	if = {
		limit = { faith != scope:new_patron.faith}
		scope:new_patron.faith = {
			if = {
				limit = {
					faith_hostility_level = {
						target = prev.faith
						value = faith_hostile_level
					}
				}
				add = -50
			}
			else_if = {
				limit = {
					faith_hostility_level = {
						target = prev.faith
						value = faith_astray_level
					}
				}
				add = -25
			}
		}
		faith = {
			if = {
				limit = {
					faith_hostility_level = {
						target = scope:new_patron.faith
						value = faith_hostile_level
					}
				}
				add = -50
			}
			else_if = {
				limit = {
					faith_hostility_level = {
						target = scope:new_patron.faith
						value = faith_astray_level
					}
				}
				add = -25
			}
		}
	}
	# COMPANY SIZE
	if = {
		limit = {
			scope:new_patron.highest_held_title_tier >= tier_kingdom
			has_realm_law = company_size_small
		}
		add = -75
	}
	else_if = {
		limit = {
			scope:new_patron.highest_held_title_tier >= tier_empire
			OR = {
				has_realm_law = company_size_small
				has_realm_law = company_size_medium
			}
		}
		add = -75
	}
	# DISTANCE
	add = {
		value = "domicile.domicile_location.squared_distance(scope:new_patron.capital_province)"
		multiply = 0.0005
	}
}

starting_acumen_value = {
	value = 0
	owner = {
		if = {
			limit = { has_realm_law = company_size_small }
			add = miniscule_acumen_value
		}
		else_if = {
			limit = { has_realm_law = company_size_medium }
			add = company_size_medium_threshold_value
			if = {
				limit = {
					OR = {
						age >= 50
						AND = {
							age >= 30
							highest_skill = stewardship
						}
					}
				}
				add = minor_acumen_value
			}
		}
		else_if = {
			limit = { has_realm_law = company_size_large }
			add = company_size_large_threshold_value
			if = {
				limit = {
					OR = {
						age >= 50
						AND = {
							age >= 30
							highest_skill = stewardship
						}
					}
				}
				add = medium_acumen_value
			}
		}
		else = { add = company_size_huge_threshold_value }

		if = {
			limit = {
				culture = { has_cultural_tradition = tradition_family_entrepreneurship }
			}
			add = minor_acumen_value
		}
		if = {
			limit = {
				culture = { has_cultural_parameter = fortune_seekers_merchant_propensity }
			}
			add = miniscule_acumen_value
		}
	}
	add = owner.age
	add = owner.stewardship
}
