# Empire Faith Gate -- Dissolve Empire CB
# Allows same-faith rulers to declare war on an excommunicated emperor to
# destroy their empire title. Does NOT transfer the title; the slot opens.

dissolve_empire_war = {
	icon = excommunication
	group = religious
	allow_hostages = no
	ai_only_against_neighbors = yes

	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		rite = {
			rite_has_parameter = excommunication_active
		}
		empire_faith_gate_enabled_trigger = yes
		NOT = { has_trait = excommunicated }
	}

	allowed_against_character = {
		scope:defender = {
			top_liege = this
			any_held_title = { tier >= tier_empire }
			has_trait = excommunicated
			faith = scope:attacker.faith
			rite = { rite_has_parameter = excommunication_active }
			# Grace period blocks only AI attackers; human players can always declare.
			trigger_if = {
				limit = { scope:attacker = { is_ai = yes } }
				efg_contested_empire_grace_period_elapsed_trigger = yes
			}
			trigger_else = {
				always = yes
			}
		}
		scope:attacker = {
			NOR = {
				liege = scope:defender
				has_trait = excommunicated
			}
			trigger_if = {
				limit = {
					is_ai = no
					highest_held_title_tier < tier_empire
				}
				any_neighboring_top_liege_realm_owner = {
					this = scope:defender
				}
			}
		}
	}

	target_de_jure_regions_above = yes

	valid_to_start = {
		always = yes
	}

	on_declaration = {
		on_declared_war = yes
	}

	should_invalidate = {
		OR = {
			scope:defender = {
				NOT = { has_trait = excommunicated }
			}
			scope:defender = {
				NOT = { any_held_title = { tier >= tier_empire } }
			}
		}
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = {
						NOT = { has_trait = excommunicated }
					}
				}
				desc = dissolve_empire_war_invalidated_no_excommunication
			}
			triggered_desc = {
				trigger = {
					scope:defender = {
						NOT = { any_held_title = { tier >= tier_empire } }
					}
				}
				desc = dissolve_empire_war_invalidated_no_empire
			}
			desc = msg_invalidate_war_title
		}
	}

	#Other religious wars have an empty block here as well, i dont mind removing it but it doesnt do anything bad -W
	on_invalidated = {
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = dissolve_empire_war_victory_desc_defender
			}
			desc = dissolve_empire_war_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_victory_effect = yes

		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		scope:attacker = {
			add_prestige = major_prestige_gain
		}

		scope:defender = {
			every_held_title = {
				limit = { tier >= tier_empire }
				save_temporary_scope_as = empire_to_destroy
				scope:defender = { destroy_title = scope:empire_to_destroy }
			}
		}

		scope:defender = {
			add_prestige = {
				value = massive_prestige_loss
			}
			add_piety = {
				value = major_piety_loss
			}
		}

		scope:attacker = {
			if = {
				limit = { exists = faith.religious_head }
				reverse_add_opinion = {
					modifier = pleased_opinion
					target = faith.religious_head
					opinion = 30
				}
			}
		}

		add_truce_attacker_victory_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = dissolve_empire_war_white_peace_desc_defender
			}
			desc = dissolve_empire_war_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_prestige = medium_prestige_loss
			stress_and_fulfillment_impact = {
				ambitious = medium_stress_impact_gain
			}
		}

		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }

		add_truce_white_peace_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = dissolve_empire_war_defeat_desc_attacker
			}
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = dissolve_empire_war_defeat_desc_defender
			}
			desc = dissolve_empire_war_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_defeat_effect = yes

		scope:defender = {
			#Again used in most CBs on defeat even if not Mandala -W
			mandala_peacemaker_perk_serenity_effect = yes
			accolade_defender_war_end_glory_gain_high_effect = yes
		}

		scope:attacker = {
			add_prestige = {
				value = major_prestige_loss
			}
			add_piety = {
				value = medium_piety_loss
			}
		}

		add_truce_attacker_defeat_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	transfer_behavior = transfer

	war_name = "DISSOLVE_EMPIRE_WAR_NAME"

	ai_score_mult = {
		value = 1
		multiply = struggle_wars_prioritise_struggle_targets_value
		if = {
			limit = {
				scope:attacker = {
					empire_faith_gate_enabled_trigger = yes
					has_contested_empire_trigger = yes
					faith_has_existing_empire_holder_trigger = yes
				}
			}
			multiply = 5
		}
	}

	cost = {
		piety = {
			add = {
				value = 200
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
		}
	}

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100

	max_ai_diplo_distance_to_title = 500
}
