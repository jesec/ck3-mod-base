trade_league_commune_cb = {
	icon = individual_county_de_jure_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		gdp_is_assembly_head_trigger = yes
		gdp_is_trade_league_head_trigger = yes
	}

	# Greyed out below the cohesion threshold
	allowed_for_character_display_regardless = {
		gdp_league_can_proclaim_communes_trigger = yes
		custom_tooltip = {
			text = league_commune_cooldown_tt
			NOT = { has_character_flag = trade_league_commune_cooldown }
		}
	}

	allowed_against_character = {
		scope:attacker = {
			ALL_FALSE = {
				top_liege = scope:defender.top_liege
				liege = scope:defender
			}
		}
		scope:defender = {
			NOT = { government_has_flag = government_is_republic }
			trigger_if = {
				limit = { exists = scope:attacker.confederation }
				NOT = { is_member_of_confederation = scope:attacker.confederation }
			}
		}
	}

	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 200
	attacker_score_from_battles_scale = 125
	defender_score_from_battles_scale = 125

	ai_can_target_all_titles = {
		# Only maritime republics should go looking for commune seats far from home
		has_realm_law = republic_type_maritime
	}
	ai_score_mult = {
		value = 1 # Base

		# Prefer same faith counties
		if = {
			limit = {
				scope:attacker.faith = scope:target.faith
			}
			add = 1
		}

		# Prefer same culture counties
		if = {
			limit = {
				scope:attacker.culture = scope:target.culture
			}
			add = 1
		}

		# Prefer counties in markets where we already have a presence
		add = {
			every_in_list = {
				list = target_titles
				duchy = {
					every_de_jure_county = {
						limit = {
							title_province = {
								exists = trade_post_district
								trade_post_owner = {
									OR = {
										this = scope:attacker
										liege = scope:attacker
									}
								}
							}
						}
						add = 0.5
					}
				}
			}
		}

		# A commune needs a town worth the name
		add = {
			every_in_list = {
				list = target_titles
				if = {
					limit = {
						title_province = { province_size_level >= 5 }
					}
					add = 1
				}
			}
		}

		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}

	valid_to_start = {
		scope:target = {
			gdp_league_commune_target_valid_trigger = { LEADER = root }
		}
	}

	should_invalidate = {
		OR = {
			war_should_invalidate_defender_lost_target_lands_trigger = yes
			# Invalidate if the county can no longer seat a republic
			NOT = {
				any_in_list = {
					list = target_titles
					AND = {
						holder ?= {
							NOT = { government_has_flag = government_is_republic }
						}
						title_province ?= {
							OR = {
								has_holding_type = city_holding
								has_holding_type = castle_holding
							}
						}
					}
				}
			}
			# Invalidate if the attacker is no longer assembly or league head
			scope:attacker = {
				OR = {
					gdp_is_assembly_head_trigger = no
					gdp_is_trade_league_head_trigger = no
				}
			}
		}
	}

	on_invalidated_desc = msg_trade_league_commune_war_invalidated_message
	on_invalidated = {
		# Refund cooldown if the war never resolved
		scope:attacker = {
			remove_character_flag = trade_league_commune_cooldown
		}
	}

	cost = {
		prestige = {
			value = 0
			add = {
				value = 150
				desc = CB_BASE_COST
			}
			add = {
				value = 5
				multiply = scope:attacker.sub_realm_size
				desc = CB_REALM_SIZE_COST
				max = 200
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes

		# Cooldown covers victory, defeat, and white peace
		scope:attacker = {
			custom_tooltip = {
				text = league_commune_cooldown_declaration_tt
				add_character_flag = {
					flag = trade_league_commune_cooldown
					years = gdp_league_commune_cooldown_years_value
				}
			}
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = trade_league_commune_cb_victory_desc_attacker
			}
			desc = trade_league_commune_cb_victory_desc
		}
	}

	on_victory = {
		on_victory_effect = yes

		# Only ever one county, since the titles are mutually exclusive
		every_in_list = {
			list = target_titles
			gdp_trade_league_proclaim_commune_effect = {
				COUNTY = this
				LEAGUE_HEAD = scope:attacker
			}
		}

		scope:attacker = {
			add_prestige_experience = medium_prestige_value
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		on_white_peace_effect = yes
		# Prestige loss for the attacker
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker # Not important as the scales are identical
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = county_conquest_cb_defeat_desc_defender
			}
			desc = county_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		on_defeat_effect = yes

		scope:defender = {
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		# Prestige loss for the attacker, and the failure is felt across the league
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss

			confederation ?= {
				change_cohesion = gdp_league_failed_commune_cohesion_value
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "TRADE_LEAGUE_COMMUNE_WAR_NAME"
	war_name_base = "TRADE_LEAGUE_COMMUNE_WAR_NAME_BASE"
	cb_name = "TRADE_LEAGUE_COMMUNE_CB_NAME"

	interface_priority = 94

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

trade_county_conquest_cb = {
	icon = trade_county_conquest
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		gdp_is_assembly_head_trigger = yes
	}

	allowed_for_character_display_regardless = {
	}

	allowed_against_character = {
		scope:attacker = {
			ALL_FALSE = {
				top_liege = scope:defender.top_liege
				liege = scope:defender
			}
		}
		scope:defender = {
			any_sub_realm_county = {
				title_province = {
					trigger_if = {
						limit = {
							exists = trade_post_district
							exists = trade_post_owner
						}
						OR = {
							trade_post_owner = scope:attacker
							trade_post_owner.liege = scope:attacker
						}
					}
					trigger_else = {
						always = no
					}
				}
			}
		}
	}
	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder
	
	attacker_score_from_occupation_scale = 200
	attacker_score_from_battles_scale = 125
	defender_score_from_battles_scale = 125

	ai_can_target_all_titles = {
		# Only maritime republics should actively take land from far away
		has_realm_law = republic_type_maritime
	}
	ai_score_mult = {
		value = 1 # Base

		# Prefer same faith counties
		add = {
			every_in_list = {
				list = target_titles
				if = {
					limit = { faith = scope:attacker.faith }
					add = 1
				}
			}
		}

		# Prefer same culture counties
		add = {
			every_in_list = {
				list = target_titles
				if = {
					limit = { culture = scope:attacker.culture }
					add = 1
				}
			}
		}

		# Prefer counties in markets where we have a larger presence
		add = {
			every_in_list = {
				list = target_titles
				duchy = {
					every_de_jure_county = {
						limit = {
							title_province = {
								exists = trade_post_district
								trade_post_owner = {
									OR = {
										this = scope:attacker
										liege = scope:attacker
									}
								}
							}
						}
						add = 0.5
					}
				}
			}
		}
		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			OR = {
				is_neighbor_to_realm = root
				is_coastal_county = yes
			}
			title_province = {
				exists = trade_post_district
				trade_post_owner ?= {
					OR = {
						this = root
						liege = root
					}
				}
			}
		}
	}

	should_invalidate = {
		war_should_invalidate_defender_lost_target_lands_trigger = yes
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	cost = {
		prestige = {
			value = 0
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			add = {
				value = 5
				multiply = scope:attacker.sub_realm_size
				desc = CB_REALM_SIZE_COST
				max = 200
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
	}

	on_victory = {
		on_victory_aggressive_war_effect = yes

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		#As this is just a county conquest, simply grab the designated county.
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}

		resolve_title_and_vassal_change = scope:change

		# Prestige Progress for the Attacker
		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = minor_prestige_value
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		on_white_peace_effect = yes
		# Prestige loss for the attacker
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker # Not important as the scales are identical
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = county_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = county_conquest_cb_defeat_desc_attacker
			}
			desc = county_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		on_defeat_aggressive_war_effect = yes

		scope:defender = {
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		# Prestige loss for the attacker
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	defender_allies_inherit = yes
