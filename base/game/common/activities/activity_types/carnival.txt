activity_carnival = {
	is_shown = {
		has_gdp_dlc_trigger = yes
		is_landed = yes
		government_has_flag = government_is_republic
		culture = { has_cultural_parameter = unlock_carnival_activity }
		highest_held_title_tier > tier_barony
		trigger_if = {
			limit = {
				is_ai = yes
			}
			is_at_war = no
			short_term_gold >= carnival_activity_base_cost
		}
	}
	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_carnival }
		is_available_adult = yes
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_capable_adult = yes
			is_imprisoned = no
			is_landed_or_landless_administrative = yes
			NOT = { is_incapable = yes }
		}
		# If no one shows up
		trigger_if = {
			limit = { is_current_phase_active = yes }
			has_attending_activity_guests = yes
		}
	}

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes
		invalidation_unlanded_effect = yes
		if = { # No one shows up
			limit = {
				has_attending_activity_guests = no
			}
			scope:activity = {
				activity_type = { save_scope_as = activity_type }
				activity_location = { save_scope_as = location }
			}
			scope:host = { trigger_event = activity_system.0100 }
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes
	}

	is_single_location = yes

	is_location_valid = {
		has_holding = yes
		this = scope:host.capital_province
	}

	max_province_icons = 1 ### only shows up in capital

	cost = {
		treasury = {
			add = {
				add = {
					value = carnival_activity_base_cost #Base level
					desc = carnival_base_cost
				}
				add = {
					value = carnival_activity_base_cost
					multiply = activity_cost_scale_by_era
					subtract = carnival_activity_base_cost
					desc = activity_cost_scale_by_era_desc
				}
			}
		}
	}

	ui_predicted_cost = {
		# All costs are balanced on County/Early Era starting point
		treasury = {
			add = { # Base cost, scaled by era
				value = carnival_activity_base_cost
				multiply = activity_cost_scale_by_era
			}
			add = { # Mid-range (Normal) option cost
				value = 7
				add = {
					value = 7
					multiply = activity_cost_scale_by_tier
					subtract = 7
				}
				add = {
					value = 7
					multiply = activity_cost_scale_by_era
					subtract = 7
				}
			}
			##Make it a multiple of 5 (rounded up)
			divide = 5
			ceiling = yes
			multiply = 5
		}
	}

	cooldown = { years = standard_carnival_cooldown_time }

	ai_will_do = {
		add = {
			value = 50

		}

		if = {

				NOR = {

			limit = {

		if = {

		}
		
		if = {

		}

		if = {
			limit = {
				has_trait = lifestyle_reveler
			}
			add = 30
		}
		# Plague
		if = {
        	limit = {
        	    any_held_title = {
					tier = tier_county
        	        any_county_province = {
        	            any_province_epidemic = {
        	                count >= 1
        	                outbreak_intensity < apocalyptic
        	            }
        	        }
        	    }
        	    would_follow_social_distancing_value >= -1
        	}
        	add = {
        	    value = -60
        	    desc = "Plague in Domain"
        	}
    	}
    	else_if = {
    	    limit = {
    	        any_realm_county = {
    	            any_county_province = {
    	                any_province_epidemic = {
    	                    count >= 1
    	                    outbreak_intensity < apocalyptic
    	                }
    	            }
    	        }
    	        would_follow_social_distancing_value >= -1
    	    }
    	    add = {
    	        value = -20
    	        desc = "Plague in Top Realm"
    	    }
    	}
    	if = {
    	    limit = {
    	        any_held_title = {
					tier = tier_county
    	            any_county_province = {
    	                any_province_epidemic = {
    	                    count >= 1
    	                    outbreak_intensity = apocalyptic
    	                }
    	            }
    	        }
    	    }
    	    add = {
    	        value = -200
    	        desc = "Apocalyptic Plague in Domain"
    	    }
    	}
    	else_if = {
    	    limit = {
    	        any_realm_county = {
    	            any_county_province = {
    	                any_province_epidemic = {
    	                    count >= 1
    	                    outbreak_intensity = apocalyptic
    	                }
    	            }
    	        }
    	    }
    	    add = {
    	        value = -40
    	        desc = "Apocalyptic Plague in Top Realm"
    	    }
    	}
	}
