GuelphGhibellinePeaceConflict = {
	type = character

	text = {
		trigger = {
			situation:guelph_ghibelline ?= { 
				OR = {
					situation_current_phase = agitation
					situation_current_phase = guelph_advantage
					situation_current_phase = ghibelline_advantage
				}
			}
		}
		localization_key = guelph_ghibelline_peace_conflict_peace
	}

	text = {
		trigger = {
			situation:guelph_ghibelline ?= { situation_current_phase = repose }
		}
		localization_key = guelph_ghibelline_peace_conflict_conflict
	}
}
