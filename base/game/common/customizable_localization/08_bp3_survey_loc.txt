SurveyArrivalIntroProsperityLoc = {
	type = character

	text = {
		trigger = {
			survey_province_prosperity_info_value >= 100
		}
		localization_key = SurveyArrivalIntroProsperityLocMaxed
	}

	text = {
		trigger = {
			survey_province_prosperity_info_value >= {
				value = root.capital_county.prosperity_level
				add = 5
			}
		}
		localization_key = SurveyArrivalIntroProsperityLocGood
	}

	text = {
		trigger = {
			survey_province_prosperity_info_value <= {
				value = root.capital_county.prosperity_level
				subtract = 5
			}
		}
		localization_key = SurveyArrivalIntroProsperityLocBad
	}

	text = {
		localization_key = SurveyArrivalIntroProsperityLocNeutral
	}
}

SurveyArrivalIntroOpinionLoc = {
	type = character

	text = {
		trigger = {
			scope:second != root
			scope:second = {
				opinion = {
					target = root
					value > 25
				}
			}
		}
		localization_key = SurveyArrivalIntroOpinionLocGood
	}

	text = {
		trigger = {
			scope:second != root
			scope:second = {
				opinion = {
					target = root
					value < 0
				}
			}
		}
		localization_key = SurveyArrivalIntroOpinionLocBad
	}

	text = {
		fallback = yes
		localization_key = SurveyArrivalIntroOpinionLocEmpty
	}
}
