# add_random_county_corruption_modifier_effect - Adds a random CC modifier the County does not already have
# remove_random_county_corruption_modifier_effect - Removes a random CC the County has

add_inefficient_tax_collection_modifier_effect = {
	save_scope_as = county
	if = {
		limit = {
			can_worsen_county_corruption_trigger = yes
			NOR = {
				has_county_modifier = county_corruption_inefficient_tax_collection_modifier_2
				has_county_modifier = county_corruption_inefficient_tax_collection_modifier_3
			}
		}
		holder = {
			send_interface_message = {
				type = msg_county_corruption_bad
				title = {
					first_valid = {
						triggered_desc = {
							trigger = { exists = scope:deficit }
							desc = gain_county_corruption.message.deficit
						}
						triggered_desc = {
							trigger = { exists = scope:low_control }
							desc = gain_county_corruption.message.low_control
						}
						desc = gain_county_corruption.message.county_corruption
					}
				}
				left_icon = scope:county
				scope:county = {
					add_county_modifier = {
						modifier = county_corruption_inefficient_tax_collection_modifier
						years = 10
					}
				}
			}
		}
	}
}

add_smuggling_ring_modifier_effect = {
	save_scope_as = county
	if = {
		limit = {
			can_worsen_county_corruption_trigger = yes
			NOR = {
				has_county_modifier = county_corruption_smuggling_ring_modifier_2
				has_county_modifier = county_corruption_smuggling_ring_modifier_3
			}
		}
		title_province.barony.trade_market = {
			save_scope_as = market
			trade_market_owner.county.holder = { save_scope_as = market_owner } # Market Owner
			inform_market_of_smuggling_modifier_effect = yes
		}
		holder = {
			send_interface_message = {
				type = msg_county_corruption_bad

						}

					}
				}
				left_icon = scope:county
				scope:county = {
					add_county_modifier = {
						modifier = county_corruption_smuggling_ring_modifier
						years = 10
					}
				}
				scope:market = {
					add_trade_market_modifier = {
						modifier = gdp_market_smuggling_trade_modifier

		}

						add_county_modifier = {
							modifier = county_corruption_smuggling_ring_modifier_3
							years = 10
						}
					}
					else_if = {
						limit = { has_county_modifier = county_corruption_smuggling_ring_modifier }
						hidden_effect = { remove_county_modifier = county_corruption_smuggling_ring_modifier }
						add_county_modifier = {
							modifier = county_corruption_smuggling_ring_modifier_2
							years = 10
						}
					}
					else = {
						add_county_modifier = {
							modifier = county_corruption_smuggling_ring_modifier
							years = 10
						}
					}
				}
				scope:market = {
					add_trade_market_modifier = {
						modifier = gdp_market_smuggling_trade_modifier
						years = 10
					}
				}
			}
		}
	}
