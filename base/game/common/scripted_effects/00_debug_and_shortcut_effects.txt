# Fires an activity or scheme event on root, setting up whichever scopes are available.
## Usage: debug_fire_event_effect = { EVENT = my_event.001 }
debug_fire_event_effect = {
	# Activity events need the activity and its host.
	involved_activity ?= {
		save_scope_as = activity
		activity_host ?= { save_scope_as = host }
	}
	# Scheme events need the scheme, its owner, and whatever it is targeting.
	random_scheme = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target_character ?= { save_scope_as = target }
		scheme_target_title ?= { save_scope_as = target }
		scheme_target_culture ?= { save_scope_as = target }
		scheme_target_faith ?= { save_scope_as = target }
	}
	trigger_event = $EVENT$
}

# Usage: debug_add_breaches_all_active_schemes_effect = { AMOUNT = 5 }
debug_add_breaches_all_active_schemes_effect = {
	every_scheme = {
		add_scheme_breach = $AMOUNT$
	}
}

# Usage: debug_add_advantages_all_active_schemes_effect = { AMOUNT = 20 }
debug_add_advantages_all_active_schemes_effect = {
	every_scheme = {
		change_opportunities = $AMOUNT$
	}
}

log_debug_variable_for_persian_struggle_effect = {
	if = {
		limit = {
			gather_debug_variables_for_persian_struggle_trigger = yes
			is_struggle_type = persian_struggle
		}
		increment_global_variable_effect = {
			VAR = sp_$VAR$
			VAL = 1
		}
	}
}


##################################################
# Shortcuts
##################################################

# Easily set-or-add to a variable.
increment_variable_effect = {
	if = {
		limit = {
			NOT = { exists = var:$VAR$ }
		}
		set_variable = {
			name = $VAR$
			value = $VAL$
		}
	}
	else = {
		change_variable = {
			name = $VAR$
			add = $VAL$
		}
	}
}

increment_variable_remove_at_zero_effect = {
	if = {
		limit = { exists = var:$VAR$ }
		change_variable = {
			name = $VAR$
			add = $VALUE$
		}
		if = {
			limit = { var:$VAR$ <= 0 }
			remove_variable = $VAR$
		}
	}
}

# Easily set-or-add to a _global_ variable - generally only for use in logging.
increment_global_variable_effect = {
	if = {
		limit = {
			NOT = { exists = global_var:$VAR$ }
		}
		set_global_variable = {
			name = $VAR$
			value = $VAL$
		}
	}
	else = {
		change_global_variable = {
			name = $VAR$
			add = $VAL$
		}
	}
}
