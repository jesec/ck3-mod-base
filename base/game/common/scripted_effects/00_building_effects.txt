add_next_building_tier_effect = {
	if = {
		limit = { has_building = $BUILDING$_4 }
		start_building_construction_for_holder = $BUILDING$_5
	}
	else_if = {
		limit = { has_building = $BUILDING$_3 }
		start_building_construction_for_holder = $BUILDING$_4
	}
	else_if = {
		limit = { has_building = $BUILDING$_2 }
		start_building_construction_for_holder = $BUILDING$_3
	}
	else_if = {
		limit = { has_building = $BUILDING$_1 }
		start_building_construction_for_holder = $BUILDING$_2
	}
	else_if = {
		limit = { can_start_building_construction_for_owner = $BUILDING$_1 }
		start_building_construction_for_holder = $BUILDING$_1
	}
}

destroy_or_downgrade_district_building_effect = {
	random_built_regular_district = {
		limit = { has_building_or_higher = $BUILDING$_1 }
		if = {
			limit = { has_building = $BUILDING$_5 }
			remove_building = $BUILDING$_5
			hidden_effect = { add_building = $BUILDING$_4 }
		}
		else_if = {
			limit = { has_building = $BUILDING$_4 }
			remove_building = $BUILDING$_4
			hidden_effect = { add_building = $BUILDING$_3 }
		}
		else_if = {
			limit = { has_building = $BUILDING$_3 }
			remove_building = $BUILDING$_3
			hidden_effect = { add_building = $BUILDING$_2 }
		}
		else_if = {
			limit = { has_building = $BUILDING$_2 }
			remove_building = $BUILDING$_2
			hidden_effect = { add_building = $BUILDING$_1 }
		}

		}
	}
}

# Province scope. Picks the building the next destroy_random_building_effect in this province hits, so its tooltip and outcome match
destroy_random_building_variable_effect = {
	if = {
		limit = { exists = scope:destroyed_building_district }
		clear_saved_scope = destroyed_building_district
	}
	random_built_regular_district = {
		limit = {
			any_district_building = { is_primary_building = no }
		}
		save_scope_as = destroyed_building_district
		random_district_building = {
			limit = { is_primary_building = no }
			building_type = { save_scope_as = destroyed_building_type }
		}
	}
}

			hidden_effect = { add_building = temple_01 }
		}
		else = {
			remove_building = temple_01
		}
	}
}

add_next_temple_tier_effect = {
	holding_district = {
		if = {
			limit = {
				NOT = { has_building_or_higher = temple_01 }
			}
			add_building = temple_01
		}
		else_if = {
			limit = { has_building = temple_01 }
			add_building = temple_02
		}
		else_if = {
			limit = { has_building = temple_02 }
			add_building = temple_03
		}
		else_if = {
			limit = { has_building = temple_03 }
			add_building = temple_04
		}
	}
}

		}
	}
}
