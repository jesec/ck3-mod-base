merchant_luxury_contract_travel_or_progress_effect = {
	if = {
		limit = { location != $DESTINATION$ }
		start_travel_plan = {
			destination = $DESTINATION$
			on_arrival_event = $ARRIVAL_EVENT$
			on_travel_planner_cancel_event = gdp_merchant_trade_opportunity.9000
			on_arrival_destinations = all_but_last
		}
	}
	else = { trigger_event = $ARRIVAL_EVENT$ }
}

manage_new_luxury_contract_effect = {
	if = {
		limit = { exists = scope:destination }
		if = {
			limit = {
				NOT = { has_character_flag = luxury_contract_frequency_cooldown }
			}
			add_character_flag = { flag = luxury_contract_frequency_cooldown years = 6 }
		}
		send_interface_message = {
			type = msg_merchant_contracts
			title = msg_luxury_contract_appear.t
			desc = msg_luxury_contract_appear.desc
			right_icon = scope:destination.county
		}
	}
	if = {
		limit = {
			is_ai = yes
			exists = scope:luxury_contract
		}
		accept_task_contract = scope:luxury_contract
	}
}

save_random_region_province_effect = {
	if = {
		limit = { exists = scope:find_curio }
		random_county_in_region = {
			region = $REGION$
			limit = {
				duchy.title_capital_county ?= this
				any_county_province = { gdp_find_curio_poi_province_trigger = yes }
			}
			alternative_limit = { duchy.title_capital_county ?= this }
			weight = {
				base = 1
				modifier = { add = prosperity_level }
				modifier = {
					factor = holder.highest_held_title_tier
					exists = holder
				}
				modifier = {
					add = {
						value = 0
						every_de_jure_county = {
							limit = {
								any_county_province = { gdp_find_curio_poi_province_trigger = yes }
							}
							add = 150
						}
					}
				}
				modifier = {
					title_province ?= { has_holding_type = tribal_holding }
					factor = 0.25
				}
			}
			duchy.title_capital_county.title_province = { save_scope_as = destination }
		}
	}
	else = {
		random_county_in_region = {
			region = $REGION$
			limit = {
				duchy.title_capital_county ?= this
				exists = root.var:target_animal_type
				title_province ?= {  
					switch = {
						trigger = root.var:target_animal_type 
						flag:lion = {
							hunt_animal_lion_trigger = yes
						}
						flag:tiger = {
							hunt_animal_tiger_trigger = yes
						}
						flag:leopard = {
							hunt_animal_leopard_trigger = yes
						}
						flag:caracal = {
							hunt_animal_caracal_trigger = yes
						}
						flag:lynx = {
							hunt_animal_lynx_trigger = yes
						}
						flag:bear = {
							hunt_animal_bear_trigger = yes
						}
						flag:elephant = {
							luxury_contract_creature_elephant_trigger = yes
						}
						fallback = { always = no }
					}
				}
			}
			weight = {
				base = 1
				modifier = { add = prosperity_level }
				modifier = {
					factor = holder.highest_held_title_tier
					exists = holder
				}
				modifier = {
					title_province ?= { has_holding_type = tribal_holding }
					factor = 0.25
				}
			}
			title_province = { save_scope_as = destination }
		}
	}
}
