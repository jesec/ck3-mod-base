gg_invalidation_missing_leader_message_effect = {
	situation:guelph_ghibelline ?= {
		#All players
		every_situation_participant = {
			limit = {
				is_ai = no
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = gg_invalidation_title
				if = {
					limit = { $MISSING_LEADER$ = flag:guelph }
					custom_tooltip = gg_invalidation_missing_guelph_leader_tt
					debug_log = "Guelphs are missing a leader! Invalidating GG"
					debug_log_scopes = yes
				}
				else_if = {
					limit = { $MISSING_LEADER$ = flag:ghibelline }
					custom_tooltip = gg_invalidation_missing_ghibelline_leader_tt
					debug_log = "Ghibellines are missing a leader! Invalidating GG"
					debug_log_scopes = yes
				}
				else_if = {
					limit = { $MISSING_LEADER$ = flag:hre }
					custom_tooltip = gg_invalidation_missing_hre_leader_tt
					debug_log = "There is no HRE holder! Invalidating GG"
					debug_log_scopes = yes
				}
				else_if = {
					limit = { $MISSING_LEADER$ = flag:1066_hre_not_leader }
					custom_tooltip = gg_invalidation_missing_hre_leader_dictatus_tt
					debug_log = "The HRE is not Ghibelline leader in Prelude! Invalidating GG"
					debug_log_scopes = yes
				}
				else_if = {
					limit = { $MISSING_LEADER$ = flag:1066_pope_not_leader }
					custom_tooltip = gg_invalidation_missing_pope_leader_dictatus_tt
					debug_log = "The Pope is not Guelph leader in Prelude! Invalidating GG"
					debug_log_scopes = yes
				}
				else_if = {
					limit = { $MISSING_LEADER$ = flag:pope_is_emperor }
					custom_tooltip = gg_invalidation_pope_is_emperor_tt
					debug_log = "The Pope and the Emperor are the same character! Invalidating GG"
					debug_log_scopes = yes
				}
				else = {
					custom_tooltip = gg_invalidation_missing_pope_leader_tt
					debug_log = "There is no Pope! Invalidating GG"
					debug_log_scopes = yes
				}
				show_as_tooltip = {
					situation:guelph_ghibelline = { end_situation = yes }
				}
			}
		}
	}
}

try_set_up_new_gg_leader_effect = {
	save_temporary_scope_as = gg_departing_leader
	#Special handling to make sure Pope stays Guelph leader, and HRE inheritor stays Ghibelline leader
	if = {
		limit = { 
			gg_phase_is_pre_gg_trigger = yes
			exists = var:guelph_leader
		}
		if = {
			limit = {
				exists = title:k_papal_state
				this != title:k_papal_state.holder
				title:k_papal_state.holder = {
					NOT = { exists = var:guelph_leader }
				}
			}
			title:k_papal_state.holder = {
				set_variable = guelph_leader
				recalculate_participant_group = situation:guelph_ghibelline
				if = {
					limit = { NOT = { exists = var:gg_involvement } }
					set_variable = {
						name = gg_involvement
						value = gg_involvement_base_value
					}
				}
			}
			remove_variable = guelph_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}		
	}
	else_if = {
		limit = { 
			gg_phase_is_pre_gg_trigger = yes
			exists = var:ghibelline_leader
		}
		if = {
			limit = { 
				exists = title:e_hre
				this != title:e_hre.holder
				title:e_hre.holder = {
					NOT = { exists = var:ghibelline_leader }
				} 
			}
			title:e_hre.holder = {
				set_variable = ghibelline_leader
				recalculate_participant_group = situation:guelph_ghibelline
				if = {
					limit = { NOT = { exists = var:gg_involvement } }
					set_variable = {
						name = gg_involvement
						value = gg_involvement_base_value
					}
				}
			}
			remove_variable = ghibelline_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}
	}
	else = {
		if = {
			limit = { has_variable = guelph_leader }
			if = {
				limit = {
					primary_heir ?= {
						this != scope:gg_departing_leader # Republics can re-elect, so the primary heir may be this dead ruler
						save_temporary_scope_as = heir_check
						situation:guelph_ghibelline ?= { 
							any_situation_participant = { this = scope:heir_check }
						}
						NOR = { 
							has_variable = ghibelline_leader
							any_held_title = { this = title:k_papal_state }
						}
						top_liege = scope:gg_departing_leader
					}
				}
				primary_heir = {
					set_variable = guelph_leader
					save_scope_as = new_gg_leader
				}
			}
			else_if = {
				limit = {
					situation:guelph_ghibelline = {
						situation_participant_group:guelph_member = {
							any_situation_group_participant = { 
								count >= 1
								NOT = { 
									any_held_title = { this = title:k_papal_state }
								}
							}
						}
					}
				}
				situation:guelph_ghibelline = {
					situation_participant_group:guelph_member = {
						ordered_situation_group_participant = {
							limit = { 
								NOT = { 
									any_held_title = { this = title:k_papal_state }
								}
							}
							order_by = primary_title.tier
							set_variable = guelph_leader
							save_scope_as = new_gg_leader
						}
					}
				}
			}
			else = {
				save_scope_value_as = {
					name = missing_leader
					value = flag:guelph
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
				situation:guelph_ghibelline = { end_situation = yes }
			}
			remove_variable = guelph_leader
		}
		if = {
			limit = { has_variable = ghibelline_leader }
			if = {
				limit = {
					primary_heir ?= {
						this != scope:gg_departing_leader
						save_temporary_scope_as = heir_check
						situation:guelph_ghibelline ?= { 
							any_situation_participant = { this = scope:heir_check }
						}
						NOT = { has_variable = guelph_leader }
						top_liege = scope:gg_departing_leader
					}
				}
				primary_heir = { 
					set_variable = ghibelline_leader
					save_scope_as = new_gg_leader
				}
			}
			else_if = {
				limit = {
					situation:guelph_ghibelline = {
						situation_participant_group:ghibelline_member = {
							any_situation_group_participant = { count >= 1 }
						}
					}
				}
				situation:guelph_ghibelline = {
					situation_participant_group:ghibelline_member = {
						ordered_situation_group_participant = {
							order_by = primary_title.tier
							set_variable = ghibelline_leader
							save_scope_as = new_gg_leader
						}
					}
				}
			}
			else = {
				save_scope_value_as = {
					name = missing_leader
					value = flag:ghibelline
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
				situation:guelph_ghibelline = { end_situation = yes }
			}
			remove_variable = ghibelline_leader
		}
		if = {
			limit = { exists = situation:guelph_ghibelline }
			custom_tooltip = {
				text = new_gg_leader_tt
				scope:new_gg_leader = { recalculate_participant_group = situation:guelph_ghibelline }
				recalculate_participant_group = situation:guelph_ghibelline
			}
		}
	}
}


gg_gather_claim_leadership_schemes_effect = {
	hidden_effect = {
		every_targeting_scheme = {
			limit = { scheme_type = gg_claim_leadership }
			add_to_temporary_list = gg_claim_leadership_schemes_to_invalidate
		}
	}
}

gg_become_new_leader_effect = {
	save_scope_as = root_scope
	save_scope_value_as = {
		name = gg_leadership_change_source
		value = $VIA_SCHEME$
	}
	if = {
		limit = { is_guelph_trigger = yes  }
		situation:guelph_ghibelline ?= {
			every_situation_participant = {
				limit = { is_guelph_leader_trigger = yes  }
				gg_gather_claim_leadership_schemes_effect = yes
				if = {
					limit = {
						NOR = {
							has_relation_friend = scope:root_scope
							opinion = {
								target = scope:root_scope
								value >= 80
							}
						}
					}
					progress_towards_rival_effect = {
						REASON = rival_overthrown_guelph_leader
						CHARACTER = scope:root_scope
						OPINION = 0
					}
					add_opinion = {
						target = scope:root_scope
						modifier = seized_guelph_leadership_opinion
						opinion = -40
					}
				}
				remove_variable = guelph_leader
				save_scope_as = affected_character #For the tooltip
				create_character_memory = {
					type = lost_guelph_leadership_memory
					participants = { usurper = scope:root_scope }
				}
				if = {
					limit = { scope:gg_leadership_change_source = flag:scheme }
					scope:new_memory = { set_variable = via_scheme }
				}
				custom_tooltip = {
					text = change_to_guelph_tt
					recalculate_participant_group = situation:guelph_ghibelline
				}
				set_variable = {
					name = recently_overthrown_gg_leader
					years = 5
				}
			}
		}
		custom_tooltip = {
			text = become_new_guelph_leader_tt
			set_variable = guelph_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}
		create_character_memory = { type = became_guelph_leader_memory }
	}
	else = {
		situation:guelph_ghibelline ?= {
			every_situation_participant = {
				limit = { is_ghibelline_leader_trigger = yes }
				gg_gather_claim_leadership_schemes_effect = yes
				if = {
					limit = {
						NOR = {
							has_relation_friend = scope:root_scope
							opinion = {
								target = scope:root_scope
								value >= 80
							}
						}
					}
					progress_towards_rival_effect = {
						REASON = rival_overthrown_ghibelline_leader
						CHARACTER = scope:root_scope
						OPINION = 0
					}
					add_opinion = {
						target = scope:root_scope
						modifier = seized_ghibelline_leadership_opinion
						opinion = -40
					}
				}
				remove_variable = ghibelline_leader
				save_scope_as = affected_character #For the tooltip
				create_character_memory = {
					type = lost_ghibelline_leadership_memory
					participants = { usurper = scope:root_scope }
				}
				if = {
					limit = { scope:gg_leadership_change_source = flag:scheme }
					scope:new_memory = { set_variable = via_scheme }
				}
				custom_tooltip = {
					text = change_to_ghibelline_tt
					recalculate_participant_group = situation:guelph_ghibelline
				}
				set_variable = {
					name = recently_overthrown_gg_leader
					years = 5
				}
			}
		}
		custom_tooltip = {
			text = become_new_ghibelline_leader_tt
			set_variable = ghibelline_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}
		create_character_memory = { type = became_ghibelline_leader_memory }
	}
	hidden_effect = {
		every_in_list = {
			list = gg_claim_leadership_schemes_to_invalidate
			invalidate_scheme = yes
		}
	}
}

gg_clean_up_pope_modifiers_effect = {
	#Remove Pope-specific GG modifiers
	if = {
		limit = {
			faith = {
				OR = {
					religion != religion:christianity_religion
					has_doctrine_parameter = no_head_of_faith
				}
			}				
		}
		if = {
			limit = { has_character_modifier = favored_by_pope_modifier }
			remove_character_modifier = favored_by_pope_modifier
		}
		if = {
			limit = { has_character_modifier = scorned_pope_modifier }
			remove_character_modifier = scorned_pope_modifier
		}
	}
}



gg_swap_to_undecided_effect = {
	if = {
		limit = { is_guelph_or_ghibelline_trigger = yes }
		if = {
			limit = {
				OR = {
					has_variable = guelph_leader
					has_variable = ghibelline_leader
				}
			}
			try_set_up_new_gg_leader_effect = yes
		}
		save_scope_as = affected_character
		if = {
			limit = { $USE_MODIFIERS$ = flag:yes }
			if = {
				limit = { is_guelph_trigger = yes }
				add_character_modifier = {
					modifier = left_guelphs_modifier
					years = 10
				}
			}
			if = {
				limit = { is_ghibelline_trigger = yes }
				add_character_modifier = {
					modifier = left_ghibellines_modifier
					years = 10
				}
			}
		}
		custom_tooltip = {
			text = change_to_undecided_tt
			gg_clean_up_participant_groups_effect = yes
			gg_clean_up_swap_variables_effect = yes
			set_variable = undecided_override
			recalculate_participant_group = situation:guelph_ghibelline
		}
	}
}

gg_involvement_cb_participant_effect = {
	$WAR_OUTCOME$ = { save_scope_as = gg_outcome }
	if = {
		limit = {
			scope:attacker = { is_$ATTACKER$_trigger = yes }
			scope:defender = { is_$DEFENDER$_trigger = yes }
		}
		scope:war ?= {
			every_war_participant = {
				limit = {
					is_$ATTACKER$_trigger = yes
					is_attacker_in_war = scope:war
					NOT = { this = scope:attacker }
				}
				save_scope_as = participant_temp
				save_scope_value_as = {
					name = participant_war_contribution_value
					value = {
						value = "scope:war.war_contribution(scope:participant_temp)"
						divide = 1000
						round = yes
					}
				}
				save_scope_value_as = {
					name = participant_actual_involvement_gain
					value = {
						add = {
							value = gg_involvement_base_addition_value
							multiply = scope:participant_war_contribution_value
							if = {
								limit = { gg_phase_gives_half_involvement_trigger = yes }
								multiply = 0.5
							}
							if = {
								limit = { gg_phase_gives_double_involvement_trigger = yes }
								multiply = 2
							}
							round = yes
							if = {
								limit = { scope:gg_outcome = flag:victory }
								max = gg_involvement_cb_participant_value
							}
							else_if = {
								limit = { scope:gg_outcome = flag:white_peace }
								max = gg_involvement_cb_failure_value
							}
							else = {
								max = {
									value = gg_involvement_cb_failure_value
									divide = 2
								}
							}
						}
					}
				}
				if = {
					limit = { 
						has_variable = gg_involvement
						scope:participant_actual_involvement_gain > gg_involvement_base_value
					}
					custom_tooltip = {
						text = gg_involvement_cb_participant_actual_tt
						gg_add_involvement_effect = { AMOUNT = scope:participant_actual_involvement_gain }
					}
				}
			}
		}
	}
}

gg_involvement_cb_effect = {
	$WAR_OUTCOME$ = { save_scope_as = gg_outcome }
	if = {
		limit = { exists = scope:war }
		scope:war ?= {
			#Calculate the involvement for the leader
			random_war_participant = {
				limit = { this = scope:attacker }
				save_scope_value_as = {
					name = winner_war_contribution_value
					value = {
						value = "scope:war.war_contribution(scope:attacker)"
						divide = 1000
						round = yes
					}
				}
				save_scope_value_as = {
					name = actual_involvement_gain
					value = {
						add = {
							value = gg_involvement_base_addition_value
							multiply = scope:winner_war_contribution_value
							if = {
								limit = { gg_phase_gives_half_involvement_trigger = yes }
								multiply = 0.5
							}
							if = {
								limit = { gg_phase_gives_double_involvement_trigger = yes }
								multiply = 2
							}
							round = yes
							if = {
								limit = { scope:gg_outcome = flag:victory }
								max = gg_involvement_cb_success_value
							}
							else_if = {
								limit = { scope:gg_outcome = flag:white_peace }
								max = gg_involvement_cb_participant_value
							}
							else = {
								max = gg_involvement_cb_failure_value
							}
						}
					}
				}
				if = {
					limit = { has_variable = gg_involvement }
					custom_tooltip = {
						text = gg_involvement_cb_success_actual_tt
						gg_add_involvement_effect = { AMOUNT = scope:actual_involvement_gain }
					}
				}
			}
			#Involvement for Guelph participant attackers
			if = {
				limit = {
					scope:attacker = { is_guelph_trigger = yes }
					scope:defender = { is_ghibelline_trigger = yes }
				}
				gg_involvement_cb_participant_effect = { 
					ATTACKER = guelph 
					DEFENDER = ghibelline 
					WAR_OUTCOME = scope:gg_outcome
				}
			}
			#Involvement for Ghibelline participant attackers
			if = {
				limit = {
					scope:attacker = { is_ghibelline_trigger = yes }
					scope:defender = { is_guelph_trigger = yes }
				}
				gg_involvement_cb_participant_effect = { 
					ATTACKER = ghibelline 
					DEFENDER = guelph 
					WAR_OUTCOME = scope:gg_outcome
				}
			}
		}
	}
	else = {
		custom_tooltip = gg_involvement_cb_success_tt
		custom_tooltip = gg_involvement_cb_participant_tt
	}
}

gg_become_independent_effect = {
	$TITLE$ = { save_scope_as = independency_title }
	#Is the title not the primary title of the holder? Invent someone to take over!
	if = {
		limit = {
			#The title isn't the holder's primary title (i.e. it goes in under a duchy or similar)
			scope:independency_title.holder.primary_title != scope:independency_title
			#You shouldn't already be the count of a neighboring county (e.g. Friuli / Treviso)
			scope:independency_title.holder.primary_title.tier > scope:independency_title.tier
		}
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = yes
		}
		scope:independency_title = { save_scope_as = target_title }
		#Create character
		create_character = {
			template = gg_ruler_template
			location = scope:target_title.title_province
			save_scope_as = new_holder
		}
		scope:independency_title = {
			change_title_holder = {
				holder = scope:new_holder
				change = scope:change
			}
		}
		scope:new_holder = {
			becomes_independent = { change = scope:change }
			save_scope_as = became_independent
		}
		resolve_title_and_vassal_change = scope:change
	}
	else = {
		scope:independency_title.holder = {
			#Otherwise just your regular independency
			if = {
				limit = { is_independent_ruler = no }
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = yes
				}
				becomes_independent = { change = scope:change }
				resolve_title_and_vassal_change = scope:change
				if = {
					limit = { scope:league_creator ?= scope:independency_title.holder }
					save_scope_as = became_independent
				}
			}
		}
	}
}

#This is in the confederation scope
try_to_add_to_lombard_league_effect = {
	$TITLE$ = { save_scope_as = title_to_check }
	#First check that the title is still relevant for the HRE, whether the holder is the Count of Friuli or Matilda
	if = {
		limit = {
			scope:title_to_check = {
				trigger_if = {
					limit = { this = title:k_venice }
					holder ?= { is_either_hre_vassal_or_independent_king_trigger = yes }
				}
				trigger_else = {
					holder ?= { is_either_hre_vassal_or_independent_duke_or_less_trigger = yes }
				}
			}
		}
		if = {
			limit = {
				OR = {
					#Then check if the holder is AI
					scope:title_to_check.holder = { is_ai = yes }
					#If Matilda is a player, the counts leave regardless
					trigger_if = {
						limit = { scope:title_to_check.holder.primary_title != scope:title_to_check }
						#We also account for counts who hold two counties, such as Friuli and Treviso
						scope:title_to_check.holder.primary_title.tier > scope:title_to_check.tier
					}
				}
			}
			actual_add_to_lombard_league_effect = { TITLE = scope:title_to_check }
		}
		else = {
			#For the tooltip
			scope:title_to_check = { save_scope_as = potential_league_title }
			custom_tooltip = {
				text = potential_league_title_is_human.tt
				scope:title_to_check.holder = { trigger_event = gdp_guelph_ghibelline.7005 }
			}
		}		
	}
}

			limit = { is_ai = no }

		}
	}
}
