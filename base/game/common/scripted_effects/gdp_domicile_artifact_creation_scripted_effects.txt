create_library_artifact_effect = {
	create_artifact_book_effect = {
		OWNER = $OWNER$
		CREATOR = $CREATOR$
		SET_SUBJECT = $SUBJECT$
		SET_TOPIC = $TOPIC$
	}
	hidden_effect_new_object = {
		scope:newly_created_artifact ?= {
			reforge_artifact = { type = library_item }
			domicile_library_artifact_modifiers_effect = yes
		}
	}
	custom_tooltip = create_library_artifact
}

create_curio_artifact_effect = {
	# Without a caller-set quality the trinket pool includes the bad tier, so roll common to rare here
	hidden_effect = {
		if = {
			limit = {
				NOT = { exists = scope:quality }
			}
			random_list = {
				55 = {
					save_scope_value_as = {
						name = quality
						value = 25
					}
				}
				35 = {
					save_scope_value_as = {
						name = quality
						value = 50
					}
				}
				10 = {
					save_scope_value_as = {
						name = quality
						value = 70
					}
				}
			}
			save_scope_value_as = {
				name = curio_rolled_quality
				value = yes
			}
		}
	}
	generate_trinket_effect = {
		TRINKET_RECEIVER = $OWNER$
		TRINKET_GIVER = $CREATOR$
		GRAB_ALL_TRINKETS = yes
		HISTORY_TYPE = given
	}
	if = {
		limit = { exists = scope:curio_rolled_quality }
		clear_saved_scope = quality
		clear_saved_scope = curio_rolled_quality
	}
	hidden_effect_new_object = {
		scope:new_trinket ?= {
			reforge_artifact = { type = curio }
			domicile_curio_artifact_modifiers_effect = yes
		}
	}
	custom_tooltip = create_curio_artifact
}

select_local_creature_effect = {

							exists = scope:harmless_local_creature

							OR = {
								hunt_animal_lion_trigger = yes

							}
						}
					}

						factor = 0.1
						AND = {
							character_has_creature_type_trigger = { CHAR = root TYPE = lion }
							character_has_creature_type_trigger = { CHAR = root TYPE = tiger }
							character_has_creature_type_trigger = { CHAR = root TYPE = leopard }

create_creature_artifact_effect = {
	# owner is who the artifact is made for
	# For luxury contracts the creator is the contract giver, otherwise it should always be a merchant
	$OWNER$ = { save_scope_as = owner }
	$CREATOR$ = { save_scope_as = creator }

	get_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes

	scope:owner = {
		if = {
			limit = {
				# Check if creature type is already set.
				NOT = { domicile_creature_animal_type_trigger = { TYPE = var:animal_type } }
			}
			random_list = {
				8 = {
					luxury_contract_standard_creature_effect = { PROVINCE = scope:creator.location }
				}
				2 = {
					luxury_contract_dangerous_creature_effect = { PROVINCE = scope:creator.location }
				}
			}
		}

		hidden_effect_new_object = {
			if = {
				limit = {
					OR = {
						var:animal_type ?= flag:peacock
						var:animal_type ?= flag:parrot
						var:animal_type ?= flag:cockatoo
						var:animal_type ?= flag:crane
					}
				}
				create_domicile_creature_artifact_object_effect = { VISUALS = creature_court_trophy_bird }
			}
			else_if = {
				limit = {
					var:animal_type ?= flag:monkey
				}
				create_domicile_creature_artifact_object_effect = { VISUALS = creature_court_trophy_monkey }
			}
			else_if = {
				limit = {
					OR = {
						var:animal_type ?= flag:caracal
						var:animal_type ?= flag:lynx
					}
				}
				create_domicile_creature_artifact_object_effect = { VISUALS = creature_court_small_hide }
			}
			else_if = {
				limit = {
					OR = {
						var:animal_type ?= flag:lion
						var:animal_type ?= flag:tiger
						var:animal_type ?= flag:leopard
						var:animal_type ?= flag:bear
						var:animal_type ?= flag:elephant
					}
				}
				create_domicile_creature_artifact_object_effect = { VISUALS = creature_court_large_hide }
			}
			else = {
				create_domicile_creature_artifact_object_effect = { VISUALS = creature }
			}
		}
	}

	# DO NOT set this variable inside of the hidden effect - the whole scripted effect will break!! 
	scope:newly_created_artifact ?= {
		set_variable = {
			name = animal_type
			value = scope:owner.var:animal_type
		}
		set_variable = {
			name = creature_location
			value = scope:creature_location
		}
		if = {
			limit = { exists = var:creature_location }
			## Supress startup errors
		}
	}

	hidden_effect = {
		# Now we set the proper name and desc.
		scope:newly_created_artifact ?= {
			switch = {
				trigger = var:animal_type
				flag:lion = {
					set_artifact_name = creature_type_lion_name
					set_artifact_description = creature_type_lion_desc
				}
				flag:caracal = {
					set_artifact_name = creature_type_caracal_name
					set_artifact_description = creature_type_caracal_desc
				}
				flag:lynx = {
					set_artifact_name = creature_type_lynx_name
					set_artifact_description = creature_type_lynx_desc
				}
				flag:tiger = {
					set_artifact_name = creature_type_tiger_name
					set_artifact_description = creature_type_tiger_desc
				}
				flag:leopard = {
					set_artifact_name = creature_type_leopard_name
					set_artifact_description = creature_type_leopard_desc
				}
				flag:bear = {
					set_artifact_name = creature_type_bear_name
					set_artifact_description = creature_type_bear_desc
				}
				flag:elephant = {
					set_artifact_name = creature_type_elephant_name
					set_artifact_description = creature_type_elephant_desc
				}
				flag:peacock = {
					set_artifact_name = creature_type_peacock_name
					set_artifact_description = creature_type_peacock_desc
				}
				flag:parrot = {
					set_artifact_name = creature_type_parrot_name
					set_artifact_description = creature_type_parrot_desc
				}
				flag:cockatoo = {
					set_artifact_name = creature_type_cockatoo_name
					set_artifact_description = creature_type_cockatoo_desc
				}
				flag:crane = {
					set_artifact_name = creature_type_crane_name
					set_artifact_description = creature_type_crane_desc
				}
				flag:monkey = {
					set_artifact_name = creature_type_monkey_name
					set_artifact_description = creature_type_monkey_desc
				}
			}
			add_scaled_artifact_modifier_prestige_effect = yes
			random_list = {
				50 = {
					add_scaled_artifact_modifier_majesty_effect = yes
				}
				50 = {
					add_scaled_artifact_modifier_rulership_effect = yes
				}
				50 = {
					add_scaled_artifact_modifier_attractiveness_effect = yes
				}
			}
			remove_artifact_modifier = artifact_placeholder_modifier
			domicile_creature_artifact_modifiers_effect = yes
		}
	}
	scope:owner = {	
		set_variable = {
			name = menagerie_pending_creature_artifact
			value = scope:newly_created_artifact
		}
		create_story = story_cycle_menagerie
	}
	custom_tooltip = create_creature_artifact
}

# For the Story Cycle, assign the animal's gender
assign_creature_gender_effect = {
	if = {
		limit = {
			NOT = { exists = var:creature_gender }
		}
		# We start with males as these are the illustrations we have
		set_variable = {
			name = creature_gender
			value = flag:male
		}
	}
	if = {
		limit = {
			exists = var:creature_gender
			var:animal_type ?= flag:lion
		}
		set_variable = {
			name = creature_gender
			value = flag:female
		}
	}
}

# Note: og_owner must be set before this runs, for loc
reforge_creature_artifact_effect = {
	save_scope_as = artifact
	save_scope_value_as = { name = legendary value = no }
	set_variable = { name = was_creature_artifact days = 1 }
	if = {
		limit = {		
			OR = {
				var:animal_type ?= flag:parrot
				var:animal_type ?= flag:cockatoo
				var:animal_type ?= flag:crane
				var:animal_type ?= flag:peacock
			}
		}
		reforge_artifact = {
			type = pedestal
			visuals = small_box
		}
	}
	else_if = {
		limit = { var:animal_type ?= flag:monkey }
		reforge_artifact = {
			type = animal_skull
			visuals = animal_trophy
		}
	}
	else_if = {
		limit = {
			OR = {
				var:animal_type ?= flag:caracal
				var:animal_type ?= flag:lynx
			}
		}
		reforge_artifact = {
			type = animal_hide
			visuals = small_animal_hide
		}
	}
	else = {
		reforge_artifact = {
			type = animal_hide_big
			visuals = large_animal_hide
		}
	}
	switch = {
		trigger = var:animal_type
		flag:lion = {
			set_artifact_name = creature_type_reforged_lion_name
			set_artifact_description = creature_type_lion_reforged_desc
		}
		flag:caracal = {
			set_artifact_name = creature_type_reforged_caracal_name
			set_artifact_description = creature_type_caracal_reforged_desc
		}
		flag:lynx = {
			set_artifact_name = creature_type_reforged_lynx_name
			set_artifact_description = creature_type_lynx_reforged_desc
		}
		flag:tiger = {
			set_artifact_name = creature_type_reforged_tiger_name
			set_artifact_description = creature_type_tiger_reforged_desc
		}
		flag:leopard = {
			set_artifact_name = creature_type_reforged_leopard_name
			set_artifact_description = creature_type_leopard_reforged_desc
		}
		flag:bear = {
			set_artifact_name = creature_type_reforged_bear_name
			set_artifact_description = creature_type_bear_reforged_desc
		}
		flag:elephant = {
			set_artifact_name = creature_type_reforged_elephant_name
			set_artifact_description = creature_type_elephant_reforged_desc
		}
		flag:peacock = {
			set_artifact_name = creature_type_reforged_peacock_name
			set_artifact_description = creature_type_peacock_reforged_desc
		}
		flag:parrot = {
			set_artifact_name = creature_type_reforged_parrot_name
			set_artifact_description = creature_type_parrot_reforged_desc
		}
		flag:cockatoo = {
			set_artifact_name = creature_type_reforged_cockatoo_name
			set_artifact_description = creature_type_cockatoo_reforged_desc
		}
		flag:crane = {
			set_artifact_name = creature_type_reforged_crane_name
			set_artifact_description = creature_type_crane_reforged_desc
		}
		flag:monkey = {
			set_artifact_name = creature_type_reforged_monkey_name
			set_artifact_description = creature_type_monkey_reforged_desc
		}
	}

# this = artifact
# We add one extra trade modifier scaled by rarity, or two if the artifact is famed or better
domicile_artifact_add_trade_modifiers_effect = {
	domicile_artifact_add_trade_modifier_effect = {
		CATEGORY = $CATEGORY$
		COMPETITIVENESS = $COMPETITIVENESS$
		EXPERTISE = $EXPERTISE$
		SELL = $SELL$
		BUY = $BUY$
	}
	if = {
		limit = { rarity_level >= famed }
		domicile_artifact_add_trade_modifier_effect = {
			CATEGORY = $CATEGORY$
			COMPETITIVENESS = $COMPETITIVENESS$
			EXPERTISE = $EXPERTISE$
			SELL = $SELL$
			BUY = $BUY$
		}
	}
	remove_variable = domicile_mod_competitiveness
	remove_variable = domicile_mod_expertise
	remove_variable = domicile_mod_sell_price
	remove_variable = domicile_mod_buy_price
}

# Adds one trade modifier the artifact does not have yet. this = artifact
domicile_artifact_add_trade_modifier_effect = {
	random_list = {
		$COMPETITIVENESS$ = {
			trigger = {
				NOT = { has_variable = domicile_mod_competitiveness }
			}
			domicile_artifact_add_tiered_trade_modifier_effect = { STAT = competitiveness }
			set_variable = domicile_mod_competitiveness
		}
		$EXPERTISE$ = {
			trigger = {
				NOT = { has_variable = domicile_mod_expertise }
			}
			domicile_artifact_add_tiered_trade_modifier_effect = { STAT = $CATEGORY$_expertise }
			set_variable = domicile_mod_expertise
		}
		$SELL$ = {
			trigger = {
				NOT = { has_variable = domicile_mod_sell_price }
			}
			domicile_artifact_add_tiered_trade_modifier_effect = { STAT = $CATEGORY$_sell_price }
			set_variable = domicile_mod_sell_price
		}
		$BUY$ = {
			trigger = {
				NOT = { has_variable = domicile_mod_buy_price }
			}
			domicile_artifact_add_tiered_trade_modifier_effect = { STAT = $CATEGORY$_buy_price }
			set_variable = domicile_mod_buy_price
		}
	}
}

# this = artifact
domicile_artifact_add_tiered_trade_modifier_effect = {
	if = {
		limit = { rarity = illustrious }
		add_artifact_modifier = gdp_domicile_artifact_$STAT$_4_modifier
	}
	else_if = {
		limit = { rarity = famed }
		add_artifact_modifier = gdp_domicile_artifact_$STAT$_3_modifier
	}
	else_if = {
		limit = { rarity = masterwork }
		add_artifact_modifier = gdp_domicile_artifact_$STAT$_2_modifier
	}
	else = {
		add_artifact_modifier = gdp_domicile_artifact_$STAT$_1_modifier
	}
}
