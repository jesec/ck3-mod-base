			}
		}
	}
}

	hidden_effect = {
		if = {

			current_travel_plan ?= {
				delay_travel_plan = {
					days = $DAYS$
				}
			}
		}
	}
}

poi_lifestyle_experience_effect = {
	if = {
		limit = {
			has_variable = visited_pois_lifestyle_var
			var:visited_pois_lifestyle_var >= 40
		}
		add_$LIFESTYLE$_lifestyle_xp = {
			value = $VALUE$
			divide = 5
		}
	}
	else_if = {
		limit = {
			has_variable = visited_pois_lifestyle_var
			var:visited_pois_lifestyle_var >= 30
		}
		add_$LIFESTYLE$_lifestyle_xp = {
			value = $VALUE$
			divide = 4
		}
	}
	else_if = {
		limit = {
			has_variable = visited_pois_lifestyle_var
			var:visited_pois_lifestyle_var >= 20
		}
		add_$LIFESTYLE$_lifestyle_xp = {
			value = $VALUE$
			divide = 3
		}
	}
	else_if = {
		limit = {
			has_variable = visited_pois_lifestyle_var
			var:visited_pois_lifestyle_var >= 10
		}
		add_$LIFESTYLE$_lifestyle_xp = {
			value = $VALUE$
			divide = 2
		}
	}
	else = {
		add_$LIFESTYLE$_lifestyle_xp = {
			value = $VALUE$
		}
	}
	if = {
		limit = {
			NOT = { has_variable = visited_pois_lifestyle_var }
		}
		set_variable = {
			name = visited_pois_lifestyle_var
			value = 1
		}
	}
	else = {
		change_variable = {
			name = visited_pois_lifestyle_var
			add = 1
		}
	}
}
