	save_temporary_scope_as = merchant_temp
	# Select the Contract Tier
	random_list = {
		10 = {
			save_scope_value_as = {
				name = contract_tier
				value = tier_county
			}
		}
		5 = {
			trigger = {
				exists = scope:destination
				scope:destination.county.prosperity_level >= 10
			}
			save_scope_value_as = {
				name = contract_tier
				value = tier_duchy
			}
		}
		2 = {
			trigger = {
				exists = scope:destination
				scope:destination.county.prosperity_level >= 25
			}
			save_scope_value_as = {
				name = contract_tier
				value = tier_kingdom
			}
		}
	}
}

trade_opportunity_scheme_scopes_effect = {
	save_scope_as = scheme
	scheme_owner = { save_scope_as = owner }
	if = {
		limit = { is_scheme_target_type = character }
		scheme_target_character ?= { save_scope_as = target }
	}
	else = {
		scheme_target_title ?= {
			save_scope_as = target_title
			holder ?= { save_scope_as = target }
		}
	}
	task_contract ?= {
		save_scope_as = task_contract
		task_contract_employer = { save_scope_as = associate }
		task_contract_location = {
			save_scope_as = task_contract_location
			county.holder = { save_scope_as = local_ruler }
			barony.trade_market = { save_scope_as = local_market }
		}
		if = {
			limit = {
				has_task_contract_type = promote_goods
				exists = var:task_contract_trade_good
			}
			var:task_contract_trade_good ?= { save_scope_as = trade_good_type }
		}
	}
}

populate_location_with_trade_opportunities_effect = {
	if = {
		limit = { is_ai = no } # The AI have their own system for getting trade opportunities.
		# Flag our pre-existing contracts if necessary.
		if = {
			limit = { exists = scope:toggle_distance_ignore }
			every_character_task_contract = { add_to_list = pre_existing_trade_opportunities_list }
		}
		# Now populate.
		while = {
			count = $AMOUNT$
			# instead we do it manually
			trigger_event = { on_action = spawn_trade_opportunity_contract_on_action }
		}
		# Then flag our new trade opportunities (again, if necessary).
		if = {
			limit = { exists = scope:toggle_distance_ignore }
			every_character_task_contract = {
				limit = {
					NOT = { is_in_list = pre_existing_trade_opportunities_list }
				}
				set_variable = {
					name = retain_despite_distance
					value = yes
				}
			}
		}
	}
}

replace_stale_trade_opportunities_effect = {
	set_local_variable = {
		name = stale_trade_opportunity_count
		value = 0
	}
	every_character_task_contract = {
		limit = {
			OR = {
				has_task_contract_group = merchant_trade_opportunity_group
				has_task_contract_group = merchant_silk_road_contract_group
			}
			exists = var:task_contract_route
			OR = {
				NOT = { var:task_contract_route.trader ?= task_contract_employer }
				AND = {
					var:task_contract_market ?= { save_temporary_scope_as = contract_market_temp }
					var:task_contract_route = {
						NOT = {
							any_trade_market_on_route = { this = scope:contract_market_temp }
						}
					}
				}
			}
		}
		if = {
			limit = { var:task_contract_route.trader ?= task_contract_employer }
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:market_left_route
			}
		}
		else = {
			set_variable = {
				name = trade_opportunity_lost_reason
				value = flag:trader_changed
			}
		}
		change_local_variable = {
			name = stale_trade_opportunity_count
			add = 1
		}
		clear_trade_opportunity_type_cooldown_effect = yes
		invalidate_contract = yes
	}
	if = {
		limit = { local_var:stale_trade_opportunity_count > 0 }
		populate_location_with_trade_opportunities_effect = { AMOUNT = local_var:stale_trade_opportunity_count }
	}
}

promote_goods_task_contract_pick_trade_good_effect = {
	scope:trade_opportunity ?= {
		var:task_contract_market ?= { save_scope_as = contract_market }
		var:task_contract_route ?= {
			save_scope_as = contract_route
			random_traded_good = {
				limit = {
					trade_good_sold_by_route_in_market_trigger = {
						ROUTE = scope:contract_route
						MARKET = scope:contract_market
					}
					save_temporary_scope_as = route_trade_good_candidate
					scope:contract_market = {
						OR = {
							any_trade_good_with_surplus = { this = scope:route_trade_good_candidate }
							any_trade_good_with_need = { this = scope:route_trade_good_candidate }
						}
					}
				}
				alternative_limit = {
					trade_good_sold_by_route_in_market_trigger = {
						ROUTE = scope:contract_route
						MARKET = scope:contract_market
					}
				}
				save_scope_as = task_contract_trade_good
			}
		}
		if = {
			limit = { exists = scope:task_contract_trade_good }
			set_variable = {
				name = task_contract_trade_good
				value = scope:task_contract_trade_good
			}
		}
	}
}

manage_new_trade_opportunity_contract_effect = {
	scope:trade_opportunity = {
		set_variable = {
			name = task_contract_route
			value = scope:route
		}
		set_variable = {
			name = task_contract_market
			value = scope:market
		}
	}
	scope:other_trade_opportunity ?= {
		set_variable = {
			name = task_contract_route
			value = scope:route
		}
		set_variable = {
			name = task_contract_market
			value = scope:market
		}
	}
	if = {
		limit = { exists = scope:destination }
		if = {
			limit = {
				NOT = { has_character_flag = trade_opportunity_frequency_cooldown }
			}
			add_character_flag = { flag = trade_opportunity_frequency_cooldown years = 6 }
		}
		send_interface_message = {
			type = msg_trade_opportunity_appear
			right_icon = scope:destination.duchy
			localization_values = {
				trade_opportunity = scope:trade_opportunity
				market = scope:market
			}
		}
	}
	if = {
		limit = {
			is_ai = yes
			exists = scope:trade_opportunity
		}
		if = {
			limit = { exists = scope:other_trade_opportunity }
			random_list = {
				10 = {
					modifier = { add = ai_greed } # exploit
					accept_task_contract = scope:trade_opportunity
				}
				10 = {
					modifier = { add = ai_compassion } # alleviate
					accept_task_contract = scope:other_trade_opportunity
				}
			}
		}
		else = { accept_task_contract = scope:trade_opportunity }
	}
	scope:trade_opportunity.task_contract_taker = {
		if = {
			limit = { exists = var:market_agent_opportunity }
			trigger_event = gdp_merchant.3031
		}
	}
	play_sound_effect = "event:/DLC/EP3/SFX/Stingers/Byzantine_Flavor/ep3_mx_sting_byzantineflavor_become_governor" # TODO_CD_GDP
}

alleviate_upheaval_modifier_removal_effect = {
	# SIEGE
	if = {
		limit = { has_trade_market_modifier = gdp_market_recently_sieged_capacity_modifier }
		remove_trade_market_modifier = gdp_market_recently_sieged_capacity_modifier
	}
	else_if = {
		limit = { has_trade_market_modifier = gdp_market_recently_raided_capacity_modifier }
		remove_trade_market_modifier = gdp_market_recently_raided_capacity_modifier
	}
	# ARMY
	if = {
		limit = { has_trade_market_modifier = gdp_market_army_movement_capacity_modifier }
		remove_trade_market_modifier = gdp_market_army_movement_capacity_modifier
	}
	# DISASTER
	if = {
		limit = { has_trade_market_modifier = gdp_natural_disaster_1_trade_modifier }
		remove_trade_market_modifier = gdp_natural_disaster_1_trade_modifier
	}
	else_if = {
		limit = { has_trade_market_modifier = gdp_natural_disaster_2_trade_modifier }
		remove_trade_market_modifier = gdp_natural_disaster_2_trade_modifier
	}
	else_if = {
		limit = { has_trade_market_modifier = gdp_natural_disaster_3_trade_modifier }
		remove_trade_market_modifier = gdp_natural_disaster_3_trade_modifier
	}
	# DISEASE
	if = {
		limit = { has_trade_market_modifier = gdp_epidemic_1_trade_modifier }
		remove_trade_market_modifier = gdp_epidemic_1_trade_modifier
	}
	else_if = {
		limit = { has_trade_market_modifier = gdp_epidemic_2_trade_modifier }
		remove_trade_market_modifier = gdp_epidemic_2_trade_modifier
	}
	else_if = {
		limit = { has_trade_market_modifier = gdp_epidemic_3_trade_modifier }
		remove_trade_market_modifier = gdp_epidemic_3_trade_modifier
	}
}
