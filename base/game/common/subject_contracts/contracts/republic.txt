republic_government_obligations = {
	obligation_levels = {
		default = {
			levies = {
				# Determine base levy
				if = {
					limit = { # Republic lieges take no levy by default
						scope:liege = {
							government_has_flag = government_is_republic
						}
					}
					value = 0
				}
				else = {
					value = 0.1
				}

				# If the liege's culture makes republics provide more levies, increase it accordingly
				if = {
					limit = {
						scope:liege.culture = {
							has_cultural_parameter = republican_vassals_pay_more
						}
						scope:subject = {
							primary_title.tier >= tier_county
						}
					}
					add = 0.05
				}

				# Provide more if liege has unlocked relevant legacy perk(s)
				if = {
					limit = {
						scope:liege.dynasty ?= {
							has_dynasty_perk = fp2_urbanism_legacy_3
						}
					}
					add = 0.15
				}
			}
			tax = {
				# Determine the base tax
				if = { # Tribal lieges have a penalty
					limit = {
						scope:liege = {
							government_has_flag = government_is_tribal
						}
					}
					value = 0.05
				}
				else_if = { # Republic lieges have a higher base than normal
					limit = {
						scope:liege = {
							government_has_flag = government_is_republic
						}
					}
					# Landed vassals pay a bit more compared to landless patricians
					if = {
						limit = {
							scope:subject = {
								is_landed = no
							}
						}
						value = 0.3
					}
					else = {
						value = 0.5
					}
				}
				else = { # Everyone else
					value = 0.4
				}

				# If the liege's culture makes republics pay more, increase taxes accordingly
				if = {
					limit = {
						scope:liege.culture = {
							has_cultural_parameter = republican_vassals_pay_more
						}
						scope:subject = {
							primary_title.tier >= tier_county
						}
					}
					add = 0.1
				}

				# Pay more if liege has unlocked relevant legacy perk(s)
				if = {
					limit = {
						scope:liege.dynasty ?= {
							has_dynasty_perk = fp2_urbanism_legacy_3
						}
					}
					add = 0.15
				}
			}
			contribution_desc = {
				first_valid = {
					triggered_desc = {
						trigger = { scope:liege = { government_has_flag = government_is_tribal } }
						desc = "tribal_liege_inefficiency"
					}
					triggered_desc = {
						trigger = {
							scope:subject = {
								primary_title.tier >= tier_county
							}
							scope:liege.culture = {
								has_cultural_parameter = republican_vassals_pay_more
							}
						}
						desc = "republican_vassals_pay_more_parameter"
					}
				}
			}
		}
	}
}
