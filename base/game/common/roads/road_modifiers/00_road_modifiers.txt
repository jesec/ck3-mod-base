## Shared road modifiers per tier
#
# Applies to every road, based on the tier it is currently at.
# An entry with no 'type' applies to all roads that no other entry claims.
# (road_modifiers_default used by code - do not remove)
#
# Each tier takes up to four blocks, all optional:
#
#	tier_3 = {
#		modifier = { road_construction_speed = 0.1 }				# the road itself
#		province_modifier = { monthly_civic_tax = 0.05 }			# provinces the road covers
#		trade_route_modifier = { trade_capacity_add = 3 }			# routes that follow the road
#		travel_modifier = { travel_speed = 30 }						# anyone travelling along it
#	}
#
# The first three blocks at entry level apply at every tier. travel_modifier is per tier only.
#
