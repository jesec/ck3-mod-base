valid_to_accept_trade_opportunity_trigger = {
	custom_tooltip = {
		text = valid_to_accept_trade_opportunity_trigger
		save_temporary_scope_as = contract_owner_temp
		OR = {
			any_character_active_contract = {
				count = 0
				NOT = { exists = var:contract_partner }
			}
			trigger_if = {
				limit = { exists = court_position:partner_court_position }
				court_position:partner_court_position = {
					NOT = { exists = var:partner_contract }
					is_available_quick = {
						adult = yes
						travel = no
						imprisoned = no
						incapable = no
					}
					is_trader = no
				}
			}
			trigger_else = {
				can_employ_court_position_type = partner_court_position
				any_courtier = {
					would_be_valid_for_court_position = {
						employer = scope:contract_owner_temp
						court_position = partner_court_position
					}
					NOT = { exists = var:partner_contract }
					is_available_quick = {
						adult = yes
						imprisoned = no
						incapable = no
					}
					trigger_if = {
						limit = { is_travelling = yes }
						merchant_trade_opportunity_partner_in_entourage_trigger = { TAKER = scope:contract_owner_temp }
					}
					is_trader = no
				}
			}
		}
	}
}

can_accept_trade_opportunity_while_traveling_trigger = {
	custom_tooltip = {
		text = can_accept_trade_opportunity_while_traveling_trigger
		save_temporary_scope_as = task_contract
		task_contract_taker = {
			save_temporary_scope_as = contract_owner_temp
			location != scope:task_contract.task_contract_location
			trigger_if = {
				limit = { exists = court_position:partner_court_position }
				court_position:partner_court_position = {
					save_temporary_scope_as = partner
					merchant_trade_opportunity_partner_available_trigger = yes
				}
			}
			trigger_else = {
				can_employ_court_position_type = partner_court_position
				any_courtier = {
					would_be_valid_for_court_position = {
						employer = scope:contract_owner_temp
						court_position = partner_court_position
					}
					merchant_trade_opportunity_partner_candidate_available_trigger = { TAKER = scope:contract_owner_temp }
				}
			}
			merchant_trade_opportunity_partner_send_blocked_trigger = no
		}
	}
}

# can a title be purchased in an interaction?
is_title_purchasable_shared_trigger = {
	$TITLE$ ?= { save_temporary_scope_as = title_temp }
	$BUYER$ ?= { save_temporary_scope_as = buyer_temp }
	$SELLER$ ?= { save_temporary_scope_as = seller_temp }
	scope:title_temp ?= {
		# Only counties/duchies
		custom_tooltip = {
			text = purchase_land_tier_tt
			tier >= tier_county
			tier <= tier_duchy
		}
		# Not capital county of seller
		custom_tooltip = {
			text = purchase_land_capital_tt
			this != scope:seller_temp.capital_county
		}
		# Not primary title of seller
		custom_tooltip = {
			text = purchase_land_primary_tt
			this != scope:seller_temp.primary_title
		}
		# Prestige req for duchies
		custom_tooltip = {
			text = purchase_land_duchy_tt
			trigger_if = {
				limit = { tier = tier_duchy }
				scope:buyer_temp.prestige_level >= 3
			}
		}
		# Duchies only allowed if seller completely controls the duchy
		custom_tooltip = {
			text = PURCHASE_LAND_COMPLETE_CONTROL_TT
			trigger_if = {
				limit = { scope:target = { tier = tier_duchy } }
				scope:recipient = {
					completely_controls = scope:target
				}
			}
		}
		# Must be held by seller
		trigger_if = {
			limit = { exists = holder }
			holder = scope:seller_temp
		}
	}
}

is_title_purchase_valid_shared_trigger = {
	$BUYER$ ?= { save_temporary_scope_as = buyer_temp }
	$SELLER$ ?= { save_temporary_scope_as = seller_temp }
	$TITLE$ ?= { save_temporary_scope_as = title_temp }
	scope:seller_temp ?= {
		is_at_war = no
		NOT = { has_strong_hook = scope:buyer_temp }
		trigger_if = {
			limit = {
				any_held_county = { count < 2 }
			}
			custom_description = {
				text = purchase_land_minimum_county_tt
				object = scope:seller_temp
				always = no
			}
		}
	}
	scope:title_temp ?= {
		### TIER
		trigger_if = { # Can't purchase Baronies
			limit = { tier = tier_barony }
			custom_description = {
				text = purchase_land_barony_tt
				always = no
			}
		}
		trigger_if = { # Can't purchase Kingdoms or Empires
			limit = { tier >= tier_kingdom }
			custom_description = {
				text = purchase_land_kingdom_tt
				always = no
			}
		}
		trigger_if = { # Can't purchase HoF titles
			limit = { is_head_of_faith = yes }
			custom_description = {
				text = purchase_land_religious_tt
				always = no
			}
		}
		### TYPE
		trigger_if = { # Can't purchase nf titles
			limit = { is_noble_family_title = yes }
			custom_description = {
				text = purchase_land_noble_family_tt
				always = no
			}
		}
		trigger_if = { # Can't purchase laamp titles
			limit = { has_variable = adventurer_creation_reason }
			custom_description = {
				text = purchase_land_landless_adventurer_tt
				always = no
			}
		}
		trigger_if = { # Can't purchase merchant titles
			limit = { has_variable = merchant_creation_reason }
			custom_description = {
				text = purchase_land_merchant_tt
				always = no
			}
		}
		trigger_if = { # Can't purchase laamp titles
			limit = { is_landless_type_title = yes }
			always = no
		}
	}
}

republic_can_grant_county_to_landed_trigger = {
	custom_tooltip = {
		text = republic_title_must_be_neighboring_or_de_jure_vassal_desc
		OR = {
			# Holds county neighboring target county
			scope:target = {
				any_title_to_title_neighboring_county = { holder = scope:recipient }
			}
			scope:recipient = { 
				OR = {
					# Holds duchy or higher that target county belongs to
					any_held_title = { is_de_jure_liege_or_above_target = scope:target }
					# Holds barony that belongs to target county
					AND = { 
						highest_held_title_tier = tier_barony
						capital_province.county = scope:target
					}
				}
			}
		}
	}
}

					count > 0
				}
			}
			scope:target = {

					any_in_de_jure_hierarchy = {
						tier = tier_county

				}

		}

	}
