burgher_liege_valid_trigger = {
	has_gdp_dlc_trigger = yes

	OR = {
		AND = {
			highest_held_title_tier >= tier_kingdom
			is_independent_ruler = yes
			government_has_flag = government_is_feudal
		}
		government_has_flag = government_is_ecclesiastical
	}
}

burgher_liege_can_be_crowned_trigger = {
	highest_held_title_tier >= tier_kingdom
	NOR = {
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_republic
	}
}

burgher_county_is_urban_trigger = {
	prosperity_level >= burgher_faction_min_prosperity_value
	OR = {
		any_county_province = {
			has_holding_type = city_holding
		}
		prosperity_level >= burgher_faction_rich_prosperity_value
	}
}

burgher_county_can_join_trigger = {
	prosperity_level >= burgher_faction_join_prosperity_value
	OR = {
		any_county_province = {
			has_holding_type = city_holding
		}
		prosperity_level >= burgher_faction_min_prosperity_value
	}
}

burgher_county_founder_faith_compatible_trigger = {
	trigger_if = {
		limit = {
			exists = scope:faction.var:burgher_county.faith
		}
		NOT = {
			faith ?= {
				faith_hostility_level = {
					target = scope:faction.var:burgher_county.faith
					value >= faith_hostile_level
				}
			}
		}
	}
	trigger_if = {
		limit = {
			exists = scope:faction.var:burgher_county.rite
		}
		NOT = {
			rite ?= {
				rite_hostility_level = {
					target = scope:faction.var:burgher_county.rite
					value >= faith_hostile_level
				}
			}
		}
	}
}

burgher_county_faction_on_cooldown_trigger = {
	has_variable = burgher_faction_cooldown
}

burgher_county_has_own_faction_option_trigger = {
	save_temporary_scope_as = burgher_option_county
	OR = {
		AND = {
			exists = holder
			can_title_create_faction = {
				type = burgher_autonomy_faction
				target = holder
			}
		}
		AND = {
			exists = holder
			holder = {
				any_targeting_faction = {
					faction_is_type = burgher_autonomy_faction
					scope:burgher_option_county = { can_title_join_faction = prev }
				}
			}
		}
	}
}

	is_alive = yes
	is_adult = yes
	is_landed = no
	is_imprisoned = no
	is_incapable = no
	NOT = { is_in_list = burgher_new_communes }
}

burgher_merchant_petitioner_trigger = {
	is_domicile_type = merchant_company
	owner ?= {
		burgher_can_head_commune_trigger = yes
		NOT = { $FACTION$.var:merchant_sponsor ?= this }
	}
}

burgher_privilege_tier_2_trigger = {
	title_province ?= {
		province_size_level >= burgher_privilege_tier_2_size_value
	}
}

burgher_privilege_tier_3_trigger = {
	title_province ?= {
		province_size_level >= burgher_privilege_tier_3_size_value
	}
}

burgher_county_holds_privilege_trigger = {
	OR = {
		has_county_modifier = civic_privilege_$PRIVILEGE$_1_county_modifier
		has_county_modifier = civic_privilege_$PRIVILEGE$_2_county_modifier
		has_county_modifier = civic_privilege_$PRIVILEGE$_3_county_modifier
	}
}

burgher_lacks_privilege_tier_trigger = {
	NOR = {
		has_county_modifier = civic_privilege_$PRIVILEGE$_3_county_modifier
		AND = {
			has_county_modifier = civic_privilege_$PRIVILEGE$_2_county_modifier
			burgher_privilege_tier_3_trigger = no
		}
		AND = {
			has_county_modifier = civic_privilege_$PRIVILEGE$_1_county_modifier
			burgher_privilege_tier_2_trigger = no
		}
	}
}

burgher_county_has_privileges_trigger = {
	OR = {
		has_county_modifier = civic_privileges_county_modifier
		burgher_county_holds_privilege_trigger = { PRIVILEGE = market_rights }
		burgher_county_holds_privilege_trigger = { PRIVILEGE = guild_rights }
		burgher_county_holds_privilege_trigger = { PRIVILEGE = tax_exemption }
		burgher_county_holds_privilege_trigger = { PRIVILEGE = militia_charter }
	}
}

	liege.highest_held_title_tier > tier_county

	}
}
