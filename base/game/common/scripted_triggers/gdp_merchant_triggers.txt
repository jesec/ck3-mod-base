# Shared with can_be_trader scripted rule (00_rules.txt)

	is_available_quick = {
		alive = yes
		adult = yes
		incapable = no
		imprisoned = no
	}
	# Merchants have no council seat besides the spouse's
	trigger_if = {
		limit = {

				liege ?= { government_has_flag = government_is_merchant }
			}
		}

			text = gdp_can_be_trader_council_tt
			OR = {

			}
		}
	}

		text = gdp_can_be_trader_partner_tt
		NOT = { has_court_position = partner_court_position }
	}
	NOT = { exists = var:partner_contract }
}

# Trigger for the Become Merchant Decision
gdp_is_valid_for_merchant_trigger = {
	save_temporary_scope_as = potential_merchant_temp
	trigger_if = {
		limit = {
			# If you are a Patrician you should absolutely have this option
			NOT = { government_allows = profit_trade_routes }
		}
		trigger_if = {
			limit = {
				OR = {
					is_ai = yes
					NOT = { government_has_mechanic = administrative }
					NOT = { top_liege = this }
					is_landed = yes
				}
			}
			custom_tooltip = {
				text = adventurer_not_nf_head_trigger_tt
				NOT = {
					any_held_title = { is_noble_family_title = yes }
				}
			}
		}
	}
	# Not in line for ceremonial liege
	trigger_if = {
		limit = { tgp_realm_has_ceremonial_liege_trigger = yes }
		custom_tooltip = {
			text = tgp_japan_imperial_branch_decision_imperial_heir_trigger
			NOT = {
				ceremonial_liege_title = {
					place_in_line_of_succession = {
						target = scope:potential_merchant_temp
						value <= 4
					}
				}
			}
		}
	}
	trigger_if = {
		limit = { is_ai = yes }
		is_available_quick = {
			adult = yes
			incapable = no
			imprisoned = no
		}
		age < 60 # Won't immediately die
		health >= fine_health # Healthy enough to venture forth
		highest_skill_value >= decent_skill_rating
		# Can't be completely unwilling to go outside.
		OR = {
			ai_boldness >= low_negative_ai_value
			ai_energy >= low_negative_ai_value
		}
		NOR = {
			has_trait = content # Too chill to trade
			has_trait = craven # Too scared to trade
			has_trait = lazy # Too lazy to trade
		}
		# Block admin members of noble families
		NOT = {
			house ?= {
				any_house_member = {
					any_held_title = { is_noble_family_title = yes }
				}
			}
			NOT = { government_allows = profit_trade_routes }
		}
	}
}

# As gdp_profit_potential_rival_trigger, but unlanded Patricians also count. Contested market checks must keep using the merchant-only version.

		ROUTE_OWNER = $ROUTE_OWNER$
		MARKET = $MARKET$

market_potential_rival_trigger = {
	exists = house
	NOR = {
		this = $ROUTE_OWNER$
		house ?= $ROUTE_OWNER$.house
	}
	potential_trade_rival = {
		target = $ROUTE_OWNER$
		trade_market = $MARKET$
	}
}

# this = trade market that is still contested for $ROUTE_OWNER$:
# - a profit rival trades in this market
# - no complete monopoly exists
# - $ROUTE_OWNER$ holds no monopoly over it
gdp_market_competitiveness_contested_trigger = {
	save_temporary_scope_as = contested_market_temp
	any_character_in_market = {
		gdp_profit_potential_rival_trigger = {
			ROUTE_OWNER = $ROUTE_OWNER$
			MARKET = scope:contested_market_temp
		}
	}
	NOR = {
		has_complete_trade_monopoly = yes
		any_trade_monopoly_for_market = {
			OR = {
				character_owner ?= $ROUTE_OWNER$
				AND = {
					exists = title_owner
					title_owner.holder ?= $ROUTE_OWNER$
				}
				AND = {
					exists = $ROUTE_OWNER$.confederation
					exists = confederation_owner
					confederation_owner ?= $ROUTE_OWNER$.confederation
				}
			}
		}
	}
}

# Merchant diplo range
gdp_merchant_diplo_range_trigger = {
	$MERCHANT$ = {
		save_temporary_scope_as = merchant_temp
		scope:merchant_temp.domicile.domicile_location ?= { save_temporary_scope_as = company_temp }
	}
	$TARGET$ = {
		save_temporary_scope_as = ruler_temp
		trigger_if = {
			limit = { exists = capital_province }
			capital_province ?= { save_temporary_scope_as = capital_temp }
		}
		trigger_else_if = {
			limit = { exists = host.capital_province }
			host.capital_province ?= { save_temporary_scope_as = capital_temp }
		}
		trigger_else = {
			location ?= { save_temporary_scope_as = capital_temp }
		}
	}
	custom_description = {
		text = gdp_merchant_diplo_range_trigger_tt
		subject = scope:merchant_temp
		object = scope:ruler_temp
		scope:company_temp ?= {
			trigger_if = { # Check range
				limit = { exists = scope:capital_temp }
				squared_distance = {
					target = scope:capital_temp
					value < 1000000 # more than adventurers
				}
			}
			trigger_else = { always = no }
		}
	}
}

		OR = {
			is_ruler = yes
			is_theocratic_lessee = yes
		}
		is_available_quick = {
			adult = yes
			alive = yes
			imprisoned = no
			incapable = no
		}
	}
	# LOCATION CHECKS
	var:route.trader ?= task_contract_target
	NOT = { task_contract_location.county.duchy = task_contract_taker.capital_province.county.duchy }
	task_contract_location.barony.trade_market = {

suzerain_tier_to_company_size_shared_valid_trigger = {
	trigger_if = {
		limit = { suzerain ?= { highest_held_title_tier >= tier_empire } }
		custom_tooltip = {
			text = merchant_charter_company_size_requirement_empire_tt
			has_realm_law = company_size_huge
		}
	}
	trigger_else_if = {
		limit = { suzerain ?= { highest_held_title_tier >= tier_kingdom } }
		custom_tooltip = {
			text = merchant_charter_company_size_requirement_kingdom_tt
			OR = {
				has_realm_law = company_size_large
				has_realm_law = company_size_huge
			}
		}
	}
	trigger_else_if = {
		limit = { suzerain ?= { highest_held_title_tier >= tier_duchy } }
		custom_tooltip = {
			text = merchant_charter_company_size_requirement_duchy_tt
			OR = {
				has_realm_law = company_size_medium
				has_realm_law = company_size_large
				has_realm_law = company_size_huge
			}
		}
	}
	trigger_else = { always = yes } # other validity checks should confirm

}

exceptional_diplo_range_trigger = {
	#limit range for landless adventurers
	trigger_if = {
		limit = {
			$CHARACTER$ = { has_government = landless_adventurer_government }
		}
		#Has to be used instead of diplo range checks in laamp to landed interactions
		ep3_laamp_diplo_range_trigger = {
			TARGET = $TARGET$
			LAAMP = $CHARACTER$
		}
	}
	# limit range for merchants
	trigger_if = {
		limit = {
			$CHARACTER$ = {
				has_government = merchant_government
			}
		}
		gdp_merchant_diplo_range_trigger = {
			TARGET = $TARGET$
			MERCHANT = $CHARACTER$
		}
	}
}
