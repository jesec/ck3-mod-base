# Trigger for when assembly approval is required
gdp_requires_assembly_approval_trigger = {
	OR = {
		has_realm_law = republic_assembly_authority_constitutional
		has_realm_law = republic_assembly_authority_legislative
	}
}

gdp_can_pass_republic_law_trigger = {
	# If we have a council assembly with voting powers, no council position can be left empty
	trigger_if = {
		limit = {
			has_realm_law = republic_assembly_composition_council
			NOT = { has_realm_law = republic_assembly_authority_ceremonial }
		}
		has_free_council_slot = no
	}
	# If we have a senate, the number of senators much be at least 90% of the total senate size
	trigger_if = {
		limit = {
			has_realm_law_flag = republic_senate_size
			NOT = { has_realm_law = republic_assembly_authority_ceremonial }
		}
		gdp_republic_senate_has_enough_senators_trigger = yes
	}
}

gdp_can_propose_law_changes_trigger = {
	OR = {
		# Republican leader may always propose changes
		this = top_liege
		liege ?= {
			NOT = { government_has_flag = government_is_republic }
		}
		# Major members may suggest changes when the assembly has voting powers
		trigger_if = {
			limit = {
				liege ?= {
					NOT = { has_realm_law = republic_assembly_authority_ceremonial }
				}
			}
			is_major_assembly_member = yes
		}
	}
}

gdp_assembly_approval_for_decrees_trigger = {
	highest_liege_matching_rule:uses_assembly = {
		has_realm_law = republic_assembly_authority_constitutional
	}
}

gdp_assembly_can_pass_decrees_trigger = {
	# Check first if assembly approval is required
	trigger_if = {
		limit = {
			gdp_assembly_approval_for_decrees_trigger = yes
		}
		highest_liege_matching_rule:uses_assembly = {
			# If we have a council assembly with voting powers, no council position can be left empty
			trigger_if = {
				limit = {
					has_realm_law = republic_assembly_composition_council
				}
				has_free_council_slot = no
			}
			# If we have a senate, the number of senators much be at least 90% of the total senate size
			trigger_if = {
				limit = {
					has_realm_law_flag = republic_senate_size
				}
				gdp_republic_senate_has_enough_senators_trigger = yes
			}
		}
	}
}

# Does the senate have enough senators for approval? (republic head is the scoped character)
gdp_republic_senate_has_enough_senators_trigger = {
	# Safety check to ensure that we have a senate
	trigger_if = {
		limit = {
			has_realm_law_flag = republic_senate_size
		}
		save_temporary_scope_value_as = {
			name = needed_senate_size
			value = {
				value = republic_senate_size_max_value
				multiply = 0.8 # Must have at least 80% of the senator seats filled
				ceiling = yes # Rounded up
			}
		}
		custom_tooltip = {
			text = must_have_at_least_senator_limit_desc
			primary_title ?= {
				any_full_assembly_member = {
					count >= scope:needed_senate_size
				}
			}
		}
	}
}

#  Does the senate have any empty seats? (republic head is the scoped character)
gdp_republic_senate_has_empty_seats_trigger = {
	# Safety check to ensure that we have a senate
	trigger_if = {
		limit = {
			has_realm_law_flag = republic_senate_size
		}
		save_temporary_scope_value_as = {
			name = max_senate_size
			value = republic_senate_size_max_value
		}
		custom_tooltip = {
			text = republic_senate_is_full_desc
			primary_title ?= {
				any_full_assembly_member = {
					count < scope:max_senate_size
				}
			}
		}
	}
}

				}
			}

	}
	trigger_if = {
		limit = {
			scope:roman_liege ?= {
				OR = {
					has_realm_law = female_only_law
					has_realm_law = female_preference_law
				}
			}
		}
		is_female = yes
	}
	is_adult = yes
	is_imprisoned = no
	is_lowborn = no
	NOT = { has_character_flag = republic_appointed_senator }
}

			house ?= {
				house_power_score >= gdp_patrician_house_score_threshold_2_value

		}
	}
	trigger_else_if = {
		limit = {
			$VALUE$ = 3
		}
		custom_description = {
			text = has_patrician_rating_threshold_desc
			object = scope:current_house_head
			value = gdp_patrician_house_score_threshold_3_value
			house ?= {
				house_power_score >= gdp_patrician_house_score_threshold_3_value

# Can the scoped character use the Free City obligation on republic vassals?
gdp_can_have_free_cities_trigger = {
	# Must be an independent emperor or hegemon
	this = top_liege
	highest_held_title_tier >= tier_empire
	# But is not a republic
	NOT = { government_has_flag = government_is_republic }
	# Must have access to free cities somehow
	OR = {
		# Always true for the HRE
		has_title = title:e_hre
		# Cultural traditions
		culture ?= { has_cultural_tradition = tradition_parochialism }
	}
}

# Is the scope character part of the same trade league as the target character
is_in_same_trade_league_as_trigger = {
	# Scopes
	$TARGET_CHARACTER$ = { save_temporary_scope_as = target_league_character }
	# This character is in a trade league
	government_has_flag = government_is_republic
	highest_liege_matching_rule:uses_assembly ?= {
		exists = primary_title.title_confederation
		primary_title.title_confederation = {
			confederation_type = confederation_type:trade_league
			save_temporary_scope_as = shared_trade_league
		}
	}
	# Target character is in the same trade league
	scope:target_league_character = {
		government_has_flag = government_is_republic # For safety
		highest_liege_matching_rule:uses_assembly.primary_title.title_confederation ?= { this = scope:shared_trade_league }
	}
}

# Valid commune CB target for $LEADER$'s league
gdp_league_commune_target_valid_trigger = {
	is_landless_type_title = no
	custom_tooltip = {
		text = league_commune_target_not_republican_tt
		holder ?= {
			NOT = { government_has_flag = government_is_republic }
		}
	}
	# City or castle required: other holdings cannot convert to a republic
	custom_tooltip = {
		text = league_commune_target_holding_tt
		title_province ?= {
			OR = {
				has_holding_type = city_holding
				has_holding_type = castle_holding
			}
		}
	}
	custom_tooltip = {
		text = league_commune_target_size_tt
		title_province ?= {
			province_size_level >= 2
		}
	}
	custom_tooltip = {
		text = league_commune_target_reach_tt
		OR = {
			is_coastal_county = yes
			is_neighbor_to_realm = $LEADER$
			AND = {
				exists = $LEADER$.confederation
				any_neighboring_county = {
					holder ?= {
						is_member_of_confederation = $LEADER$.confederation
					}
				}
			}
		}
	}
}

# $LEAGUE$ trades here, outside $RIVAL_LEAGUE$'s seats

# Holds assets $LEAGUE$ can contest
gdp_league_holds_contested_trade_assets_trigger = {
	OR = {
		any_owned_trade_post_district = {
			district_province.barony.trade_market ?= {
				gdp_league_contested_market_trigger = {
					LEAGUE = $LEAGUE$
					RIVAL_LEAGUE = $RIVAL_LEAGUE$
				}
			}
		}
		any_trade_monopoly_for_character = {
			monopoly_trade_market ?= {
				gdp_league_contested_market_trigger = {
					LEAGUE = $LEAGUE$
					RIVAL_LEAGUE = $RIVAL_LEAGUE$
				}
			}
		}
		any_held_title = {
			any_trade_monopoly_for_title = {
				monopoly_trade_market ?= {
					gdp_league_contested_market_trigger = {
						LEAGUE = $LEAGUE$
						RIVAL_LEAGUE = $RIVAL_LEAGUE$
					}
				}
			}
		}
	}
}

# Yes when a character can adopt the republic government
# This shows the Adopt Republican Ideals decision and the Request City Charter interaction
gdp_basic_can_adopt_republic_government_trigger = {
	has_gdp_dlc_trigger = yes
	# You are not one of the following governments
	NOR = {
		government_has_flag = government_is_republic
		government_has_flag = government_is_tribal
		government_has_flag = government_has_herd
		government_has_flag = government_is_theocracy
		government_is_japanese_trigger = yes
	}
	# Your realm isn't already too large, at which point we hide this altogether
	sub_realm_size < 75
	# You own land
	is_landed = yes
	# Is duchy tier or less
	highest_held_title_tier <= tier_duchy
}
