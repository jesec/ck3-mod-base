#Scripted Modifiers used for buildings

# Scales positive in-tier context adds so they reorder buildings within a priority tier
@ai_building_in_tier_adjustment_scale = 0.1

ai_rural_building_modifier = {
	modifier = { # Rural identity, keep building the rural districts
		add = ai_building_priority_medium
		ai_urban_identity_trigger = no
	}
	modifier = {
		factor = ai_building_identity_mismatch_factor
		ai_urban_identity_trigger = yes
	}
}

ai_rural_synergy_modifier = {
	modifier = { # Manor houses and religious community build on the villages a rural province already has
		add = {
			value = ai_building_priority_strong
			if = {
				limit = {
					NOR = {
						province_has_district = manor_houses_district
						province_has_district = religious_community_district
					}
				}
				add = ai_building_priority_soft
			}
		}
		ai_urban_identity_trigger = no
	}
}

ai_urban_building_modifier = {
	modifier = { #Prioritize this where the size is higher - it is an urban place 
		add = {
			if = {
				limit = {
					province_size > 200
				}
				add = {
					value = province_size
					divide = 10
					multiply = @ai_building_in_tier_adjustment_scale
				}
			}
			else = {
				subtract = 50
			}
		}
	}
}

ai_opinion_baseline_building_modifier = {
	modifier = { # Prioritize places with low popular opinion

		add = {
			value = 0
			if = {
				limit = {
					county.base_popular_opinion < 25
				}
				add = 20
			}
			else_if = {
				limit = {
					county.base_popular_opinion < 50
				}
				add = 10
			}
			else_if = {
				limit = {
					county.base_popular_opinion < 75
				}
				add = 5
			}
			multiply = {
				value = $VALUE$
				divide = ai_building_urgency_reference_grant
				min = 0.4
				max = 2
			}
			multiply = @ai_building_in_tier_adjustment_scale
		}
	}
}
