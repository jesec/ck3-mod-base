merchant_find_curio_poi_modifier = {
	modifier = {
		exists = scope:find_curio
		add = {
			value = 0
			every_county_in_region = {
				region = $REGION$
				limit = {
					any_county_province = { gdp_find_curio_poi_province_trigger = yes }
				}
				add = 5
			}
		}
	}
}

	modifier = {
		$MERCHANT$ = { save_temporary_scope_as = merchant_temp }

			any_trade_category = {

				has_trade_good_category_xp_track = yes
				"scope:merchant_temp.primary_title.trade_good_category_level(scope:category_temp)" < $BELOW$
			}
		}
		add = $WEIGHT$
	}
}

diversify_commission_route_market_need_modifier = {
	modifier = { # Weight from category needs
		save_temporary_scope_as = market_temp
		add = {
			every_trade_category_with_unfulfilled_need = {
				limit = { save_temporary_scope_as = category_temp }
				add = "scope:market_temp.trade_category_unfulfilled_need(scope:category_temp)"
			}
			multiply = 0.01
		}
	}
	modifier = { # Weight from specific good needs
		save_temporary_scope_as = market_temp
		add = {
			every_trade_good_with_unfulfilled_need = {
				limit = { save_temporary_scope_as = target_temp }
				add = {
					value = "scope:market_temp.trade_good_unfulfilled_need(scope:target_temp)"
					divide = 2 # offset from category need
				}
			}
			multiply = 0.01
		}
	}
}

diversify_commission_route_category_routes_modifier = {
	modifier = {
		save_temporary_scope_as = category_temp
		factor = {
			value = 1
			scope:commissioner ?= {
				every_tributary = {
					limit = { government_has_flag = government_is_merchant }
					every_trade_route = {
						limit = {
							is_commission_route = yes
							commissioner ?= scope:commissioner
							any_traded_good = {
								any_trade_category = { this = scope:category_temp }
							}
						}
						divide = 2
					}
				}
			}
			min = 0.1
		}
	}
}
