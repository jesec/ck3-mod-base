# 15. In Sweden we have a System...
# Wine and Beer share the Alcohol category, so one expertise track and one monopoly cover both
ep5_15_in_sweden_we_have_a_system_achievement = {
	possible = {
		has_gdp_dlc_trigger = yes
	}
	happened = {
		custom_description = {
			text = ep5_15_in_sweden_we_have_a_system_achievement_trigger
			any_held_title = {
				trade_good_category_level = {
					target = trade_good_category:trade_category_alcohol
					value >= merchant_monopoly_trade_expertise_level_max_value
				}
			}
			capital_barony.trade_market ?= {
				any_trade_monopoly_for_market = {
					is_trade_monopoly_for_trade_good_category = trade_category_alcohol
					ep5_achievement_monopoly_owned_by_trigger = { OWNER = root }
				}
			}
		}
	}
}
