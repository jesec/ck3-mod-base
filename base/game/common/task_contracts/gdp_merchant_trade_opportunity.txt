challenge_competitor = {
	group = merchant_trade_opportunity_group
	icon = "gfx/interface/icons/scheme_types/icon_scheme_challenge_status.dds"
	travel = no
	use_diplomatic_range = no

	weight = { value = challenge_competitor_trade_opportunity_weight_value }

	valid_to_create = {
		valid_merchant_contract_basic_trigger = yes
		NOR = {
			any_character_task_contract = { has_task_contract_type = challenge_competitor }
			has_character_flag = trade_opportunity_cooldown_challenge_competitor
		}
		any_trade_route = {
			has_trader = yes
			current_capacity < max_capacity
			any_trade_market_on_route = {
				gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
				gdp_market_competitiveness_contested_trigger = { ROUTE_OWNER = root }
				NOR = {
					has_competitiveness_modifier_for_character = {
						character = root
						modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
					}
					has_competitiveness_modifier_for_character = {
						character = root
						modifier = gdp_challenge_competitor_trade_opportunity_success_modifier
					}
					has_competitiveness_modifier_for_character = {
						character = root
						modifier = gdp_challenge_competitor_trade_opportunity_failure_modifier
					}
				}
			}
		}
	}

	valid_to_keep = {
		valid_trade_opportunity_to_continue_trigger = yes
		task_contract_target ?= {
			gdp_is_merchant_or_unlanded_patrician_trigger = yes
			house != root.task_contract_taker.house
		}
	}

	valid_to_accept = {
		valid_merchant_contract_basic_trigger = yes
		valid_to_accept_trade_opportunity_trigger = yes
	}

	can_accept_while_traveling = {
		can_accept_trade_opportunity_while_traveling_trigger = yes
	}

	valid_to_continue = {
		valid_trade_opportunity_to_continue_trigger = yes
		task_contract_target.house.house_head ?= {
			gdp_is_merchant_or_unlanded_patrician_trigger = yes
			house != root.task_contract_taker.house
		}
		var:task_contract_route ?= {
			any_trade_market_on_route = { trade_market_owner.title_province = root.task_contract_location }
		}
	}

	on_create = {
		scope:contract.task_contract_taker ?= { # Set a cooldown
			add_character_flag = {
				flag = trade_opportunity_cooldown_challenge_competitor
				years = 10
			}
		}
	}

	on_accepted = {
		trade_opportunity_on_accepted_effect = yes
	}

	on_completed = {
		merchant_trade_opportunity_travel_complete_or_invalidated_effect = yes # required to ensure travel does not get stuck
	}

	on_invalidated = {
		merchant_trade_opportunity_contract_invalidated_message_effect = yes
		merchant_trade_opportunity_travel_complete_or_invalidated_effect = yes # required to ensure travel does not get stuck
	}

	task_contract_reward = {
		success_standard = {
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract"
					add_prestige = {
						value = minor_prestige_gain
						multiply = trade_opportunity_contract_tier_value
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_value
					trade_opportunities_competitive_pressure_success_effect = yes
					merchant_binding_contracts_trade_opportunity_success_effect = yes
				}
				var:task_contract_market ?= {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_challenge_competitor_trade_opportunity_success_modifier
						years = 10
					}
				}
				scope:task_contract.task_contract_taker = { save_scope_as = char }
				scope:task_contract.task_contract_target = { save_scope_as = target_char }
				task_contract_target.house = {
					if = {
						limit = {
							NOT = { has_house_relation_with = scope:task_contract.task_contract_taker.house }
						}
						set_house_relation = {
							target = scope:task_contract.task_contract_taker.house
							level = quarrel
							description = house_relation_reason_challenge_competitor_decrease_desc
						}
					}
					else = {
						change_house_relation_effect = {
							HOUSE = scope:task_contract.task_contract_taker.house
							VALUE = house_relation_damage_minor_value
							REASON = challenge_competitor_decrease
							CHAR = scope:task_contract.task_contract_taker
							TARGET_CHAR = scope:task_contract.task_contract_target
							TITLE = scope:dummy_gender
						}
					}
				}
			}
		}
		success_critical = {
			should_print_on_complete = yes
			visible = no
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract"
					add_prestige = {
						value = medium_prestige_gain
						multiply = trade_opportunity_contract_tier_value
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_critical_value
					trade_opportunities_competitive_pressure_critical_success_effect = yes
					merchant_binding_contracts_trade_opportunity_success_effect = yes
					merchant_port_factors_trade_opportunity_success_effect = yes
				}
				var:task_contract_market ?= {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
						years = 10
					}
				}
				scope:task_contract.task_contract_taker = { save_scope_as = char }
				scope:task_contract.task_contract_target = { save_scope_as = target_char }
				task_contract_target.house = {
					add_house_modifier = {
						modifier = gdp_challenge_competitor_trade_opportunity_success_critical_modifier
						years = 10
					}
					if = {
						limit = {
							NOT = { has_house_relation_with = scope:task_contract.task_contract_taker.house }
						}
						set_house_relation = {
							target = scope:task_contract.task_contract_taker.house
							level = cordial
							description = house_relation_reason_challenge_competitor_increase_desc
						}
					}
					else = {
						change_house_relation_effect = {
							HOUSE = scope:task_contract.task_contract_taker.house
							VALUE = house_relation_improve_minor_value
							REASON = challenge_competitor_increase
							CHAR = scope:task_contract.task_contract_taker
							TARGET_CHAR = scope:task_contract.task_contract_target
							TITLE = scope:dummy_gender
						}
					}
				}
			}
		}
		failure_standard = {
			positive = no
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract"
					merchant_contract_failure_reputation_effect = yes
					trade_opportunities_competitive_pressure_failure_effect = yes
				}
				var:task_contract_market ?= {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_challenge_competitor_trade_opportunity_failure_modifier
						years = 10
					}
				}
				scope:task_contract.task_contract_taker = { save_scope_as = char }
				scope:task_contract.task_contract_target = { save_scope_as = target_char }
				task_contract_target.house = {
					if = {
						limit = {
							NOT = { has_house_relation_with = scope:task_contract.task_contract_taker.house }
						}
						set_house_relation = {
							target = scope:task_contract.task_contract_taker.house
							level = quarrel
							description = house_relation_reason_challenge_competitor_decrease_desc
						}
					}
					else = {
						change_house_relation_effect = {
							HOUSE = scope:task_contract.task_contract_taker.house
							VALUE = house_relation_damage_minor_value
							REASON = challenge_competitor_decrease
							CHAR = scope:task_contract.task_contract_taker
							TARGET_CHAR = scope:task_contract.task_contract_target
							TITLE = scope:dummy_gender
						}
					}
				}
			}
		}
	}
}
