patron_hobnob = {
	group = merchant_patron_contract_group
	icon = "gfx/interface/icons/message_feed/merchant.dds" # TODO_CD_GDP
	travel = yes
	use_diplomatic_range = no

	weight = { value = 1 }

	valid_to_create = {
		valid_merchant_contract_basic_trigger = yes
		suzerain ?= {
			is_available_quick = {
				adult = yes
				travel = no
				imprisoned = no
				in_army = no
				incapable = no
			}
		}
		NOR = {
			any_character_task_contract = { has_task_contract_type = patron_hobnob }
			has_character_flag = patron_contract_cooldown_hobnob
		}
	}

	valid_to_keep = {
		task_contract_taker = {
			valid_merchant_contract_basic_trigger = yes
			suzerain ?= {
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				exists = capital_province
			}
		}
		trigger_if = {
			limit = { exists = task_contract_employer }
			task_contract_employer ?= task_contract_taker.suzerain
			task_contract_location ?= task_contract_taker.suzerain.capital_province
		}
	}

	valid_to_accept = {
		valid_merchant_contract_basic_trigger = yes
		merchant_trade_opportunity_valid_to_accept_contract_limit_trigger = yes
	}

	valid_to_continue = {
		save_temporary_scope_as = task_contract
		task_contract_taker = {
			valid_merchant_contract_basic_trigger = yes
			suzerain ?= {
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				exists = capital_province
			}
		}
		task_contract_employer ?= task_contract_taker.suzerain
		task_contract_location ?= task_contract_taker.suzerain.capital_province
	}
	
	on_create = {
		scope:contract.task_contract_taker ?= { # Set a cooldown
			add_character_flag = {
				flag = patron_contract_cooldown_hobnob
				years = 10
			}
		}
	}

	on_accepted = {
		task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_accept_contract" }
		save_scope_as = task_contract # save contract
		task_contract_location.barony = { save_scope_as = task_contract_location } # save location
		task_contract_employer = {
			save_scope_as = task_contract_employer # save target
			save_scope_as = patron
		}
		task_contract_taker = {
			start_scheme = {
				type = merchant_patron_hobnob_patron_contract_scheme
				contract = scope:task_contract
				target_character = scope:task_contract_employer
			}
			scope:task_contract.scheme = { save_scope_as = task_contract_scheme } # save scheme
			trigger_event = gdp_merchant_patron_contract.0001
		}
	}

	on_invalidated = {
		if = {
			limit = { merchant_trade_opportunity_invalidated_message_trigger = yes }
			merchant_patron_contract_invalidated_message_effect = yes
		}
	}

	task_contract_reward = {
		success_standard = {
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				# Standard Rewards.
				task_contract_taker = {
					add_prestige = {
						value = medium_prestige_gain
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_value
					reverse_add_opinion = {
						modifier = friendliness_opinion
						target = suzerain
						opinion = 10
					}
				}
				task_contract_employer.capital_county.title_province.barony.trade_market = {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_impressed_patron_modifier
						years = 10
					}
				}
			}
		}
		success_critical = {
			should_print_on_complete = yes
			visible = no
			effect = {
				save_scope_as = task_contract
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				# Critical Rewards.
				task_contract_taker = {
					add_prestige = {
						value = major_prestige_gain
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_critical_value
					reverse_add_opinion = {
						modifier = friendliness_opinion
						target = suzerain
						opinion = 20
					}
				}
				task_contract_employer.capital_county.title_province.barony.trade_market = {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_impressed_patron_critical_modifier
						years = 10
					}
				}
			}
		}
		failure_standard = {
			positive = no
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract" }
				# Standard Penalties.
				task_contract_taker = {
					add_stress = {
						value = minor_stress_gain
						multiply = trade_opportunity_contract_tier_value
					}
					reverse_add_opinion = {
						modifier = insulted_opinion
						target = suzerain
						opinion = -10
					}
				}
				task_contract_employer.capital_county.title_province.barony.trade_market = {
					add_competitiveness_modifier_for_character = {
						character = scope:task_contract.task_contract_taker
						modifier = gdp_dismissive_patron_modifier
						years = 10
					}
				}
			}
		}
	}
}
