silk_road_network = { # PAIRED: silk_road_caravan 
	group = merchant_silk_road_contract_group
	icon = "gfx/interface/icons/situation_types/silk_road_situation.dds"
	travel = no
	use_diplomatic_range = no

	weight = { value = silk_road_network_trade_opportunity_weight_value }

	valid_to_create = {
		valid_merchant_contract_basic_trigger = yes
		NOR = {
			any_character_task_contract = {
				OR = {
					has_task_contract_type = silk_road_network
					has_task_contract_type = silk_road_caravan
				}
			}
			has_character_flag = trade_opportunity_cooldown_silk_road_network
		}
		merchant_is_silk_road_participant_trigger = yes
		any_trade_route = {
			has_trader = yes
			NOR = {
				has_trade_route_modifier = gdp_silk_road_network_trade_opportunity_success_modifier
				has_trade_route_modifier = gdp_silk_road_network_trade_opportunity_success_critical_modifier
				has_trade_route_modifier = gdp_silk_road_network_trade_opportunity_failure_modifier
				has_trade_route_modifier = gdp_silk_road_caravan_trade_opportunity_success_modifier
				has_trade_route_modifier = gdp_silk_road_caravan_trade_opportunity_success_critical_modifier
				has_trade_route_modifier = gdp_silk_road_caravan_trade_opportunity_failure_modifier
			}
			any_trade_market_on_route = {
				gdp_trade_opportunity_market_on_route_destination_available_trigger = { MERCHANT = root }
				is_target_in_global_variable_list = {
					name = silk_road_markets
					target = this
				}
			}
		}
	}

	valid_to_keep = {
		valid_trade_opportunity_to_continue_trigger = yes
	}

	valid_to_accept = {
		valid_merchant_contract_basic_trigger = yes
		valid_to_accept_trade_opportunity_trigger = yes
	}

	can_accept_while_traveling = {
		can_accept_trade_opportunity_while_traveling_trigger = yes
	}

	valid_to_continue = {
		valid_trade_opportunity_to_continue_trigger = yes
		task_contract_taker = { merchant_is_silk_road_participant_trigger = yes }
		var:task_contract_route ?= {
			trader ?= root.task_contract_employer
			any_trade_market_on_route = {
				trade_market_owner.title_province = root.task_contract_location
				is_target_in_global_variable_list = {
					name = silk_road_markets
					target = this
				}
			}
		}
	}

	on_create = {
		scope:contract.task_contract_taker ?= { # Set a cooldown
			add_character_flag = {
				flag = trade_opportunity_cooldown_silk_road_network
				years = 10
			}
		}
	}

	on_accepted = {
		trade_opportunity_on_accepted_effect = yes
	}

	on_completed = {
		merchant_trade_opportunity_travel_complete_or_invalidated_effect = yes # required to ensure travel does not get stuck
	}

	on_invalidated = {
		merchant_trade_opportunity_contract_invalidated_message_effect = yes
		merchant_trade_opportunity_travel_complete_or_invalidated_effect = yes # required to ensure travel does not get stuck
	}

	task_contract_reward = {
		success_standard = {
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract"
					add_prestige = {
						value = minor_prestige_gain
						multiply = trade_opportunity_contract_tier_value
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_value
					trade_opportunities_competitive_pressure_success_effect = yes
					merchant_binding_contracts_trade_opportunity_success_effect = yes
				}
				var:task_contract_route ?= {
					add_trade_route_modifier = {
						modifier = gdp_silk_road_network_trade_opportunity_success_modifier
						years = 10
					}
				}
			}
		}
		success_critical = {
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract"
					add_prestige = {
						value = medium_prestige_gain
						multiply = trade_opportunity_contract_tier_value
						multiply = merchant_contract_prestige_reward_mult_value
					}
					add_short_term_gold = merchant_contract_gold_reward_critical_value
					trade_opportunities_competitive_pressure_critical_success_effect = yes
					merchant_binding_contracts_trade_opportunity_success_effect = yes
					merchant_port_factors_trade_opportunity_success_effect = yes
				}
				var:task_contract_route ?= {
					add_trade_route_modifier = {
						modifier = gdp_silk_road_network_trade_opportunity_success_critical_modifier
						years = 10
					}
				}
			}
		}
		failure_standard = {
			positive = no
			should_print_on_complete = yes
			effect = {
				save_scope_as = task_contract
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract"
					merchant_contract_failure_expertise_effect = yes
					trade_opportunities_competitive_pressure_failure_effect = yes
				}
				var:task_contract_route ?= {
					add_trade_route_modifier = {
						modifier = gdp_silk_road_network_trade_opportunity_failure_modifier
						years = 10
					}
				}
			}
		}
	}
}
