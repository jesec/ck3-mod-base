### Negotiating with [Ruler]
# Small opinion gain with the local county holder.
# Requires: local holder must be alive and not at war with the route owner.
gdp_trader_negotiating_with_ruler_scheme = {
	skill = diplomacy
	desc = gdp_trader_negotiating_with_ruler_scheme_desc_general
	icon = merchant
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = nothing
	category = personal

	speed_per_skill_point = -1
	uses_resistance = no
	base_progress_goal = 120
	base_maximum_success = 75
	minimum_success = 5

	base_success_chance = {
		base = 0
		gdp_trader_scheme_success_chance_modifier = { SKILL = DIPLOMACY }
	}

	allow = {
		gdp_trader_is_valid_for_scheme_trigger = yes
		location.county.holder = {
			is_alive = yes
			NOR = {
				is_at_war_with = root.assigned_trade_route.trade_route_owner
				this = root.assigned_trade_route.trade_route_owner
			}
		}
	}
	valid = {
		gdp_trader_is_valid_for_scheme_trigger = yes
	}

	on_phase_completed = {
		gdp_trader_scheme_setup_effect = yes

		if = {
			limit = { exists = scope:gdp_trader_scheme_success }
			scope:route_owner = {
				send_interface_message = {
					type = msg_trader_activity_success
					title = gdp_trader_negotiating_with_ruler_scheme_success_title
					desc = gdp_trader_negotiating_with_ruler_scheme_success_desc
					left_icon = scope:trader
					scope:trader.location.county.holder = {
						add_opinion = {
							modifier = merchant_polished_introductions_opinion
							target = scope:route_owner
							years = 5
						}
					}
				}
			}
		}
		gdp_trader_scheme_complete_effect = yes
	}

	on_invalidated = {
		gdp_trader_scheme_invalidated_effect = yes
	}
}

### Entertaining Merchants
# Build a friendship with one other merchant company head at this market.
# Requires: at least one other AI merchant with opinion >= 25 toward the route owner active in this market.
gdp_trader_entertaining_merchants_scheme = {
	skill = diplomacy
	desc = gdp_trader_entertaining_merchants_scheme_desc_general
	icon = merchant
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = nothing
	category = personal

	speed_per_skill_point = -1
	uses_resistance = no
	base_progress_goal = 120
	base_maximum_success = 75
	minimum_success = 5

	base_success_chance = {
		base = 0
		gdp_trader_scheme_success_chance_modifier = { SKILL = DIPLOMACY }
	}

	allow = {
		gdp_trader_is_valid_for_scheme_trigger = yes
		location.barony.trade_market = {
			any_trade_route_in_market = {
				trade_route_owner = {
					is_ai = yes
					government_has_flag = government_is_merchant
					this != root.assigned_trade_route.trade_route_owner
					opinion = {
						target = root.assigned_trade_route.trade_route_owner
						value >= 25
					}
				}
			}
		}
	}
	valid = {
		gdp_trader_is_valid_for_scheme_trigger = yes
	}

	on_phase_completed = {
		gdp_trader_scheme_setup_effect = yes

		if = {
			limit = { exists = scope:gdp_trader_scheme_success }
			scope:route_owner = {
				send_interface_message = {
					type = msg_trader_activity_success
					title = gdp_trader_entertaining_merchants_scheme_success_title
					desc = gdp_trader_entertaining_merchants_scheme_success_desc
					left_icon = scope:trader
					scope:trader.location.barony.trade_market = {
						random_trade_route_in_market = {
							limit = {
								trade_route_owner = {
									is_ai = yes
									government_has_flag = government_is_merchant
									this != scope:route_owner
									opinion = {
										target = scope:route_owner
										value >= 25
									}
								}
							}
							trade_route_owner = {
								add_opinion = {
									modifier = merchant_polished_introductions_opinion
									target = scope:route_owner
									years = 5
								}
							}
						}
					}
				}
			}
		}
		gdp_trader_scheme_complete_effect = yes
	}

	on_invalidated = {
		gdp_trader_scheme_invalidated_effect = yes
	}
}

### Hiring Local Talent
# Chance to recruit a local character as a company associate.
# Requires: route owner must have fewer than 10 courtiers.
gdp_trader_hiring_local_talent_scheme = {
	skill = diplomacy
	desc = gdp_trader_hiring_local_talent_scheme_desc_general
	icon = merchant
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = nothing
	category = personal

	speed_per_skill_point = -1
	uses_resistance = no
	base_progress_goal = 120
	base_maximum_success = 75
	minimum_success = 5

	base_success_chance = {
		base = 0
		gdp_trader_scheme_success_chance_modifier = { SKILL = DIPLOMACY }
	}

	allow = {
		gdp_trader_is_valid_for_scheme_trigger = yes
		assigned_trade_route.trade_route_owner = {
			any_courtier = { count < 10 always = yes }
		}
	}
	valid = {
		gdp_trader_is_valid_for_scheme_trigger = yes
		assigned_trade_route.trade_route_owner = {
			any_courtier = { count < 10 always = yes }
		}
	}

	on_phase_completed = {
		gdp_trader_scheme_setup_effect = yes

		if = {
			limit = { exists = scope:gdp_trader_scheme_success }
			# Recruit first, then notify
			random_pool_character = {
				province = scope:trader.location
				limit = {
					is_adult = yes
					is_imprisoned = no
					NOR = {
						has_trait = incapable
						is_ruler = yes
					}
				}
				save_scope_as = hired_character
				add_courtier = scope:route_owner
			}
			scope:route_owner = {
				send_interface_message = {
					type = msg_trader_activity_success
					title = gdp_trader_hiring_local_talent_scheme_success_title
					desc = gdp_trader_hiring_local_talent_scheme_success_desc
					left_icon = scope:trader
					right_icon = scope:hired_character
				}
			}
		}
		gdp_trader_scheme_complete_effect = yes
	}

	on_invalidated = {
		gdp_trader_scheme_invalidated_effect = yes
	}
}
