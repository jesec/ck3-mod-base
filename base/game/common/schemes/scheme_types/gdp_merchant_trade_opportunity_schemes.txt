@merchant_base_trade_opportunity_phases_per_agent_charge = 1

# INFO
# For laamp contracts that are broadly applicable: that is, they should be able to trigger virtually anywhere and with a minimum of triggers.
# The idea is that any given laamp led by a character skilled in any one thing should always be viable — maybe through crime, maybe by demeaning themselves, but they should always have some contracts around to work with.

# Challenge Competitor
merchant_challenge_competitor_trade_opportunity_scheme = {
	# Basic Setup
	skill = stewardship
	desc = merchant_challenge_competitor_trade_opportunity_scheme_desc_general
	success_desc = merchant_challenge_competitor_trade_opportunity_scheme_desc_success
	icon = merchant
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = title
	category = contract

	# Parameters
	speed_per_skill_point = t2_spsp_owner_value
	base_progress_goal = t3_base_phase_length_value # Trade opportunities should be snappy
	base_maximum_success = t1_base_max_success_value
	phases_per_agent_charge = @merchant_base_trade_opportunity_phases_per_agent_charge
	success_chance_growth_per_skill_point = merchant_trade_opportunity_scgpsp_value
	spymaster_speed_per_skill_point = 0
	uses_resistance = no

	# Core Triggers
	valid = {
		scope:owner = {
			is_landless_merchant = yes
		}
		scope:target.holder ?= {
			gdp_is_merchant_or_unlanded_patrician_trigger = yes
		}
	}

	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_laamp_contract_modifier = yes # close enough
	}
	valid_agent = { is_valid_agent_standard_trigger = yes }
	agent_groups_owner_perspective = { courtiers }

	# Base Chances
	base_success_chance = {
		base = 0
		scheme_type_skill_success_chance_modifier = { SKILL = STEWARDSHIP }
		merchant_trade_opportunity_scheme_basic_success_chance_modifier = yes
	}
	
	# On Actions
	on_start = {
		trade_opportunity_scheme_scopes_effect = yes
		# Try to guarantee an event in the first six months.
		scope:owner = {
			trigger_event = {
				on_action = merchant_challenge_competitor_trade_opportunity_scheme_on_going_events
				months = { 1 3 }
			}
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		merchant_base_challenge_competitor_trade_opportunity_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { merchant_base_challenge_competitor_trade_opportunity_scheme_prep_effect = yes }
	on_push_for_advantages = {
		scheme_owner = {
			trigger_event = { on_action = trade_opportunity_push_for_advantages_events }
		}
	}
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			merchant_base_challenge_competitor_trade_opportunity_scheme_prep_effect = yes
		}
	}
	on_monthly = { fire_generic_trade_opportunity_scheme_on_going_events_effect = yes }
	on_invalidated = {
		save_scope_as = scheme
		scheme_owner = {
			send_interface_message = {
				title = scheme_invalidated_title
				type = trade_opportunity_scheme_invalidated_message
				right_icon = scope:scheme.scheme_target_title.holder
				show_as_tooltip = {
					scope:scheme = { end_scheme = yes }
				}
			}
		}
	}
}
