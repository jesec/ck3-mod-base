		}
		trigger_else = { always = no }
	}

	valid = {
		scope:target = {

			NOR = {
				any_character_war = {
					using_cb = displace_guelph_ghibelline_ruler_cb
					primary_defender = scope:target
				}
				any_character_war = {
					using_cb = reinstate_guelph_ghibelline_ruler_cb
					primary_defender = scope:target
				}
			}
		}
		trigger_if = {
			limit = { is_guelph_trigger = yes }
			scope:target = { is_guelph_leader_trigger = yes }
		}
		trigger_else_if = {
			limit = { is_ghibelline_trigger = yes }
			scope:target = { is_ghibelline_leader_trigger = yes }
		}
		trigger_else = { always = no }
	}

	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_hostile_general_suite_modifier = yes
	}
	valid_agent = {	
		is_valid_agent_standard_trigger = yes #Is adult, is NOT imprisoned
		#Don't help the other side
		NOR = {
			AND = {
				scope:owner = { is_ghibelline_trigger = yes }
				is_guelph_trigger = yes
			}
			AND = {
				scope:owner = { is_guelph_trigger = yes }
				is_ghibelline_trigger = yes
			}
		}
	} 
	agent_groups_target_character_perspective = { peer_vassals vassals scripted_relations family situation_participant_group }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_diplomacy_value
		add = agent_groups_target_character_perspective_value
		add = odds_gg_is_kaiser_negative_value
		add = odds_gg_is_pope_negative_value
		add = odds_gg_claim_leadership_scheme_misc_value
		min = 0
	}

	# Base Chances
	base_success_chance = {

		}

		#Prestige level compared to the current leader
		modifier = {
			add = gg_claim_leadership_prestige_level_difference_value
			desc = SCHEME_GG_PRESTIGE_LEVEL_COMPARED_TO_LEADER
			gg_claim_leadership_prestige_level_difference_value > 0
		}

		#Weakened leader
		modifier = {
			add = 25
			desc = OVERTHROW_GOVERNMENT_TARGET_EXCOMMUNICATED
			scope:target = { has_trait = excommunicated }
		}
		modifier = {
			add = 25
			desc = ELOPE_SUCCESS_IMPRISONED
			scope:target = { is_imprisoned = yes }
		}
		modifier = {
			add = 10
			desc = SCHEME_GG_TARGET_UNCROWNED
			scope:target = { has_realm_law = uncrowned }
		}

		#Rank tier difference (landed target/target whose liege doesn't care)
		modifier = { #3 or more higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 25
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value >= 3
				}
			}
		}
		modifier = { #2 higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 10
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value = 2
				}
			}
		}
		modifier = { #1 higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 5
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value = 1
				}
			}
		}
		modifier = { #1 lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -10
			desc = "LOWER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value = -1
				}
			}
		}
		modifier = { #2 lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -25
			desc = "LOWER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value = -2
				}
			}
		}
		modifier = { #3 or less lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -50
			desc = "LOWER_RANK_THAN_SCHEME_TARGET"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = no
			}
			scope:owner = {
				tier_difference = {
					target = scope:target
					value <= -3
				}
			}
		}
		#Rank tier difference (unlanded character)
		modifier = { #3 or more higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 10
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value >= 3
				}
			}
		}
		modifier = { #2 higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 5
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value = 2
				}
			}
		}
		modifier = { #1 higher rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = 2
			desc = "HIGHER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
				NOT = {
					is_theocratic_lessee = yes
				}
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value = 1
				}
			}
		}
		modifier = { #1 lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -5
			desc = "LOWER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value = -1
				}
			}
		}
		modifier = { #2 lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -10
			desc = "LOWER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value = -2
				}
			}
		}
		modifier = { #3 or less lower rank
			trigger = { personal_scheme_should_not_evaluate_tier_differences_trigger = yes }
			add = -15
			desc = "LOWER_RANK_THAN_SCHEME_TARGET_LIEGE"
			scope:target = {
				personal_scheme_success_compare_target_liege_tier_trigger = yes
			}
			scope:owner = {
				tier_difference = {
					target = scope:target.liege
					value <= -3
				}
			}
		}

		#Positive
		scheme_type_skill_success_chance_modifier = { SKILL = DIPLOMACY }
		diplomacy_scheme_base_chance_modifier = yes
		political_scheme_base_chance_modifier = yes
		house_feud_hostile_scheme_success_modifier = yes

		#Negative
		target_is_kaiser_modifier = yes
		target_is_pope_modifier = yes

		# Countermeasures.
		apply_political_scheme_success_chance_adjustments_modifier = yes
	}
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	# On Actions
	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:political
		}
		# Fallback agents — a balanced config if you start the scheme from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_diplomat
			add_agent_slot = agent_diplomat
			add_agent_slot = agent_scribe
			add_agent_slot = agent_cleric
			add_agent_slot = agent_cleric
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		gg_claim_leadership_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { gg_claim_leadership_scheme_prep_effect = yes }
	on_monthly = {
		hostile_scheme_monthly_discovery_chance_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}
			scheme_owner = {
				trigger_event = {
					on_action = gg_claim_leadership_ongoing
					days = { 1 30 }
				}
			}
		}
	}
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			gg_claim_leadership_scheme_prep_effect = yes
			purge_ai_scheme_slots_effect = yes #Leaving this in, for now
		}
	}

	## Invalidation
	on_invalidated = {
		scheme_target_character = {
			save_scope_as = target
		}
		scheme_owner = {
			save_scope_as = owner
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = gg_claim_leadership_invalidated_title
				left_icon = scope:target
				if = { # Invalidation due to death
					limit = {
						scope:target = { is_alive = no }
						NOT = { block_death_event_trigger = { DEAD = scope:target } }
					}
					custom_tooltip = scheme_target_died
				}
				else_if = { # Diplo range: is it relevant here?
					limit = {
						scope:target = { 
							NOT = { in_diplomatic_range = scope:owner } 
						}
					}
					custom_tooltip = scheme_target_not_in_diplomatic_range
				}
				else_if = { # The GG Situation stopped existing!
					limit = {
						NOT = { exists = situation:guelph_ghibelline }
					}
					custom_tooltip = scheme_gg_situation_ended
				}
				else_if = { # You're no longer aligned!
					limit = {
						scope:owner = {
							is_guelph_trigger = no
							is_ghibelline_trigger = no
						}
					}
					custom_tooltip = scheme_owner_not_aligned
				}
				else_if = { # They're no longer aligned!
					limit = {
						scope:target = {
							is_guelph_trigger = no
							is_ghibelline_trigger = no
						}
					}
					custom_tooltip = scheme_target_not_aligned
				}
				else_if = { # You're not part of the GG Situation anymore!
					limit = {
						situation:guelph_ghibelline ?= {
							NOT = { 
								any_situation_participant = { this = scope:owner } 
							}
						}
					}
					custom_tooltip = scheme_owner_not_in_gg_situation
				}
				else_if = { # They're not part of the GG Situation anymore!
					limit = {
						situation:guelph_ghibelline ?= {
							NOT = { 
								any_situation_participant = { this = scope:target } 
							}
						}
					}
					custom_tooltip = scheme_target_not_in_gg_situation
				}
				else_if = { # They actually don't own anything anymore
					limit = {
						scope:target = { is_landless_adventurer = yes }
					}
					custom_tooltip = scheme_target_is_titleless
				}
				else_if = { #This character is currently the target of a Displacement War
					limit = {
						scope:target = { 
							any_character_war = {
								using_cb = displace_guelph_ghibelline_ruler_cb
								primary_defender = scope:target
							}
						}
					}
					custom_tooltip = scheme_target_might_get_displaced
				}
				else_if = { #This character is currently the target of a Reinstatement War
					limit = {
						scope:target = { 
							any_character_war = {
								using_cb = reinstate_guelph_ghibelline_ruler_cb
								primary_defender = scope:target
							}
						}
					}
					custom_tooltip = scheme_target_might_get_reinstated
				}
				else_if = { #You're already the leader of that side
					limit = {
						scope:owner = {
							OR = {
								AND = {
									is_guelph_trigger = yes
									is_guelph_leader_trigger = yes
								}
								AND = {
									is_ghibelline_trigger = yes
									is_ghibelline_leader_trigger = yes
								}
							}	
						}
					}
					custom_tooltip = scheme_owner_is_already_leader
				}
				else_if = { #They're no longer the leader of your side
					limit = {
						scope:owner = {
							OR = {
								AND = {
									is_guelph_trigger = yes
									scope:target = { is_guelph_leader_trigger = no }
								}
								AND = {
									is_ghibelline_trigger = yes
									scope:target = { is_ghibelline_leader_trigger = no }
								}
							}
						}
					}
					custom_tooltip = scheme_target_no_longer_gg_leader
				}
				else = {
					custom_tooltip = scheme_cancelled
				}
			}
		}
	}
}
