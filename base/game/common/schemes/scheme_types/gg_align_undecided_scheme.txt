	# On Actions
	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:indirect
		}
		# Fallback agents — a balanced config if you start the scheme from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_diplomat
			add_agent_slot = agent_diplomat
			add_agent_slot = agent_scribe
			add_agent_slot = agent_cleric
			add_agent_slot = agent_cleric
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		gg_align_undecided_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { gg_align_undecided_scheme_prep_effect = yes }
	on_monthly = {
		scheme_owner = {
			trigger_event = {
				on_action = gg_align_undecided_ongoing
				days = { 1 30 }
			}
		}
	}
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			gg_align_undecided_scheme_prep_effect = yes
			purge_ai_scheme_slots_effect = yes #Leaving this in, for now
		}
	}

	## Invalidation
	on_invalidated = {
		scheme_target_character = {
			save_scope_as = target
		}
		scheme_owner = {
			save_scope_as = owner
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = gg_align_undecided_invalidated_title
				left_icon = scope:target
				if = { # Invalidation due to death
					limit = {
						scope:target = { is_alive = no }
						NOT = { block_death_event_trigger = { DEAD = scope:target } }
					}
					custom_tooltip = scheme_target_died
				}
				else_if = { # Diplo range: is it relevant here?
					limit = {
						scope:target = { 
							NOT = { in_diplomatic_range = scope:owner } 
						}
					}
					custom_tooltip = scheme_target_not_in_diplomatic_range
				}
				else_if = { # The GG Situation stopped existing!
					limit = {
						NOT = { exists = situation:guelph_ghibelline }
					}
					custom_tooltip = scheme_gg_situation_ended
				}
				else_if = { # You're no longer aligned!
					limit = {
						scope:owner = {
							is_guelph_trigger = no
							is_ghibelline_trigger = no
						}
					}
					custom_tooltip = scheme_owner_not_aligned
				}
				else_if = { # They're not undecided!
					limit = {
						scope:target = { 
							OR = {
								is_guelph_trigger = yes
								is_ghibelline_trigger = yes
							}
						}
					}
					custom_tooltip = scheme_target_not_undecided
				}
				else_if = { # You're not part of the GG Situation anymore!
					limit = {
						situation:guelph_ghibelline ?= {
							NOT = { 
								any_situation_participant = { this = scope:owner } 
							}
						}
					}
					custom_tooltip = scheme_owner_not_in_gg_situation
				}
				else_if = { # They're not part of the GG Situation anymore!
					limit = {
						situation:guelph_ghibelline ?= {
							NOT = { 
								any_situation_participant = { this = scope:target } 
							}
						}
					}
					custom_tooltip = scheme_target_not_in_gg_situation
				}
				else_if = { # They actually don't own anything anymore
					limit = {
						scope:target = { is_landless_adventurer = yes }
					}
					custom_tooltip = scheme_target_is_titleless
				}
				else = {
					custom_tooltip = scheme_cancelled
				}
			}
		}
