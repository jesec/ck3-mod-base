@merchant_base_patron_contract_phases_per_agent_charge = 1

# Hobnob with patron
merchant_patron_hobnob_patron_contract_scheme = {
	# Basic Setup
	skill = diplomacy
	desc = merchant_patron_hobnob_patron_contract_scheme_desc_general
	success_desc = merchant_patron_hobnob_patron_contract_scheme_desc_success
	icon = diplomacy
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	target_type = character
	category = contract
	
	# Parameters
	speed_per_skill_point = t2_spsp_owner_value
	base_progress_goal = t3_base_phase_length_value # Should be snappy
	base_maximum_success = t3_base_max_success_value # # Should be snappy
	phases_per_agent_charge = @merchant_base_patron_contract_phases_per_agent_charge
	success_chance_growth_per_skill_point = laamp_contract_scgpsp_value
	spymaster_speed_per_skill_point = 0
	uses_resistance = no
	
	# Core Triggers
	valid = {
		scope:owner = {
			is_landless_merchant = yes
			suzerain ?= scope:scheme.scheme_target_character
		}
	}
	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_laamp_contract_modifier = yes
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_personal_inoffensive_modifier = yes
	}
	valid_agent = {
		exists = scope:owner
		is_valid_agent_standard_trigger = yes
	}
	agent_groups_owner_perspective = { courtiers }
	agent_groups_target_character_perspective = { courtiers guests }

	# Base Chances
	base_success_chance = {
		base = 0
		scheme_type_skill_success_chance_modifier = { SKILL = DIPLOMACY }
		merchant_trade_opportunity_scheme_basic_success_chance_modifier = yes # close enough
	}
	
	# On Actions
	on_start = {
		patron_contract_scheme_scopes_effect = yes
		# Try to guarantee an event in the first six months.
		scheme_owner = {
			trigger_event = {
				on_action = merchant_patron_hobnob_patron_contract_scheme_ongoing_events
				months = { 1 3 }
			}
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		merchant_base_patron_hobnob_patron_contract_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { merchant_base_patron_hobnob_patron_contract_scheme_prep_effect = yes }
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			merchant_base_patron_hobnob_patron_contract_scheme_prep_effect = yes
		}
	}
	on_monthly = { fire_generic_patron_contract_scheme_on_going_events_effect = yes }
	on_invalidated = {
		save_scope_as = scheme
		scheme_owner = {
			send_interface_message = {
				title = scheme_invalidated_title
				type = diplo_contract_scheme_invalidated_message
				right_icon = scope:scheme.scheme_target_character
				show_as_tooltip = {
					scope:scheme = { end_scheme = yes }
				}
			}
		}
	}
}
