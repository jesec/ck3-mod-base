##################################################
# Trade Schemes
# Self-directed schemes for Merchants and Republics developing their own trade network.
##################################################

############################
# Improve Trade Routes
# Three self-targeted schemes, mechanically identical apart from the governing
# skill and the modifier applied on success. Trade Routes cannot be scheme
# targets, so the chosen route is carried as var:focused_trade_route.
#
# Losing that route is not a valid check. on_trade_route_destroyed ends the
# scheme and sends its own toast. Other invalidations are silent.
############################

gdp_improve_trade_capacity = {
	# Basic Setup
	skill = stewardship
	desc = gdp_improve_trade_capacity_desc_general
	success_desc = "GDP_IMPROVE_TRADE_CAPACITY_SUCCESS_DESC"
	icon = merchant
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = character
	category = trade
	is_secret = no
	uses_resistance = no
	hide_target_name = yes

	# Parameters
	speed_per_skill_point = t1_spsp_owner_value
	spymaster_speed_per_skill_point = t1_spsp_owner_spy_value
	base_progress_goal = gdp_trade_scheme_base_phase_length_value
	base_maximum_success = t1_base_max_success_value
	minimum_success = 5
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t1_scgpsp_value

	# Core Triggers
	allow = {
		gdp_improve_trade_scheme_valid_trigger = yes
	}
	valid = {
		scope:owner = { gdp_improve_trade_scheme_valid_trigger = yes }
	}

	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
	}
	valid_agent = { is_valid_agent_standard_trigger = yes }
	agent_groups_owner_perspective = { courtiers guests scripted_relations councillors vassals }

	odds_prediction = {
		add = base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_stewardship_value
		add = agent_groups_owner_perspective_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 0
		scheme_type_skill_success_chance_modifier = { SKILL = STEWARDSHIP }

		# The more of your network is already running, the more there is to improve on.
		modifier = {
			desc = GDP_IMPROVE_TRADE_ROUTES_STAFFED_ROUTES_DESC
			add = {
				value = 0
				scope:owner = {
					every_trade_route = {
						limit = { has_trader = yes }
						add = 5
					}
				}
			}
		}
		# A mature network is easier to tune than one you only just laid down.
		modifier = {
			desc = GDP_IMPROVE_TRADE_ROUTES_ESTABLISHED_ROUTES_DESC
			add = {
				value = 0
				scope:owner = {
					every_trade_route = {
						limit = { age_in_years >= 5 }
						add = 3
					}
				}
			}
		}
	}

	# On Actions
	on_start = {
		# Fallback agents, a balanced config if the scheme is started from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_broker
			add_agent_slot = agent_appraiser
			add_agent_slot = agent_bookkeeper
			add_agent_slot = agent_overseer
			add_agent_slot = agent_supplier
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		gdp_improve_trade_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { gdp_improve_trade_scheme_prep_effect = yes }
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			gdp_improve_trade_scheme_prep_effect = yes
			purge_ai_scheme_slots_effect = yes
		}
	}
	on_monthly = { fire_generic_contract_scheme_on_going_events_effect = yes }

	## Invalidation
	on_invalidated = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		if = { # No longer a trade government
			limit = {
				scope:owner = {
					NOT = { gdp_is_trade_ruler_trigger = yes }
				}
			}
			scope:owner = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = gdp_improve_trade_capacity_invalidated_title
					left_icon = scope:owner
					custom_tooltip = gdp_improve_trade_routes_owner_not_trade_ruler
				}
			}
		}
		else_if = { # Network gone
			limit = {
				scope:owner = {
					NOT = { any_trade_route = { count >= 1 } }
				}
			}
			scope:owner = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = gdp_improve_trade_capacity_invalidated_title
					left_icon = scope:owner
					custom_tooltip = gdp_improve_trade_routes_owner_has_no_routes
				}
			}
		}
	}
}

############################
# Disrupt Trade Routes
# On success one of the target's Trade Routes loses either Capacity or
# Competitiveness. A route sharing a market with the owner is always preferred.
############################

gdp_disrupt_trade_routes = {
	# Basic Setup
	skill = intrigue
	desc = gdp_disrupt_trade_routes_desc_general
	success_desc = "GDP_DISRUPT_TRADE_ROUTES_SUCCESS_DESC"
	discovery_desc = "GDP_DISRUPT_TRADE_ROUTES_DISCOVERY_DESC"
	icon = icon_raid_estate
	illustration = "gfx/interface/illustrations/event_scenes/gdp_italian_market.dds"
	target_type = character
	category = hostile
	is_secret = yes

	# Parameters
	speed_per_skill_point = t2_spsp_owner_value
	speed_per_target_skill_point = t2_spsp_target_value
	spymaster_speed_per_skill_point = t2_spsp_owner_spy_value
	target_spymaster_speed_per_skill_point = 0
	tier_speed = 1
	base_progress_goal = t3_base_phase_length_value # One phase resolves the whole scheme, so keep it snappy
	base_maximum_success = t2_base_max_success_value
	minimum_success = 5
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t2_scgpsp_value
	maximum_secrecy = 95
	maximum_breaches = 5
	# Without this the monthly expose check has no baseline and countermeasures skip their secrecy penalty
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	# Core Triggers
	allow = {
		gdp_disrupt_trade_routes_scheme_owner_valid_trigger = yes
		scope:target ?= {
			gdp_disrupt_trade_routes_scheme_target_valid_trigger = { SCHEME_OWNER = root }
		}
	}
	valid = {
		scope:owner = { gdp_disrupt_trade_routes_scheme_owner_valid_trigger = yes }
		scope:target ?= {
			gdp_disrupt_trade_routes_scheme_target_valid_trigger = { SCHEME_OWNER = scope:owner }
		}
	}

	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
	}
	valid_agent = { is_valid_agent_standard_trigger = yes }
	agent_groups_owner_perspective = { courtiers guests scripted_relations }
	agent_groups_target_character_perspective = { courtiers guests scripted_relations }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = agent_groups_owner_perspective_value
		add = agent_groups_target_character_perspective_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 0
		modifier = {
			desc = SCHEME_YOUR_INTRIGUE
			add = {
				value = 0
				if = {
					limit = { exists = scope:owner }
					add = {
						value = scope:owner.intrigue
						multiply = 0.5
					}
				}
			}
		}
		modifier = {
			desc = SCHEME_TARGET_INTRIGUE
			add = {
				value = 0
				if = {
					limit = { exists = scope:target }
					add = {
						value = scope:target.intrigue
						multiply = -0.5
					}
				}
			}
		}
		modifier = {
			desc = SCHEME_YOUR_STEWARDSHIP
			add = {
				value = 0
				if = {
					limit = { exists = scope:owner }
					add = {
						value = scope:owner.stewardship
						multiply = 0.5
					}
				}
			}
		}
		modifier = {
			desc = SCHEME_TARGET_STEWARDSHIP
			add = {
				value = 0
				if = {
					limit = { exists = scope:target }
					add = {
						value = scope:target.stewardship
						multiply = -0.5
					}
				}
			}
		}
		modifier = {
			desc = GDP_DISRUPT_TRADE_ROUTES_SHARED_MARKET_DESC
			exists = scope:owner
			exists = scope:target
			scope:target = {
				any_trade_route = {
					gdp_trade_route_shares_market_with_trigger = { CHARACTER = scope:owner }
				}
			}
			add = 10
		}
		# Countermeasures
		apply_calculated_scheme_success_chance_adjustments_modifier = yes
	}

	# On Actions
	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:calculating
		}
		# Fallback agents, a balanced config if the scheme is started from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_thief
			add_agent_slot = agent_bookkeeper
			add_agent_slot = agent_infiltrator
			add_agent_slot = agent_overseer
			add_agent_slot = agent_lookout
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		gdp_disrupt_trade_routes_scheme_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click = { gdp_disrupt_trade_routes_scheme_prep_effect = yes }
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			gdp_disrupt_trade_routes_scheme_prep_effect = yes
			purge_ai_scheme_slots_effect = yes
		}
	}
	on_monthly = { fire_generic_contract_scheme_on_going_events_effect = yes }

	## Invalidation
	on_invalidated = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		if = { # Owner no longer runs profit routes
			limit = {
				scope:owner = {
					NOT = { gdp_is_trade_ruler_trigger = yes }
				}
			}
			scope:owner = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = gdp_disrupt_trade_routes_invalidated_title
					left_icon = scope:target
					custom_tooltip = gdp_disrupt_trade_routes_owner_not_trade_ruler
				}
			}
		}
		else_if = { # Target no longer runs profit routes
			limit = {
				scope:target ?= {
					NOT = { gdp_is_trade_ruler_trigger = yes }
				}
			}
			scope:owner = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = gdp_disrupt_trade_routes_invalidated_title
					left_icon = scope:target
					custom_tooltip = gdp_disrupt_trade_routes_target_not_trade_ruler
				}
			}
		}
		else_if = { # Target's network is gone
			limit = {
				scope:target ?= {
					NOT = { any_trade_route = { count >= 1 } }
				}
			}
			scope:owner = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = gdp_disrupt_trade_routes_invalidated_title
					left_icon = scope:target
					custom_tooltip = gdp_disrupt_trade_routes_target_has_no_routes
				}
			}
		}
	}
}
