republic_no_ambition = {
	show_in_main_hud = yes
	is_default = yes
	is_shown = {
		house_head ?= {
			government_has_flag = government_is_republic
			highest_held_title_tier >= tier_county
		}
		# Only show this if we don't have an ambition set
		NOR = {
			has_house_aspiration_parameter = republic_commerce
			has_house_aspiration_parameter = republic_aggrandizement
			has_house_aspiration_parameter = republic_communal
			has_house_aspiration_parameter = republic_signorial
			has_house_aspiration_parameter = republic_agitation
		}
	}

	cooldown = { months = 0 }

	level = {
		parameters = {
			republic_no_ambition = yes
		}
		ai_score = {
			value = 0
		}
	}

	illustration = "gfx/interface/icons/culture_pillars/ethos_communal.dds"
}
