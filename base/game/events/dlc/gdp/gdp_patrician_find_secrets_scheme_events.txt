namespace = gdp_patrician_find_secrets

gdp_patrician_find_secrets.0001 = {
	hidden = yes

	immediate = {
		scope:owner = {
			save_scope_as = councillor
			save_scope_as = councillor_liege
		}

		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = patrician_find_secrets_successful
				value = yes
			}
		}

		if = {
			limit = { exists = scope:patrician_find_secrets_successful }

			scope:target.house ?= {
				every_house_member = {
					limit = {
						is_alive = yes
						top_liege = scope:target.top_liege
						has_any_secrets = yes
					}
					every_secret = {
						limit = {
							OR = {
								spymaster_task_find_secrets_suitable_major_secret_trigger = yes
								spymaster_task_find_secrets_suitable_minor_secret_trigger = yes
							}
							NOT = { is_in_list = suitable_patrician_secrets }
						}
						add_to_temporary_list = suitable_patrician_secrets
					}
				}
			}

			if = {
				limit = {
					any_in_list = { list = suitable_patrician_secrets }
				}
				random_in_list = {
					list = suitable_patrician_secrets
					weight = {
						base = 1
						modifier = {
							add = 4
							spymaster_task_find_secrets_suitable_major_secret_trigger = yes
						}
						modifier = {
							add = 3
							secret_owner = scope:target
						}
					}
					save_scope_as = secret_to_reveal
					secret_owner = { save_scope_as = secret_holder }
				}
				scope:owner = { trigger_event = gdp_patrician_find_secrets.0002 }
			}
			else = {
				scope:owner = { trigger_event = gdp_patrician_find_secrets.0003 }
			}
		}
		else = {
			scope:owner = { trigger_event = gdp_patrician_find_secrets.0003 }
		}
	}
}

gdp_patrician_find_secrets.0002 = {
	type = character_event
	window = scheme_successful_event
	title = gdp_patrician_find_secrets.0002.t
	desc = gdp_patrician_find_secrets.0002.desc
	theme = intrigue

	left_portrait = {
		character = scope:secret_holder
		animation = worry
	}
	lower_right_portrait = {
		trigger = {
			scope:secret_holder != scope:target
		}
		character = scope:target
	}

	option = {
		name = gdp_patrician_find_secrets.0002.a
		scope:secret_to_reveal = {
			reveal_to_without_events_effect = {
				CHARACTER = root
			}
		}
		scope:scheme = { end_scheme = yes }
	}
}

gdp_patrician_find_secrets.0003 = {
	type = character_event
	window = scheme_failed_event
	title = gdp_patrician_find_secrets.0003.t
	desc = {
		triggered_desc = {
			trigger = { exists = scope:patrician_find_secrets_successful }
			desc = gdp_patrician_find_secrets.0003.desc.nothing_to_find
		}
		triggered_desc = {
			trigger = { NOT = { exists = scope:patrician_find_secrets_successful } }
			desc = gdp_patrician_find_secrets.0003.desc.outmatched
		}
	}
	theme = intrigue

	left_portrait = {
		character = root
		animation = disappointed
	}
	lower_right_portrait = scope:target

	immediate = {
		if = {
			limit = {
				NOT = { exists = scope:patrician_find_secrets_successful }
			}
			stress_and_fulfillment_impact = {
				base = minor_stress_impact_gain
				ambitious = minor_stress_impact_gain
				deceitful = minor_stress_impact_gain
			}
		}
	}

	option = {
		name = gdp_patrician_find_secrets.0003.a
		scope:scheme = { end_scheme = yes }
	}
}
