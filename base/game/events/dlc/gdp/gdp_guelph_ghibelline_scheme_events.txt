#Align/Switch Sides: Look! Isn't my side great!
gdp_guelph_ghibelline_scheme.1000 = {
	type = character_event
	title = gdp_guelph_ghibelline_scheme.1000.t
	desc = gdp_guelph_ghibelline_scheme.1000.desc
	theme = guelph_ghibelline
	override_background = { reference = market_crowd }
	left_portrait = {
		character = root
		animation = flirtation
	}
	right_portrait = {
		character = scope:target
		animation = contemplating
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	trigger = {
		NOT = {
			scope:scheme = { exists = var:had_gg_scheme_1000 }
		}
		scope:target = { is_available = yes }
		is_available = yes
		#Same location, or nearby at least
		OR = {
			location = scope:target.location
			location = { 
				any_neighboring_province = { this = scope:target.location }
			}
		}
	}
	immediate = {
		scope:scheme = {
			set_variable = had_gg_scheme_1000
		}
	}
	#Roll to get a better outcome
	option = {
		name = gdp_guelph_ghibelline_scheme.1000.a
		duel = {
			skills = { diplomacy intrigue }
			target = scope:target
			#Success
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				desc = gdp_guelph_ghibelline_scheme.1000.a.desc.success
				send_interface_toast = {
					type = event_toast_text_and_effect_good
					title = gdp_guelph_ghibelline_scheme.1000.t.success
					desc = gdp_guelph_ghibelline_scheme.1000.a.desc.success
					left_icon = root
					scope:scheme = {
						add_scheme_modifier = { type = scheme_clever_coaxing_modifier }
					}
				}
			}
			#Failure
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = 10
				}
				desc = gdp_guelph_ghibelline_scheme.1000.a.desc.failure
				send_interface_toast = {
					type = event_toast_text_and_effect_bad
					title = gdp_guelph_ghibelline_scheme.1000.t.failure
					desc = gdp_guelph_ghibelline_scheme.1000.a.desc.failure
					left_icon = root
					stress_and_fulfillment_impact = {
						base = medium_stress_impact_gain
					}
				}
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				diplomacy > scope:target.diplomacy
				intrigue > scope:target.intrigue
				add = 20
			}
		}
	}
	#Pay some resource for a guaranteed outcome
	option = {
		name = gdp_guelph_ghibelline_scheme.1000.b
		remove_short_term_gold = minor_gold_value
		scope:scheme = {
			add_scheme_modifier = { type = scheme_innocent_knickknack_modifier }
		}
		ai_chance = {
			base = 20
			modifier = {
				short_term_gold <= minor_gold_value
				factor = 0
			}
		}
	}
	#I'm good
	option = {
		name = gdp_guelph_ghibelline_scheme.1000.c
		ai_chance = {
			base = 25
		}
	}
}
