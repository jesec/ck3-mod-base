# GENERAL SCRIPTED TRIGGERS & EFFECTS

# Syncs stress lost from successfully completing a trade deal scheme.
scripted_effect merchant_base_patron_schemes_acknowledge_success_stress_effect = {
	stress_and_fulfillment_impact = { arrogant = minor_stress_impact_loss }
}

# Syncs stress gained from failing a trade deal scheme.
scripted_effect merchant_base_patron_schemes_acknowledge_failure_stress_effect = {
	if = {
		limit = { scope:task_contract_tier >= 2 }
		stress_and_fulfillment_impact = {
			base = miniscule_stress_impact_gain
			humble = minor_stress_impact_loss
			arrogant = minor_stress_impact_gain
		}
	}
}

#	Success event.
gdp_merchant_patron_contract.0002 = {
	type = character_event
	window = scheme_successful_event
	title = gdp_merchant_patron_contract.0002.t
	desc = gdp_merchant_patron_contract.0002.desc
	theme = diplomacy
	center_portrait = {
		character = root
		animation = hero_flex
	}
	right_portrait = {
		character = scope:patron
		animation = admiration
	}
	override_background = { reference = sitting_room }

	immediate = {
		merchant_base_patron_contract_schemes_set_up_outcome_immediate_effect = yes
		random_list = {
			67 = {
				scope:scheme.task_contract = { complete_task_contract = success_standard }
			}
			33 = {
				scope:scheme.task_contract = { complete_task_contract = success_critical }
			}
		}
		scope:patron = { add_stress = major_stress_loss }
		# And wrap things up.
		scope:scheme ?= { end_scheme = yes }
	}

	# Acknowledge.
	option = {
		name = gdp_merchant_patron_contract.0002.a
		flavor = merchant_base_patron_contract_schemes.0000.f.contract_concluded
		merchant_base_patron_schemes_acknowledge_success_stress_effect = yes
	}
}
