# Butter up a noble
gdp_schmooze_events.1000 = {
	type = activity_event
	title = gdp_schmooze_events.1000.t
	desc = gdp_schmooze_events.1000.desc
	theme = trade
	left_portrait = {
		character = root
		animation = admiration
	}
	right_portrait = {
		character = scope:schmoozee
		animation = war_over_tie
	}
	cooldown = { months = 1 }
	override_background = { reference = ep2_involved_activity }

	trigger = {
		has_activity_intent = schmooze_intent
		scope:activity = {
			any_attending_character = {
				is_landed = yes
				NOT = { government_has_flag = government_is_merchant }
				save_temporary_scope_as = schmoozee_temp
				NOT = {
					root = {
						is_target_in_variable_list = {
							name = schmoozed
							target = scope:schmoozee_temp
						}
					}
				}
			}
		}
	}

	immediate = {
		save_scope_as = schmoozer
		scope:activity = {
			random_attending_character = {
				limit = {
					is_landed = yes
					NOT = { government_has_flag = government_is_merchant }
					save_temporary_scope_as = schmoozee_temp
					NOT = {
						scope:schmoozer = {
							is_target_in_variable_list = {
								name = schmoozed
								target = scope:schmoozee_temp
							}
						}
					}
				}
				save_scope_as = schmoozee
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_merchant
				any_trade_route_with_goods_category_trigger = { CATEGORY = luxury_items }
			}
			save_random_trade_route_with_goods_category_effect = { CATEGORY = luxury_items }
			scope:goods_route ?= {
				random_trade_market_on_route = {
					limit = {
						NOT = { has_trade_market_modifier = activity_used_trade_market_luxury_items_modifier }
						trade_market_production_of_trade_good_need_category = {
							target = trade_good_category:trade_category_luxury_items
							value > 0
						}
					}
					alternative_limit = {
						trade_market_production_of_trade_good_need_category = {
							target = trade_good_category:trade_category_luxury_items
							value > 0
						}
					}
					alternative_limit = { always = yes }
					save_scope_as = goods_market
				}
			}
		}
	}

	option = { #I am an luxury merchant!
		name = gdp_schmooze_events.1000.d
		trigger = {
			custom_tooltip = {
				text = trade_category_luxury_items_trade_route_trigger
				exists = scope:goods_market
			}
		}
		reason = have_luxury_items_route
		show_as_unavailable = { government_has_flag = government_is_merchant }
		reverse_add_opinion = {
			target = scope:schmoozee
			modifier = impressed_opinion
			opinion = 20
		}
		scope:goods_market ?= {
			add_trade_market_modifier = {
				modifier = activity_used_trade_market_luxury_items_modifier
				years = 10
			}
		}
		gdp_log_schmooze_successful_effect = {
			CHAR = scope:schmoozer
			TARGET = scope:schmoozee
			MARKET = scope:province.barony.trade_market
		}
		add_to_variable_list = {
			name = schmoozed
			target = scope:schmoozee
			months = 6
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_trait = ambitious
			}
			ai_value_modifier = {
				ai_boldness = 0.5
			}
		}
	}

	option = {
		name = gdp_schmooze_events.1000.a
		trigger = { gold >= scope:schmoozee.minor_gold_value }
		show_as_unavailable = { gold < scope:schmoozee.minor_gold_value }
		pay_short_term_gold = {
			gold = scope:schmoozee.minor_gold_value
			target = scope:schmoozee
		}
		reverse_add_opinion = {
			target = scope:schmoozee
			modifier = impressed_opinion
			opinion = 20
		}
		gdp_log_schmooze_successful_effect = {
			CHAR = scope:schmoozer
			TARGET = scope:schmoozee
			MARKET = scope:province.barony.trade_market
		}
		add_to_variable_list = {
			name = schmoozed
			target = scope:schmoozee
			months = 6
		}
		ai_chance = {
			base = 25
			modifier = {
				gold < scope:schmoozee.medium_gold_value
				factor = 0
			}
			ai_value_modifier = {
				ai_greed = -0.5
			}
		}
	}

	option = {
		name = gdp_schmooze_events.1000.b
		duel = {
			skill = diplomacy
			value = decent_skill_rating
			10 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				min = 5
				desc = gdp_schmooze_events.1000.b.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = gdp_schmooze_events.1000.b.success
					right_icon = scope:schmoozee
					gdp_log_schmooze_successful_effect = {
						CHAR = scope:schmoozer
						TARGET = scope:schmoozee
						MARKET = scope:province.barony.trade_market
					}
					reverse_add_opinion = {
						target = scope:schmoozee
						modifier = impressed_opinion
						opinion = 10
					}
				}
			}
			10 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				min = 5
				desc = gdp_schmooze_events.1000.b.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = gdp_schmooze_events.1000.b.failure
					right_icon = scope:schmoozee
					add_prestige = miniscule_prestige_loss

			name = schmoozed

		this = root

	}

		any_character_in_market = { this = root }
	}
	save_temporary_scope_as = schmoozee_temp
	NOT = {
		root = {
			is_target_in_variable_list = {
				name = schmoozed
				target = scope:schmoozee_temp
			}
		}
	}
