hair_religious = {

	usage = game
	selection_behavior = weighted_random
	priority = 3

	## Orthodox ##

	# Not containing tonsures but a selection of other fitting hairstyles for Orthodox clergy. 

	pam_religious_orthodox = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = hairstyles
				template = pam_orthodox_hairstyles
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 130
				portrait_religious_orthodox_trigger = yes
				OR = {
					portrait_religious_devoted_trigger = yes
					portrait_religious_high_trigger = yes
					portrait_religious_high_no_poverty_trigger = yes
				}
			}
		}
	}

	## Catholic ##

	# Catholic tonsure for Western Christian men in religious life (monks, priests, military order members). 

	catholic_tonsure_basegame = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = hairstyles
				template = catholic_devoted_hairstyles # Basegame tonsure (male_hair_western_10)
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 50 # Kept under 100 so the auto-select-at-100 rule doesn't bypass weighted_random.
				exists = this
				is_female = no
				portrait_religious_western_christian_trigger = yes
				age >= 10
				OR = {
					portrait_religious_devoted_trigger = yes
					portrait_religious_high_trigger = yes
					portrait_religious_high_no_poverty_trigger = yes
					has_trait = order_member
					government_has_flag = government_is_monastic_holy_order
				}
			}
		}
	}

	catholic_tonsure_pam = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = hairstyles
				template = pam_catholic_hairstyles # PAM tonsure (m_hair_pam_christian_03)
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 50 # Matches catholic_tonsure_basegame so PAM owners roll 50/50 between the two.
				exists = this
				is_female = no
				has_pam_dlc_trigger = yes # Non-PAM owners fall through to the basegame tonsure only.
				portrait_religious_western_christian_trigger = yes
				age >= 10
				OR = {
					portrait_religious_devoted_trigger = yes
					portrait_religious_high_trigger = yes
					portrait_religious_high_no_poverty_trigger = yes
					has_trait = order_member
					government_has_flag = government_is_monastic_holy_order
				}
			}
		}
	}
}
