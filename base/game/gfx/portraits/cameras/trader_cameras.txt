# Trader task portrait pans. Same body framing as camera_body (radius/height/angle),
# only look_at L/R changes — pans the character in frame without orbiting.
# look_at = { L/R  U/D  F/B }: negative L/R shifts the character right in frame;
# positive L/R shifts them left. Tweak the first number per pose.

	camera = {
		position = cylindrical{ 335 -10 0 }
		position_node = {

			default = camera_torso_look_at
		}
